3//! Build-time-generated cell-model-IR descriptors embedded into the binary
4//! (ADR 0019 D7).
5//!
6//! `build.rs` runs the `liberty-to-cellir` converter as a library over the
7//! pinned vendored PDKs and writes each descriptor into `$OUT_DIR`; this module
8//! `include_str!`s those artifacts so a released binary carries them with no
9//! runtime `vendor/` dependency. The JSON is **not** checked in — it is a
10//! deterministic function of the pinned submodules (D7), regenerated whenever
11//! the vendored Liberty changes.
12//!
13//! In C3.1 this sits *alongside* the legacy per-PDK paths: the descriptors are
14//! available and loadable, and a run opts in via `--bundled-descriptor <name>`
15//! (with `--cell-descriptor <path>` remaining the explicit file override). The
16//! C3.2 selection work turns this into prefix-driven auto-selection, and C3.3
17//! makes it the default.
18
19use std::path::Path;
20
21use cell_model_ir::CellModelIr;
22
23/// GF180MCU 7-track descriptor, generated from
24/// `vendor/gf180mcu_fd_sc_mcu7t5v0` at the `tt_025C_5v00` corner.
25pub const GF180MCU_7T_JSON: &str =
26    include_str!(concat!(env!("OUT_DIR"), "/gf180mcu_7t.cellir.json"));
27
28/// GF180MCU 9-track descriptor, generated from
29/// `vendor/gf180mcu_fd_sc_mcu9t5v0` at the `tt_025C_5v00` corner.
30pub const GF180MCU_9T_JSON: &str =
31    include_str!(concat!(env!("OUT_DIR"), "/gf180mcu_9t.cellir.json"));
32
33/// A bundled descriptor's stable selector name and embedded JSON, paired so
34/// callers can list, match, or load them uniformly.
35pub struct BundledDescriptor {
36    /// Stable selector accepted by `--bundled-descriptor`.
37    pub name: &'static str,
38    /// The embedded JSON (the byte-for-byte build-time artifact).
39    pub json: &'static str,
40}
41
42/// All descriptors embedded at build time. Order is stable.
43pub const ALL: &[BundledDescriptor] = &[
44    BundledDescriptor {
45        name: "gf180mcu_7t",
46        json: GF180MCU_7T_JSON,
47    },
48    BundledDescriptor {
49        name: "gf180mcu_9t",
50        json: GF180MCU_9T_JSON,
51    },
52];
53
54/// Comma-separated list of valid `--bundled-descriptor` names, for error
55/// messages and `--help`.
56pub fn names() -> String {
57    ALL.iter().map(|d| d.name).collect::<Vec<_>>().join(", ")
58}
59
60/// Look up a bundled descriptor's raw JSON by selector name.
61pub fn json_by_name(name: &str) -> Option<&'static str> {
62    ALL.iter().find(|d| d.name == name).map(|d| d.json)
63}
64
65/// Parse a bundled descriptor by selector name into a [`CellModelIr`].
66///
67/// Panics if the embedded JSON fails to parse — that would be a build-time
68/// regression in the converter, not a user error.
69pub fn load(name: &str) -> Option<CellModelIr> {
70    json_by_name(name).map(|json| {
71        CellModelIr::from_json(json)
72            .unwrap_or_else(|e| panic!("embedded bundled descriptor '{name}' failed to parse: {e}"))
73    })
74}
75
76/// Resolve the cell-model-IR descriptor a run should consume, applying the
77/// ADR 0019 precedence: an explicit `--cell-descriptor <path>` file always
78/// wins over a bundled `--bundled-descriptor <name>` selection; absent both,
79/// `None` (the legacy per-PDK path).
80///
81/// Panics with an actionable message on a bad path or unknown bundled name.
82pub fn resolve(explicit: Option<&Path>, bundled: Option<&str>) -> Option<CellModelIr> {
83    if let Some(path) = explicit {
84        let ir = CellModelIr::read_from(path)
85            .unwrap_or_else(|e| panic!("loading --cell-descriptor {}: {e}", path.display()));
86        return Some(ir);
87    }
88    if let Some(name) = bundled {
89        let ir = load(name).unwrap_or_else(|| {
90            panic!(
91                "--bundled-descriptor '{name}' not found; available: {}",
92                names()
93            )
94        });
95        return Some(ir);
96    }
97    None
98}

100/// Auto-select a bundled descriptor by matching a netlist's cell-type names
101/// against each descriptor's declared D8 `library.prefixes` (ADR 0019 D8).
102///
103/// This is the automatic tier of the C3.2 precedence chain, sitting *below*
104/// the explicit `--cell-descriptor <file>` and `--bundled-descriptor <name>`
105/// selectors handled by [`resolve`] and *above* the legacy per-PDK Rust path.
106/// It replaces `pdk::PdkVariant`'s hardcoded prefix detection for the descriptor
107/// choice: instead of collapsing every GF180 cell to one variant (the #130 bug,
108/// where a 9t netlist was served by 7t models), it keys on the full vendor
109/// prefix — which carries the track designation (`…mcu7t5v0__` vs `…mcu9t5v0__`)
110/// — so a 9t netlist auto-selects the 9t descriptor.
111///
112/// Outcomes:
113/// - `Ok(Some(ir))` — exactly one bundled descriptor's declared prefixes cover
114///   the netlist; its parsed [`CellModelIr`] is returned.
115/// - `Ok(None)` — no bundled descriptor declares a prefix the netlist uses
116///   (e.g. an AIGPDK or SKY130 netlist, or a submodule-absent build whose
117///   embedded descriptors declare no prefixes). The caller falls back to the
118///   legacy per-PDK path, keeping today's behaviour.
119/// - `Err(msg)` — the netlist matches more than one bundled descriptor (e.g. a
120///   netlist mixing GF180 7t and 9t cells). That can't be auto-resolved, so the
121///   message tells the user to pin the choice with `--bundled-descriptor` or
122///   `--cell-descriptor`.
123pub fn auto_select<'a>(
124    cell_types: impl Iterator<Item = &'a str>,
125) -> Result<Option<CellModelIr>, String> {
126    // Parse each bundled descriptor once (there are only a handful). A
127    // submodule-absent build embeds valid *empty* descriptors that declare no
128    // prefixes; those match nothing and drop out below.
129    let mut parsed: Vec<(&'static str, CellModelIr)> = ALL
130        .iter()
131        .map(|d| {
132            (
133                d.name,
134                load(d.name).expect("listed bundled descriptor must load"),
135            )
136        })
137        .collect();
138
139    // Collect distinct cell types so the match scan is O(types × descriptors)
140    // rather than O(cells × descriptors) on a million-instance netlist.
141    let mut distinct: Vec<&str> = cell_types.collect();
142    distinct.sort_unstable();
143    distinct.dedup();
144
145    // A descriptor matches when any netlist cell type starts with one of its
146    // declared (non-empty) D8 prefixes.
147    let matched: Vec<usize> = parsed
148        .iter()
149        .enumerate()
150        .filter(|(_, (_, ir))| {
151            ir.library
152                .prefixes
153                .iter()
154                .any(|p| !p.is_empty() && distinct.iter().any(|ct| ct.starts_with(p.as_str())))
155        })
156        .map(|(i, _)| i)
157        .collect();
158
159    match matched.as_slice() {
160        [] => Ok(None),
161        &[i] => {
162            let (name, ir) = parsed.swap_remove(i);
163            clilog::info!(
164                "Auto-selected bundled cell-model-IR descriptor '{}' \
165                 ({}) from netlist cell-name prefix",
166                name,
167                ir.library.name
168            );
169            Ok(Some(ir))
170        }
171        many => {
172            let names: Vec<&str> = many.iter().map(|&i| parsed[i].0).collect();
173            Err(format!(
174                "netlist cell-name prefixes match multiple bundled descriptors \
175                 ({}); cannot auto-select. Pin the choice with \
176                 --bundled-descriptor <name> or --cell-descriptor <file>.",
177                names.join(", ")
178            ))
179        }
180    }
181}

183#[cfg(test)]
184mod tests {
185    use super::*;
186
187    #[test]
188    fn embedded_descriptors_parse_and_are_nonempty() {
189        for d in ALL {
190            // Always: the embedded JSON must parse (a build-time regression
191            // otherwise).
192            let ir = load(d.name).expect("listed descriptor must load");
193
194            // A build without the GF180 submodules embeds a valid *empty*
195            // descriptor (see build.rs). The real-data assertions only apply
196            // when the submodule was present at build time; the Unit Tests CI
197            // job checks GF180 out, so they are live there.
198            if ir.cells.is_empty() {
199                eprintln!(
200                    "skipping non-empty assertions for {}: empty descriptor \
201                     (GF180 submodule absent at build time)",
202                    d.name
203                );
204                continue;
205            }
206            assert!(
207                ir.cells.len() > 100,
208                "{} embedded descriptor has only {} cells",
209                d.name,
210                ir.cells.len()
211            );
212            assert!(
213                !ir.library.prefixes.is_empty(),
214                "{} descriptor should declare a D8 selection prefix",
215                d.name
216            );
217        }
218    }
219
220    #[test]
221    fn unknown_name_is_none() {
222        assert!(load("nonexistent_pdk").is_none());
223        assert!(json_by_name("nonexistent_pdk").is_none());
224    }
225
226    /// True only when the GF180 submodules were present at build time, so the
227    /// embedded descriptors carry real D8 prefixes. In a submodule-absent build
228    /// (some CI jobs) the descriptors are empty and auto-match is a no-op; the
229    /// real-data assertions are skipped there (mirrors
230    /// `embedded_descriptors_parse_and_are_nonempty`).
231    fn descriptors_have_prefixes() -> bool {
232        ALL.iter()
233            .all(|d| !load(d.name).unwrap().library.prefixes.is_empty())
234    }
235
236    #[test]
237    fn auto_select_picks_9t_for_a_9t_netlist() {
238        if !descriptors_have_prefixes() {
239            eprintln!("skipping: GF180 submodules absent at build time");
240            return;
241        }
242        // A 9t netlist's cell types carry the 9t track in their prefix; the 9t
243        // descriptor's declared prefix is `gf180mcu_fd_sc_mcu9t5v0__`, the 7t's
244        // is `gf180mcu_fd_sc_mcu7t5v0__` — disjoint, so exactly one matches.
245        let cells = [
246            "gf180mcu_fd_sc_mcu9t5v0__inv_2",
247            "gf180mcu_fd_sc_mcu9t5v0__nand2_1",
248            "gf180mcu_fd_sc_mcu9t5v0__dffq_1",
249        ];
250        let ir = auto_select(cells.iter().copied())
251            .expect("9t netlist must auto-match without ambiguity")
252            .expect("9t netlist must match a bundled descriptor");
253        assert_eq!(ir.library.name, "gf180mcu_fd_sc_mcu9t5v0__tt_025C_5v00");
254    }
255
256    #[test]
257    fn auto_select_picks_7t_for_a_7t_netlist() {
258        if !descriptors_have_prefixes() {
259            eprintln!("skipping: GF180 submodules absent at build time");
260            return;
261        }
262        let cells = [
263            "gf180mcu_fd_sc_mcu7t5v0__inv_2",
264            "gf180mcu_fd_sc_mcu7t5v0__nand2_1",
265        ];
266        let ir = auto_select(cells.iter().copied())
267            .expect("7t netlist must auto-match without ambiguity")
268            .expect("7t netlist must match a bundled descriptor");
269        assert_eq!(ir.library.name, "gf180mcu_fd_sc_mcu7t5v0__tt_025C_5v00");
270    }
271
272    #[test]
273    fn auto_select_no_match_for_aigpdk() {
274        // AIGPDK / unknown cells declare no GF180 prefix → no bundled match,
275        // caller falls back to the legacy per-PDK path.
276        let cells = ["AND2_00_0", "INV", "DFF"];
277        let r = auto_select(cells.iter().copied());
278        assert_eq!(r, Ok(None));
279    }
280
281    #[test]
282    fn auto_select_mixed_tracks_are_ambiguous() {
283        if !descriptors_have_prefixes() {
284            eprintln!("skipping: GF180 submodules absent at build time");
285            return;
286        }
287        // A netlist mixing 7t and 9t cells matches BOTH descriptors; neither can
288        // be auto-chosen, so the caller must pin the choice explicitly.
289        let cells = [
290            "gf180mcu_fd_sc_mcu7t5v0__inv_2",
291            "gf180mcu_fd_sc_mcu9t5v0__nand2_1",
292        ];
293        let r = auto_select(cells.iter().copied());
294        assert!(
295            r.is_err(),
296            "mixed-track netlist must be ambiguous, got {r:?}"
297        );
298        let msg = r.unwrap_err();
299        assert!(msg.contains("gf180mcu_7t") && msg.contains("gf180mcu_9t"));
300        assert!(msg.contains("--bundled-descriptor"));
301    }
