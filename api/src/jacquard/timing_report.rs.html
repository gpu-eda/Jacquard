1//! End-of-run structured timing report (`--timing-report <path.json>`).
2//!
3//! Schema versioning policy mirrors the timing IR (ADR 0002): explicit
4//! `schema_version` field, additive-only extension, breaking changes
5//! require a major bump and migration notes (per ADR 0008's stability
6//! contract).
7//!
8//! `worst_slack` is currently populated only from violation events.
9//! Closest-to-violation tracking when no violation ever occurred needs
10//! GPU-side near-miss instrumentation and is not in this module's scope.
11
12use std::cmp::Ordering;
13use std::path::Path;

15use serde::{Deserialize, Serialize};

17/// Schema version for the timing report JSON. Bump the major component
18/// for breaking changes; minor for additive fields per the ADR 0008
19/// stability contract.
20///
21/// 1.0.0 — initial schema (commit 58a7a04).
22/// 1.1.0 — additive: `stats.violations_truncated`; per-cycle violations
23///         array now bounded by default (worst-slack + per-word + totals
24///         still reflect every event).
25pub const SCHEMA_VERSION: &str = "1.2.0";

27/// One setup or hold violation record. Mirrors the `EventType::*Violation`
28/// payload, with `word_id` resolved to a human-readable `site` string via
29/// the caller's `WordSymbolMap`-backed resolver.
30#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
31pub struct ViolationRecord {
32    pub cycle: u32,
33    pub kind: ViolationKind,
34    pub word_id: u32,
35    pub site: String,
36    pub arrival_ps: u32,
37    pub constraint_ps: u32,
38    pub slack_ps: i32,
39}

42#[serde(rename_all = "snake_case")]
43pub enum ViolationKind {
44    Setup,
45    Hold,
46}

60/// Aggregate counters mirroring `SimStats` violation-related fields,
61/// plus a `violations_truncated` count for the case where the per-cycle
62/// `violations` array hit its cap and dropped further records.
63#[derive(Debug, Clone, Default, Serialize, Deserialize)]
64pub struct ReportStats {
65    pub setup_violations: u32,
66    pub hold_violations: u32,
67    pub events_dropped: u32,
68    /// Count of violation events observed but not retained in the
69    /// per-cycle `violations` array because the cap was reached. The
70    /// `setup_violations`/`hold_violations` totals and `worst_slack`
71    /// rankings always reflect every observed event; only the
72    /// per-cycle list is bounded.
73    #[serde(default)]
74    pub violations_truncated: u32,
75    /// Total negative slack (TNS / THS): the sum of every negative setup /
76    /// hold slack observed across the run, in picoseconds. Complements the
77    /// worst-single-slack (WNS / WHS) values carried in `worst_slack`. Always
78    /// ≤ 0; reflects every observed event regardless of the per-cycle cap.
79    #[serde(default)]
80    pub total_setup_slack_ps: i64,
81    #[serde(default)]
82    pub total_hold_slack_ps: i64,
83}

85/// Per-word aggregate: violation counts and worst slack across the run.
86/// `site` matches the formatted descriptor `WordSymbolMap` produces for
87/// `word_id`. `Option`s distinguish "no violation of this kind seen"
88/// from a literal-zero observation.
89#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
90pub struct PerWordSummary {
91    pub word_id: u32,
92    pub site: String,
93    pub setup_violations: u32,
94    pub hold_violations: u32,
95    pub worst_setup_slack_ps: Option<i32>,
96    pub worst_hold_slack_ps: Option<i32>,
97    pub worst_arrival_ps: Option<u32>,
98}

107/// Top-level report document.
108#[derive(Debug, Clone, Serialize, Deserialize)]
109pub struct TimingReport {
110    pub schema_version: String,
111    pub metadata: RunMetadata,
112    pub stats: ReportStats,
113    pub violations: Vec<ViolationRecord>,
114    pub per_word: Vec<PerWordSummary>,

118impl TimingReport {
119    pub fn new(metadata: RunMetadata) -> Self {
120        Self {
121            schema_version: SCHEMA_VERSION.to_string(),
122            metadata,
123            stats: ReportStats::default(),
124            violations: Vec::new(),
125            per_word: Vec::new(),
126            worst_slack: WorstSlack::default(),
127        }
128    }
129
130    /// Serialize as pretty JSON to the given writer.
131    pub fn write_pretty<W: std::io::Write>(&self, w: &mut W) -> serde_json::Result<()> {
132        serde_json::to_writer_pretty(w, self)
133    }
134
135    /// Convenience wrapper: write the report to a file path. Creates
136    /// parent directories if needed.
137    pub fn write_to_path(&self, path: &Path) -> std::io::Result<()> {
138        if let Some(parent) = path.parent() {
139            if !parent.as_os_str().is_empty() {
140                std::fs::create_dir_all(parent)?;
141            }
142        }
143        let f = std::fs::File::create(path)?;
144        let mut w = std::io::BufWriter::new(f);
145        self.write_pretty(&mut w)
146            .map_err(|e| std::io::Error::other(e.to_string()))?;
147        Ok(())
148    }
149
150    /// Render the report as a multi-line text summary suitable for
151    /// stdout. Matches `--timing-summary`. Stable enough for human
152    /// inspection but explicitly NOT a parseable contract per ADR 0008
153    /// (`--timing-report` JSON is the parseable form).
154    ///
155    /// ADR 0008 also lists "corner" and "margin percentage" as desired
156    /// summary content; both are deferred — corner because the
157    /// metadata struct doesn't carry it yet (next IR roundtrip), margin
158    /// because it duplicates information the consumer can compute from
159    /// `slack_ps` and `clock_period_ps` already in the report.
160    pub fn format_summary(&self) -> String {
161        // Top-N for the per-word count breakdown. Distinct from
162        // jacquard.rs::WORST_SLACK_TOP_N (which gates the JSON ranking
163        // capacity); this one bounds the human-facing table width.
164        const SUMMARY_TOP_N: usize = 5;
165
166        use std::fmt::Write as _;
167        let mut out = String::new();
168        // `writeln!` into a String via fmt::Write is infallible.
169        writeln!(out, "=== Jacquard Timing Summary ===").unwrap();
170        if let Some(d) = &self.metadata.design {
171            writeln!(out, "Design:        {d}").unwrap();
172        }
173        match &self.metadata.vector_source {
174            Some(v) => writeln!(
175                out,
176                "Vectors:       {v} ({} cycles)",
177                self.metadata.cycles_run
178            )
179            .unwrap(),
180            None => writeln!(out, "Vectors:       {} cycles", self.metadata.cycles_run).unwrap(),
181        }
182        writeln!(out, "Clock period:  {} ps", self.metadata.clock_period_ps).unwrap();
183        if let Some(t) = &self.metadata.timing_source {
184            writeln!(out, "Timing source: {t}").unwrap();
185        }
186        writeln!(out).unwrap();
187
188        writeln!(out, "Violations:").unwrap();
189        writeln!(out, "  Setup: {}", self.stats.setup_violations).unwrap();
190        writeln!(out, "  Hold:  {}", self.stats.hold_violations).unwrap();
191        let total = self.stats.setup_violations + self.stats.hold_violations;
192        writeln!(out, "  Total: {total}").unwrap();
193        if self.stats.events_dropped > 0 {
194            writeln!(
195                out,
196                "  Note:  {} events dropped (buffer overflow)",
197                self.stats.events_dropped
198            )
199            .unwrap();
200        }
201        if self.stats.violations_truncated > 0 {
202            writeln!(
203                out,
204                "  Note:  {} violation records dropped from the per-cycle list (cap reached); totals + worst-slack still reflect all events",
205                self.stats.violations_truncated
206            )
207            .unwrap();
208        }
209        writeln!(out).unwrap();
210
211        writeln!(out, "Worst slack:").unwrap();
212        write_worst(&mut out, "Setup", self.worst_slack.setup.first());
213        write_worst(&mut out, "Hold", self.worst_slack.hold.first());
214        writeln!(
215            out,
216            "  Total negative: setup (TNS)={}ps  hold (THS)={}ps",
217            self.stats.total_setup_slack_ps, self.stats.total_hold_slack_ps
218        )
219        .unwrap();
220
221        if !self.per_word.is_empty() {
222            writeln!(out).unwrap();
223            let shown = self.per_word.len().min(SUMMARY_TOP_N);
224            writeln!(
225                out,
226                "Top {shown} by violation count (of {} total words with violations):",
227                self.per_word.len()
228            )
229            .unwrap();
230            for w in self.per_word.iter().take(shown) {
231                let total = w.setup_violations + w.hold_violations;
232                writeln!(
233                    out,
234                    "  {} ({} violations): worst setup={} hold={} arrival={}",
235                    w.site,
236                    total,
237                    fmt_optional_ps(w.worst_setup_slack_ps),
238                    fmt_optional_ps(w.worst_hold_slack_ps),
239                    fmt_optional_ps(w.worst_arrival_ps.map(|a| a as i64)),
240                )
241                .unwrap();
242            }
243        }
244
245        out
246    }
247}

249fn write_worst(out: &mut String, label: &str, v: Option<&ViolationRecord>) {
250    use std::fmt::Write as _;
251    match v {
252        Some(v) => writeln!(
253            out,
254            "  {label}: {}ps  at {}  (cycle {})",
255            v.slack_ps, v.site, v.cycle
256        )
257        .unwrap(),
258        None => writeln!(out, "  {label}: (none)").unwrap(),
259    }
260}

262/// Render an `Option<i64>` (representing picoseconds) as `"-"` for
263/// `None` and `"<n>ps"` for `Some`. Generic over the integer side via
264/// the `Into<i64>` callsite cast above.
265fn fmt_optional_ps(v: Option<impl Into<i64>>) -> String {
266    match v {
267        Some(n) => format!("{}ps", n.into()),
268        None => "-".into(),
269    }
270}

277/// Accumulator that observes every violation as it occurs and keeps:
278/// running per-word aggregates, top-N worst-slack rankings, and a
279/// bounded per-cycle list. Per-word + worst-slack reflect every
280/// observed violation regardless of the per-cycle cap. Materialised
281/// into a `TimingReport` at end of run via `finalize`.
282pub struct ReportBuilder {
283    metadata: RunMetadata,
284    violations: Vec<ViolationRecord>,
285    per_word: std::collections::HashMap<u32, PerWordSummary>,
286    setup_worst: WorstSlackTracker,
287    hold_worst: WorstSlackTracker,
288    /// `None` = unbounded; `Some(n)` = keep the first `n` records.
289    max_violations: Option<usize>,
290    truncated: u32,
291    /// Running TNS / THS sums (negative slacks only), materialised into
292    /// `ReportStats` at `finalize`.
293    total_setup_slack_ps: i64,
294    total_hold_slack_ps: i64,
295}
296
297impl ReportBuilder {
298    /// Build a new accumulator. `max_violations` of `None` disables the
299    /// per-cycle cap (use with caution on violation-storm runs);
300    /// `Some(n)` retains the first `n` records and counts the rest in
301    /// `stats.violations_truncated`.
302    pub fn new(metadata: RunMetadata, worst_slack_n: usize, max_violations: Option<usize>) -> Self {
303        Self {
304            metadata,
305            violations: Vec::new(),
306            per_word: std::collections::HashMap::new(),
307            setup_worst: WorstSlackTracker::with_capacity(worst_slack_n),
308            hold_worst: WorstSlackTracker::with_capacity(worst_slack_n),
309            max_violations,
310            truncated: 0,
311            total_setup_slack_ps: 0,
312            total_hold_slack_ps: 0,
313        }
314    }
315
316    pub fn observe(&mut self, v: ViolationRecord) {
317        // Worst-slack, per-word, and the TNS/THS sums always see every
318        // violation; only the per-cycle list is bounded. Guard on
319        // `slack_ps < 0` so TNS/THS stay true "total *negative* slack"
320        // even if a zero/positive-slack record is ever observed.
321        match v.kind {
322            ViolationKind::Setup => {
323                if v.slack_ps < 0 {
324                    self.total_setup_slack_ps += v.slack_ps as i64;
325                }
326                self.setup_worst.push(v.clone())
327            }
328            ViolationKind::Hold => {
329                if v.slack_ps < 0 {
330                    self.total_hold_slack_ps += v.slack_ps as i64;
331                }
332                self.hold_worst.push(v.clone())
333            }
334        }
335        update_per_word(&mut self.per_word, &v);
336        match self.max_violations {
337            Some(cap) if self.violations.len() >= cap => {
338                self.truncated = self.truncated.saturating_add(1);
339            }
340            _ => self.violations.push(v),
341        }
342    }
343
344    /// Finalise into a `TimingReport`. The builder fills in
345    /// `cycles_run` (from the argument) and `violations_truncated`
346    /// (from its own counter). The caller-supplied
347    /// `stats.violations_truncated` is overwritten.
348    pub fn finalize(mut self, cycles_run: u32, mut stats: ReportStats) -> TimingReport {
349        self.metadata.cycles_run = cycles_run;
350        stats.violations_truncated = self.truncated;
351        stats.total_setup_slack_ps = self.total_setup_slack_ps;
352        stats.total_hold_slack_ps = self.total_hold_slack_ps;
353        let per_word = sort_per_word(self.per_word.into_values().collect());
354        TimingReport {
355            schema_version: SCHEMA_VERSION.to_string(),
356            metadata: self.metadata,
357            stats,
358            violations: self.violations,
359            per_word,
360            worst_slack: WorstSlack {
361                setup: self.setup_worst.into_sorted_vec(),
362                hold: self.hold_worst.into_sorted_vec(),
363            },
364        }
365    }
366}

368/// Bounded top-N tracker keeping the most-negative slacks (closest to
369/// or beyond violation). Internally a `Vec` since N is small (default 10).
370struct WorstSlackTracker {
371    capacity: usize,
372    items: Vec<ViolationRecord>,
373}
374
375impl WorstSlackTracker {
376    fn with_capacity(capacity: usize) -> Self {
377        Self {
378            capacity: capacity.max(1),
379            items: Vec::with_capacity(capacity.max(1)),
380        }
381    }
382
383    fn push(&mut self, v: ViolationRecord) {
384        // Worst = most negative slack; we keep the smallest N.
385        if self.items.len() < self.capacity {
386            self.items.push(v);
387            return;
388        }
389        // Find the largest (least negative) currently in the set; replace if v is smaller.
390        let (idx, max) = self
391            .items
392            .iter()
393            .enumerate()
394            .max_by(|(_, a), (_, b)| a.slack_ps.cmp(&b.slack_ps))
395            .expect("non-empty by construction");
396        if v.slack_ps < max.slack_ps {
397            self.items[idx] = v;
398        }
399    }
400
401    fn into_sorted_vec(mut self) -> Vec<ViolationRecord> {
402        self.items
403            .sort_by(|a, b| match a.slack_ps.cmp(&b.slack_ps) {
404                Ordering::Equal => a.cycle.cmp(&b.cycle),
405                other => other,
406            });
407        self.items
408    }
409}

411/// Fold one violation into the running per-word table. Used during
412/// `ReportBuilder::observe` so the aggregate sees every event even
413/// when the per-cycle list is capped.
414fn update_per_word(
415    by_word: &mut std::collections::HashMap<u32, PerWordSummary>,
416    v: &ViolationRecord,
417) {
418    let entry = by_word.entry(v.word_id).or_insert_with(|| PerWordSummary {
419        word_id: v.word_id,
420        site: v.site.clone(),
421        setup_violations: 0,
422        hold_violations: 0,
423        worst_setup_slack_ps: None,
424        worst_hold_slack_ps: None,
425        worst_arrival_ps: None,
426    });
427    match v.kind {
428        ViolationKind::Setup => {
429            entry.setup_violations += 1;
430            entry.worst_setup_slack_ps = Some(match entry.worst_setup_slack_ps {
431                Some(s) => s.min(v.slack_ps),
432                None => v.slack_ps,
433            });
434        }
435        ViolationKind::Hold => {
436            entry.hold_violations += 1;
437            entry.worst_hold_slack_ps = Some(match entry.worst_hold_slack_ps {
438                Some(s) => s.min(v.slack_ps),
439                None => v.slack_ps,
440            });
441        }
442    }
443    entry.worst_arrival_ps = Some(match entry.worst_arrival_ps {
444        Some(a) => a.max(v.arrival_ps),
445        None => v.arrival_ps,
446    });
447}

449/// Sort the per-word table for output: descending by total violation
450/// count, then ascending by word_id for deterministic ties.
451fn sort_per_word(mut summaries: Vec<PerWordSummary>) -> Vec<PerWordSummary> {
452    summaries.sort_by(|a, b| {
453        let total_a = a.setup_violations + a.hold_violations;
454        let total_b = b.setup_violations + b.hold_violations;
455        total_b.cmp(&total_a).then(a.word_id.cmp(&b.word_id))
456    });
457    summaries
458}

598        }
599        b.finalize(0, ReportStats::default()).per_word
600    }

781    /// fields added in later minor versions (e.g. `stats.violations_truncated`

783    /// If this ever fails to parse against the current types, the
784    /// schema has drifted in a non-additive way and the stability
785    /// contract has been broken.
786    #[test]
787    fn sample_fixture_parses_with_serde_default_for_newer_fields() {
788        let path = std::path::Path::new(env!("CARGO_MANIFEST_DIR"))
789            .join("tests/timing_ir/sample_reports/two_violations.json");
790        let text = std::fs::read_to_string(&path)
791            .unwrap_or_else(|e| panic!("read {}: {e}", path.display()));
792        let parsed: TimingReport =
793            serde_json::from_str(&text).expect("sample fixture failed to parse");
794        assert_eq!(parsed.schema_version, "1.0.0");
795        assert_eq!(parsed.violations.len(), 3);
796        assert_eq!(parsed.per_word.len(), 2);
797        assert_eq!(parsed.worst_slack.setup.len(), 2);
798        assert_eq!(parsed.worst_slack.hold.len(), 1);
799        // 1.1.0 field absent in the fixture; serde default kicks in.
