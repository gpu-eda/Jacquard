3//! Live signal-streaming tap for cosim (observe-only).
4//!
5//! Samples a configured bundle of design output nets per triggered edge
6//! and streams packed bytes out a UNIX-domain socket to an external
7//! renderer (first use: a live C64 video debug view). Reuses the per-edge
8//! `vcd_snapshot` ring — no GPU code — and never drives design inputs, so
9//! it stays out of the reactive path and imposes no `batch=1` penalty.
10//!
11//! Design: `docs/spikes/video-streaming-tap.md`.
12//!
13//! ## Wire format
14//!
15//! Cosim listens on the socket; a renderer connects. After each GPU batch,
16//! the tap writes one frame per batch that produced samples:
17//!
18//! ```text
19//! [ u64 first_tick LE ][ u32 sample_count LE ][ sample_count × bytes_per_sample payload ]
20//! ```
21//!
22//! Each sample packs the bundle's signals LSB-first in config order
23//! (`signals[0]` → bit 0 of byte 0). `bytes_per_sample = ceil(n_signals / 8)`.
24//! The `first_tick` header lets a renderer resync after a dropped batch.
25//! Backpressure or a gone client drops the batch and the connection; the
26//! renderer is expected to reconnect.
27
28use std::io::Write;
29use std::os::unix::net::{UnixListener, UnixStream};
30use std::path::PathBuf;
31
32use crate::sim::models::{read_bit, Edge, EdgeDetector};
33use crate::testbench::{SignalStreamConfig, StreamTrigger};
34
35/// Bytes in the per-batch frame header: `u64 first_tick` + `u32 sample_count`.
36const FRAME_HEADER_LEN: usize = 12;
37
38/// When the tap emits a sample.
39enum TriggerState {
40    /// Emit every `n`-th edge seen.
41    Every { n: u32, counter: u32 },
42    /// Emit on the rising edge of the net at `pos` in the output state.
43    Strobe { pos: u32, edge: EdgeDetector },
44}
45
46/// Trigger + packing, split out from the socket so it is unit-testable
47/// without binding a listener.
48struct Sampler {
49    /// Output-state bit positions for each bundle signal, in pack order.
50    positions: Vec<u32>,
51    bytes_per_sample: usize,
52    trigger: TriggerState,
53}
54
55impl Sampler {
56    fn new(positions: Vec<u32>, trigger: TriggerState) -> Self {
57        let bytes_per_sample = positions.len().div_ceil(8).max(1);
58        Self {
59            positions,
60            bytes_per_sample,
61            trigger,
62        }
63    }
64
65    /// Evaluate the trigger for one edge, advancing its state. If it fires,
66    /// append one packed sample (`bytes_per_sample` bytes, LSB-first) to
67    /// `out` and return `true`. `output_state` is the design-output half of
68    /// the per-edge snapshot (same frame as `resolve_to_state_pos`). Packs
69    /// in place — no per-edge allocation in the drain loop's hot path.
70    fn sample_into(&mut self, output_state: &[u32], out: &mut Vec<u8>) -> bool {
71        let fire = match &mut self.trigger {
72            TriggerState::Every { n, counter } => {
73                let f = *counter == 0;
74                *counter = (*counter + 1) % (*n).max(1);
75                f
76            }
77            TriggerState::Strobe { pos, edge } => {
78                matches!(edge.update(read_bit(output_state, *pos) != 0), Edge::Rising)
79            }
80        };
81        if !fire {
82            return false;
83        }
84        let base = out.len();
85        out.resize(base + self.bytes_per_sample, 0);
86        for (i, &pos) in self.positions.iter().enumerate() {
87            if read_bit(output_state, pos) != 0 {
88                out[base + i / 8] |= 1 << (i % 8);
89            }
90        }
91        true
92    }
93}

95/// A resolved, ready-to-run streaming tap. Constructed once per
96/// [`SignalStreamConfig`] after its signals resolve to state positions.
97pub struct SignalStreamTap {
98    name: String,
99    socket_path: PathBuf,
100    listener: UnixListener,
101    client: Option<UnixStream>,
102    sampler: Sampler,
103    /// Samples accumulated within the current batch.
104    batch_buf: Vec<u8>,
105    /// Tick of the first sample in `batch_buf`; meaningful only while
106    /// `batch_buf` is non-empty.
107    batch_first_tick: u64,
108}
109
110impl SignalStreamTap {
111    /// Bind the socket and build the tap. `positions` are the resolved
112    /// output-state positions of `cfg.signals` (pack order); `strobe_pos`
113    /// is the resolved position of the strobe net, required iff the trigger
114    /// is [`StreamTrigger::Strobe`]. The caller resolves these (it holds the
115    /// AIG / netlist / script) and skips the tap if any signal is unresolved.
116    pub fn new(
117        cfg: &SignalStreamConfig,
118        positions: Vec<u32>,
119        strobe_pos: Option<u32>,
120    ) -> std::io::Result<Self> {
121        let socket_path = PathBuf::from(&cfg.socket);
122
123        // Remove a stale socket from a previous run so bind() succeeds, but
124        // refuse to clobber a non-socket file the user may have named by
125        // mistake.
126        if let Ok(meta) = std::fs::symlink_metadata(&socket_path) {
127            use std::os::unix::fs::FileTypeExt;
128            if meta.file_type().is_socket() {
129                let _ = std::fs::remove_file(&socket_path);
130            } else {
131                return Err(std::io::Error::new(
132                    std::io::ErrorKind::AlreadyExists,
133                    format!("path {} exists and is not a socket", cfg.socket),
134                ));
135            }
136        }
137
138        let listener = UnixListener::bind(&socket_path)?;
139        listener.set_nonblocking(true)?;
140
141        let trigger = match &cfg.trigger {
142            StreamTrigger::Every(n) => TriggerState::Every {
143                n: (*n).max(1),
144                counter: 0,
145            },
146            StreamTrigger::Strobe(_) => TriggerState::Strobe {
147                pos: strobe_pos.expect("strobe trigger requires a resolved strobe position"),
148                edge: EdgeDetector::new(false),
149            },
150        };
151
152        Ok(Self {
153            name: cfg.name.clone(),
154            socket_path,
155            listener,
156            client: None,
157            sampler: Sampler::new(positions, trigger),
158            batch_buf: Vec::new(),
159            batch_first_tick: 0,
160        })
161    }
162
163    /// Sample one edge. Appends a packed sample to the batch buffer if the
164    /// trigger fires, stamping the batch's first-sample tick.
165    pub fn on_edge(&mut self, output_state: &[u32], tick: u64) {
166        let was_empty = self.batch_buf.is_empty();
167        if self.sampler.sample_into(output_state, &mut self.batch_buf) && was_empty {
168            self.batch_first_tick = tick;
169        }
170    }
171
172    /// Flush the batch's samples to the connected client. Non-blocking:
173    /// accepts a pending connection, and drops the batch + connection on any
174    /// write error (backpressure or a gone renderer).
175    pub fn flush_batch(&mut self) {
176        if self.client.is_none() {
177            if let Ok((stream, _)) = self.listener.accept() {
178                let _ = stream.set_nonblocking(true);
179                self.client = Some(stream);
180                clilog::info!("signal-stream `{}`: client connected", self.name);
181            }
182        }
183
184        if self.batch_buf.is_empty() {
185            return;
186        }
187
188        if let Some(stream) = self.client.as_mut() {
189            let n_samples = (self.batch_buf.len() / self.sampler.bytes_per_sample) as u32;
190            let mut header = [0u8; FRAME_HEADER_LEN];
191            header[..8].copy_from_slice(&self.batch_first_tick.to_le_bytes());
192            header[8..].copy_from_slice(&n_samples.to_le_bytes());
193            let wrote = stream
194                .write_all(&header)
195                .and_then(|_| stream.write_all(&self.batch_buf));
196            if wrote.is_err() {
197                clilog::debug!(
198                    "signal-stream `{}`: client write failed, dropping",
199                    self.name
200                );
201                self.client = None;
202            }
203        }
204
205        self.batch_buf.clear();
206    }
207}
208
209impl Drop for SignalStreamTap {
210    fn drop(&mut self) {
211        // Best-effort cleanup of the socket file we created.
212        let _ = std::fs::remove_file(&self.socket_path);
213    }
214}

216#[cfg(test)]
217mod tests {
218    use super::*;
219
220    /// Build a small output state with the given bit positions set.
221    fn state(bits: &[u32]) -> Vec<u32> {
222        let mut s = vec![0u32; 4];
223        for &b in bits {
224            s[(b >> 5) as usize] |= 1 << (b & 31);
225        }
226        s
227    }
228
229    /// Run one edge through the sampler; return the packed sample if it fired.
230    fn fire(s: &mut Sampler, st: &[u32]) -> Option<Vec<u8>> {
231        let mut buf = Vec::new();
232        s.sample_into(st, &mut buf).then_some(buf)
233    }
234
235    #[test]
236    fn every_fires_on_multiples() {
237        let mut s = Sampler::new(vec![0], TriggerState::Every { n: 3, counter: 0 });
238        let st = state(&[0]);
239        // Fires on edge 0, 3, 6, …
240        assert!(fire(&mut s, &st).is_some()); // 0
241        assert!(fire(&mut s, &st).is_none()); // 1
242        assert!(fire(&mut s, &st).is_none()); // 2
243        assert!(fire(&mut s, &st).is_some()); // 3
244    }
245
246    #[test]
247    fn every_one_fires_each_edge() {
248        let mut s = Sampler::new(vec![0], TriggerState::Every { n: 1, counter: 0 });
249        let st = state(&[]);
250        assert!(fire(&mut s, &st).is_some());
251        assert!(fire(&mut s, &st).is_some());
252    }
253
254    #[test]
255    fn strobe_fires_only_on_rising_edge() {
256        // Strobe net at pos 10; sampled net at pos 0.
257        let mut s = Sampler::new(
258            vec![0],
259            TriggerState::Strobe {
260                pos: 10,
261                edge: EdgeDetector::new(false),
262            },
263        );
264        let low = state(&[0]); // strobe=0
265        let high = state(&[0, 10]); // strobe=1
266        assert!(fire(&mut s, &low).is_none()); // 0→0
267        assert!(fire(&mut s, &high).is_some()); // 0→1 rising
268        assert!(fire(&mut s, &high).is_none()); // 1→1 held
269        assert!(fire(&mut s, &low).is_none()); // 1→0 falling
270        assert!(fire(&mut s, &high).is_some()); // 0→1 rising again
271    }
272
273    #[test]
274    fn packs_lsb_first_in_signal_order() {
275        // 6 signals → 1 byte. positions map signal i → some state bit.
276        // signals: [pos2, pos5, pos0, pos1, pos7, pos3]
277        let mut s = Sampler::new(
278            vec![2, 5, 0, 1, 7, 3],
279            TriggerState::Every { n: 1, counter: 0 },
280        );
281        // Drive state bits 2,0,7 high → signals index 0,2,4 high.
282        let sample = fire(&mut s, &state(&[2, 0, 7])).unwrap();
283        assert_eq!(sample.len(), 1);
284        // bits 0,2,4 set → 0b0001_0101 = 0x15
285        assert_eq!(sample[0], 0b0001_0101);
286    }
287
288    #[test]
289    fn packs_multibyte_bundle() {
290        // 10 signals → 2 bytes.
291        let positions: Vec<u32> = (0..10).collect();
292        let mut s = Sampler::new(positions, TriggerState::Every { n: 1, counter: 0 });
293        // Set state bits 0,1,8,9 → signals 0,1,8,9.
294        let sample = fire(&mut s, &state(&[0, 1, 8, 9])).unwrap();
295        assert_eq!(sample.len(), 2);
296        assert_eq!(sample[0], 0b0000_0011); // signals 0,1 in byte 0
297        assert_eq!(sample[1], 0b0000_0011); // signals 8,9 in byte 1
298    }
