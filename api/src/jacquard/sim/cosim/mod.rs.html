494/// Resolve every configured bus's pin positions + build its CPU-side decoder
495/// lane. Backend-agnostic (uses the agnostic `resolve_to_state_pos`); shared by
496/// the Metal backend (which packs the positions into the GPU `BusTraceParams`)
497/// and the CPU backend (which runs the beat FSM directly). Protocol-gated to
498/// APB3 — other protocols are skipped with a warning (Phase 2). The returned
499/// `Vec<BusTracePositions>` and `Vec<BusTraceLane>` are index-aligned: vec index
500/// == GPU `bus_id` (packed into the beat `flags>>8`).
501pub(crate) fn build_bus_trace(
502    aig: &AIG,
503    netlistdb: &NetlistDB,
504    script: &FlattenedScriptV1,
505    configs: &[crate::testbench::BusTraceConfig],
506) -> (Vec<BusTracePositions>, Vec<BusTraceLane>) {
507    use crate::sim::models::bus_trace::{pin_basename, BusTraceDecoder};
508    use crate::sim::trace_signals::resolve_to_state_pos;
509    use crate::testbench::BusProtocol;
510
511    let mut positions: Vec<BusTracePositions> = Vec::new();
512    let mut lanes: Vec<BusTraceLane> = Vec::new();
513
514    let resolve =
515        |name: &str| resolve_to_state_pos(aig, netlistdb, script, name).unwrap_or(0xFFFFFFFF);
516
517    for cfg in configs.iter() {
518        if positions.len() >= MAX_BUS_TRACES {
519            clilog::warn!(
520                "bus-trace: more than {} buses configured; `{}` and later ignored",
521                MAX_BUS_TRACES,
522                cfg.name
523            );
524            break;
525        }
526        match cfg.protocol {
527            BusProtocol::Apb3 => {}
528            other => {
529                clilog::warn!(
530                    "bus-trace `{}`: protocol {:?} not yet implemented (Phase 2); skipping",
531                    cfg.name,
532                    other
533                );
534                continue;
535            }
536        }
537
538        let addr_bits = cfg.addr_bits.min(BUS_TRACE_MAX_ADR_BITS);
539        let data_bits = cfg.data_bits.min(BUS_TRACE_MAX_DAT_BITS);
540        if cfg.addr_bits > BUS_TRACE_MAX_ADR_BITS || cfg.data_bits > BUS_TRACE_MAX_DAT_BITS {
541            clilog::warn!(
542                "bus-trace `{}`: addr/data width capped at {}/{} bits (Phase 1)",
543                cfg.name,
544                BUS_TRACE_MAX_ADR_BITS,
545                BUS_TRACE_MAX_DAT_BITS
546            );
547        }
548
549        let mut p = BusTracePositions {
550            protocol_apb3: true,
551            addr_bits,
552            data_bits,
553            sel_pos: resolve(&pin_basename(cfg, "psel")),
554            enable_pos: resolve(&pin_basename(cfg, "penable")),
555            ready_pos: resolve(&pin_basename(cfg, "pready")),
556            write_pos: resolve(&pin_basename(cfg, "pwrite")),
557            resp_pos: resolve(&pin_basename(cfg, "pslverr")),
558            ..Default::default()
559        };
560        let abase = pin_basename(cfg, "paddr");
561        for b in 0..addr_bits {
562            p.addr_pos[b] = resolve(&format!("{abase}[{b}]"));
563        }
564        let wbase = pin_basename(cfg, "pwdata");
565        let rbase = pin_basename(cfg, "prdata");
566        for b in 0..data_bits {
567            p.wdata_pos[b] = resolve(&format!("{wbase}[{b}]"));
568            p.rdata_pos[b] = resolve(&format!("{rbase}[{b}]"));
569        }
570
571        if p.sel_pos == 0xFFFFFFFF || p.enable_pos == 0xFFFFFFFF {
572            clilog::warn!(
573                "bus-trace `{}`: psel/penable did not resolve (prefix `{}`) — \
574                 this bus will not capture. Check the prefix / `signals` overrides.",
575                cfg.name,
576                cfg.prefix
577            );
578        } else {
579            let n_addr = p.addr_pos.iter().filter(|&&x| x != 0xFFFFFFFF).count();
580            clilog::info!(
581                "bus-trace `{}` (APB3): psel/penable resolved, addr {}/{} bits, \
582                 pready={} pslverr={}",
583                cfg.name,
584                n_addr,
585                addr_bits,
586                p.ready_pos != 0xFFFFFFFF,
587                p.resp_pos != 0xFFFFFFFF
588            );
589        }
590
591        positions.push(p);
592        lanes.push(BusTraceLane {
593            name: cfg.name.clone(),
594            decoder: BusTraceDecoder::new(cfg.protocol),
595        });
596    }
597
598    (positions, lanes)
599}

601/// Per-clock-domain flag bit positions in the packed state buffer.
602///
603/// Each clock domain (identified by its netlistdb pin ID) has its own set of
604/// posedge/negedge flag bits that gate DFF writeout for that domain.
605struct ClockDomainFlags {
606    /// Netlistdb pin ID for this clock (key from `clock_pin2aigpins`).
607    #[allow(dead_code)]
608    clock_pinid: usize,
609    /// Human-readable name from pin (e.g. "io$clk$i").
610    name: String,
611    /// State positions for this domain's posedge flags.
612    posedge_flag_bits: Vec<u32>,
613    /// State positions for this domain's negedge flags.
614    negedge_flag_bits: Vec<u32>,
615    /// State position of the clock input port itself (if it's a primary input).
616    clock_input_pos: Option<u32>,
617    /// GPIO index if this clock is driven from a GPIO.
618    clock_gpio: Option<usize>,
619}
620
621/// Maps GPIO pin indices to bit positions in the packed u32 state buffer.
622pub(crate) struct GpioMapping {
623    /// gpio_in[idx] → (aigpin, state bit position)
624    input_bits: HashMap<usize, u32>,
625    /// gpio_out[idx] → state bit position in output_map
626    output_bits: HashMap<usize, u32>,
627    /// Per-clock-domain flag bit positions (grouped by netlistdb pin ID).
628    clock_domains: Vec<ClockDomainFlags>,
629    /// Named input port → state bit position (for non-GPIO ports like por_l, resetb_h)
630    named_input_bits: HashMap<String, u32>,
631}

633/// Set a single bit in a packed u32 state buffer.
634#[inline]
635fn set_bit(state: &mut [u32], pos: u32, val: u8) {
636    let word = &mut state[(pos >> 5) as usize];
637    let mask = 1u32 << (pos & 31);
638    if val != 0 {
639        *word |= mask;
640    } else {
641        *word &= !mask;
642    }
643}
644
645/// Clear a single bit in a packed u32 state buffer.
646#[inline]
647fn clear_bit(state: &mut [u32], pos: u32) {
648    state[(pos >> 5) as usize] &= !(1u32 << (pos & 31));
649}

651/// Convert a 0/1 bit value to a VCD scalar value.
652#[inline]
653fn bit_to_vcd_value(bit: u8) -> vcd_ng::Value {
654    if bit != 0 {
655        vcd_ng::Value::V1
656    } else {
657        vcd_ng::Value::V0
658    }
659}

661/// Build GPIO-to-state-buffer mapping from AIG + FlattenedScript.
662///
663/// When `port_mapping` is provided, maps GPIO indices to port names explicitly
664/// (for designs like ChipFlow that use named ports instead of gpio_in[N]/gpio_out[N]).
665/// Falls back to parsing gpio_in[N]/gpio_out[N] from pin names when no mapping given.
666pub(crate) fn build_gpio_mapping(
667    aig: &AIG,
668    netlistdb: &NetlistDB,
669    script: &FlattenedScriptV1,
670    port_mapping: Option<&PortMapping>,
671) -> GpioMapping {
672    let mut input_bits: HashMap<usize, u32> = HashMap::new();
673    let mut output_bits: HashMap<usize, u32> = HashMap::new();
674    let mut named_input_bits: HashMap<String, u32> = HashMap::new();
675
676    // Group clock flags by pinid → ClockDomainFlags.
677    // BTreeMap gives deterministic iteration order (ascending pinid).
678    let mut clock_domain_map: std::collections::BTreeMap<usize, (Vec<u32>, Vec<u32>)> =
679        std::collections::BTreeMap::new();
680
681    // Build reverse lookup: port_name → gpio_index for port_mapping mode
682    let input_name_to_gpio: HashMap<String, usize> = port_mapping
683        .map(|pm| {
684            pm.inputs
685                .iter()
686                .filter_map(|(k, v)| k.parse::<usize>().ok().map(|idx| (v.clone(), idx)))
687                .collect()
688        })
689        .unwrap_or_default();
690    let output_name_to_gpio: HashMap<String, usize> = port_mapping
691        .map(|pm| {
692            pm.outputs
693                .iter()
694                .filter_map(|(k, v)| k.parse::<usize>().ok().map(|idx| (v.clone(), idx)))
695                .collect()
696        })
697        .unwrap_or_default();
698
699    // Map input ports → state buffer positions
700    for (aigpin_idx, driv) in aig.drivers.iter().enumerate() {
701        match driv {
702            DriverType::InputPort(pinid) => {
703                let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
704                // Try port_mapping first, then fall back to gpio_in[N] parsing
705                let gpio_idx = input_name_to_gpio
706                    .get(&pin_name)
707                    .copied()
708                    .or_else(|| parse_gpio_index(&pin_name, "gpio_in"));
709                if let Some(gpio_idx) = gpio_idx {
710                    if let Some(&pos) = script.input_map.get(&aigpin_idx) {
711                        input_bits.insert(gpio_idx, pos);
712                        clilog::debug!(
713                            "input[{}] = pin '{}' → state pos {}",
714                            gpio_idx,
715                            pin_name,
716                            pos
717                        );
718                    }
719                }
720                // Also store by name for constant_ports
721                if let Some(&pos) = script.input_map.get(&aigpin_idx) {
722                    named_input_bits.insert(pin_name.clone(), pos);
723                }
724            }
725            DriverType::InputClockFlag(pinid, is_negedge) => {
726                if let Some(&pos) = script.input_map.get(&aigpin_idx) {
727                    let entry = clock_domain_map
728                        .entry(*pinid)
729                        .or_insert_with(|| (Vec::new(), Vec::new()));
730                    if *is_negedge == 0 {
731                        entry.0.push(pos);
732                    } else {
733                        entry.1.push(pos);
734                    }
735                    let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
736                    clilog::debug!(
737                        "ClockFlag aigpin={} pin={} (pinid={}) negedge={} pos={}",
738                        aigpin_idx,
739                        pin_name,
740                        pinid,
741                        is_negedge,
742                        pos
743                    );
744                }
745            }
746            _ => {}
747        }
748    }
749
750    // Reverse-lookup map for extra observables: name → aigpin. Used
751    // below to resolve top-level `inout` ports — those have
752    // Direction::Unknown so they don't appear in output_map under
753    // their own name, but the AIG-side bidir-pad observability code
754    // (see `aig.rs::resolve_bi_24t_observable_base`) registers the
755    // core-drive-out aigpin under the synthetic `<port>__out` name.
756    // Looking that up gives peripheral observers (UART RX decoder,
757    // GPIO output watchers) the right signal. See #79.
758    let mut extra_observable_by_name: HashMap<String, usize> = HashMap::new();
759    for (&aigpin, names) in aig.extra_observable_names.iter() {
760        for name in names {
761            extra_observable_by_name.insert(name.clone(), aigpin);
762        }
763    }
764
765    // Map output ports → state buffer positions
766    for i in netlistdb.cell2pin.iter_set(0) {
767        let pin_name = netlistdb.pinnames[i].dbg_fmt_pin();
768        // Try port_mapping first, then fall back to gpio_out[N] parsing
769        let gpio_idx = output_name_to_gpio
770            .get(&pin_name)
771            .copied()
772            .or_else(|| parse_gpio_index(&pin_name, "gpio_out"));
773        if let Some(gpio_idx) = gpio_idx {
774            let mut aigpin_iv = aig.pin2aigpin_iv[i];
775            // Top-level inout: pin2aigpin_iv on a wafer.space-style
776            // `bi_24t` pad resolves to the input-side `Y` aigpin
777            // (PAD→core), which is non-MAX but absent from
778            // output_map. The OUTPUT direction's aigpin (core `A`→PAD)
779            // is registered under `<pin>__out` in
780            // extra_observable_names; falling back unconditionally
781            // (when found) overrides the input-side resolution with
782            // the actual output driver. Safe when the synthetic name
783            // is absent: aigpin_iv stays as-is and we fall through to
784            // the existing diagnostic.
785            if netlistdb.pindirect[i] == Direction::Unknown {
786                let synth_name = format!("{pin_name}__out");
787                if let Some(&aigpin) = extra_observable_by_name.get(&synth_name) {
788                    aigpin_iv = aigpin;
789                    clilog::debug!(
790                        "output[{}] inout pin '{}' resolved via bidir-pad \
791                         observable '{}' → aigpin {}",
792                        gpio_idx,
793                        pin_name,
794                        synth_name,
795                        aigpin
796                    );
797                }
798            }
799            if aigpin_iv == usize::MAX {
800                clilog::info!(
801                    "output[{}] pin '{}' has no AIG connection (usize::MAX)",
802                    gpio_idx,
803                    pin_name
804                );
805                continue;
806            }
807            if aigpin_iv <= 1 {
808                clilog::info!(
809                    "output[{}] pin '{}' is constant (aigpin_iv={})",
810                    gpio_idx,
811                    pin_name,
812                    aigpin_iv
813                );
814                continue;
815            }
816            if let Some(&pos) = script.output_map.get(&aigpin_iv) {
817                output_bits.insert(gpio_idx, pos);
818                clilog::debug!(
819                    "output[{}] = pin '{}' → state pos {}",
820                    gpio_idx,
821                    pin_name,
822                    pos
823                );
824            } else if let Some(&pos) = script.output_map.get(&(aigpin_iv ^ 1)) {
825                output_bits.insert(gpio_idx, pos);
826                clilog::debug!(
827                    "output[{}] = pin '{}' → state pos {} (flipped inv)",
828                    gpio_idx,
829                    pin_name,
830                    pos
831                );
832            } else {
833                clilog::info!(
834                    "output[{}] pin '{}' aigpin_iv={} (aigpin={}) not in output_map (dir={:?})",
835                    gpio_idx,
836                    pin_name,
837                    aigpin_iv,
838                    aigpin_iv >> 1,
839                    netlistdb.pindirect[i]
840                );
841            }
842        }
843    }
844
845    // Build ClockDomainFlags from grouped clock flags
846    // Also resolve GPIO index for each clock domain by looking up the clock pin
847    // in the input_bits mapping.
848    let mut clock_domains: Vec<ClockDomainFlags> = Vec::new();
849    // Build reverse: pinid → (pin_name, gpio_idx, state_pos) from input_bits
850    let mut pinid_to_input_info: HashMap<usize, (String, Option<usize>, Option<u32>)> =
851        HashMap::new();
852    for (aigpin_idx, driv) in aig.drivers.iter().enumerate() {
853        if let DriverType::InputPort(pinid) = driv {
854            let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
855            let gpio_idx = input_name_to_gpio
856                .get(&pin_name)
857                .copied()
858                .or_else(|| parse_gpio_index(&pin_name, "gpio_in"));
859            let pos = script.input_map.get(&aigpin_idx).copied();
860            pinid_to_input_info.insert(*pinid, (pin_name, gpio_idx, pos));
861        }
862    }
863
864    for (pinid, (posedge_bits, negedge_bits)) in &clock_domain_map {
865        let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
866        // Try to find the GPIO index and state position for this clock pin
867        let (clock_gpio, clock_input_pos) =
868            if let Some((_, gpio_idx, pos)) = pinid_to_input_info.get(pinid) {
869                (*gpio_idx, *pos)
870            } else {
871                (None, None)
872            };
873        clock_domains.push(ClockDomainFlags {
874            clock_pinid: *pinid,
875            name: pin_name.clone(),
876            posedge_flag_bits: posedge_bits.clone(),
877            negedge_flag_bits: negedge_bits.clone(),
878            clock_input_pos,
879            clock_gpio,
880        });
881        clilog::info!(
882            "Clock domain '{}' (pinid={}): {} posedge flags, {} negedge flags, gpio={:?}",
883            pin_name,
884            pinid,
885            posedge_bits.len(),
886            negedge_bits.len(),
887            clock_gpio
888        );
889    }
890
891    let total_posedge: usize = clock_domains
892        .iter()
893        .map(|d| d.posedge_flag_bits.len())
894        .sum();
895    let total_negedge: usize = clock_domains
896        .iter()
897        .map(|d| d.negedge_flag_bits.len())
898        .sum();
899    clilog::info!(
900        "GPIO mapping: {} inputs, {} outputs, {} clock domains ({} posedge flags, {} negedge flags)",
901        input_bits.len(),
902        output_bits.len(),
903        clock_domains.len(),
904        total_posedge,
905        total_negedge
906    );
907
908    clilog::info!("Named input ports: {} mapped", named_input_bits.len());
909
910    GpioMapping {
911        input_bits,
912        output_bits,
913        clock_domains,
914        named_input_bits,
915    }
916}

918/// Resolve an internal signal name to its output state bit position.
919///
920/// Searches DFF Q output pins (which appear as cell output pins driving named nets)
921/// for names containing `pattern`. Returns the state buffer bit position, or 0xFFFFFFFF.
922fn resolve_signal_pos(
923    aig: &AIG,
924    netlistdb: &NetlistDB,
925    script: &FlattenedScriptV1,
926    pattern: &str,
927) -> u32 {
928    // Strategy: scan ALL cell pins looking for names matching pattern.
929    // DFF Q outputs are cell output pins whose names match the net they drive.
930    for cellid in 0..netlistdb.num_cells {
931        for pinid in netlistdb.cell2pin.iter_set(cellid) {
932            let pin_name = netlistdb.pinnames[pinid].dbg_fmt_pin();
933            if !pin_name.contains(pattern) {
934                continue;
935            }
936            // Only care about output pins (Q of DFFs) — check for output_map presence
937            let aigpin_iv = aig.pin2aigpin_iv[pinid];
938            if aigpin_iv == usize::MAX || aigpin_iv <= 1 {
939                continue;
940            }
941            if let Some(&pos) = script.output_map.get(&aigpin_iv) {
942                return pos;
943            }
944            if let Some(&pos) = script.output_map.get(&(aigpin_iv ^ 1)) {
945                return pos;
946            }
947        }
948    }
949    0xFFFFFFFF
950}

972/// Build WbTraceParams by resolving all bus signal names to state positions.
973#[cfg(any(feature = "metal", feature = "cuda", feature = "hip"))]
974fn build_wb_trace_params(
975    aig: &AIG,
976    netlistdb: &NetlistDB,
977    script: &FlattenedScriptV1,
978) -> WbTraceParams {
979    let mut params = WbTraceParams {
980        ibus_cyc_pos: 0xFFFFFFFF,
981        ibus_stb_pos: 0xFFFFFFFF,
982        ibus_adr_pos: [0xFFFFFFFF; WB_TRACE_MAX_ADR_BITS],
983        ibus_rdata_pos: [0xFFFFFFFF; WB_TRACE_MAX_DAT_BITS],
984        dbus_cyc_pos: 0xFFFFFFFF,
985        dbus_stb_pos: 0xFFFFFFFF,
986        dbus_we_pos: 0xFFFFFFFF,
987        dbus_adr_pos: [0xFFFFFFFF; WB_TRACE_MAX_ADR_BITS],
988        spiflash_ack_pos: 0xFFFFFFFF,
989        sram_ack_pos: 0xFFFFFFFF,
990        csr_ack_pos: 0xFFFFFFFF,
991        has_trace: 0,
992    };
993
994    // ibus (instruction bus)
995    params.ibus_cyc_pos = resolve_signal_pos(aig, netlistdb, script, "cpu.fetch.ibus__cyc");
996    params.ibus_stb_pos = resolve_signal_pos(aig, netlistdb, script, "cpu.fetch.ibus__stb");
997    for i in 0..WB_TRACE_MAX_ADR_BITS {
998        params.ibus_adr_pos[i] =
999            resolve_bus_signal(aig, netlistdb, script, "cpu.fetch.ibus__adr", i);
1000    }
1001    for i in 0..WB_TRACE_MAX_DAT_BITS {
1002        params.ibus_rdata_pos[i] =
1003            resolve_bus_signal(aig, netlistdb, script, "cpu.fetch.ibus_rdata", i);
1004    }
1005
1006    // dbus (data bus)
1007    params.dbus_cyc_pos = resolve_signal_pos(aig, netlistdb, script, "cpu.loadstore.dbus__cyc");
1008    params.dbus_stb_pos = resolve_signal_pos(aig, netlistdb, script, "cpu.loadstore.dbus__stb");
1009    params.dbus_we_pos = resolve_signal_pos(aig, netlistdb, script, "cpu.loadstore.dbus__we");
1010    for i in 0..WB_TRACE_MAX_ADR_BITS {
1011        params.dbus_adr_pos[i] =
1012            resolve_bus_signal(aig, netlistdb, script, "cpu.loadstore.dbus__adr", i);
1013    }
1014
1015    // peripheral acks
1016    params.spiflash_ack_pos =
1017        resolve_signal_pos(aig, netlistdb, script, "spiflash.ctrl.wb_bus__ack");
1018    params.sram_ack_pos = resolve_signal_pos(aig, netlistdb, script, "sram.wb_bus__ack");
1019    params.csr_ack_pos = resolve_signal_pos(aig, netlistdb, script, "wb_to_csr.wb_bus__ack");
1020
1021    // Check if we resolved any signals
1022    let found = [
1023        params.ibus_cyc_pos,
1024        params.ibus_stb_pos,
1025        params.dbus_cyc_pos,
1026    ]
1027    .iter()
1028    .filter(|&&p| p != 0xFFFFFFFF)
1029    .count();
1030    if found > 0 {
1031        params.has_trace = 1;
1032        let ibus_adr_count = params
1033            .ibus_adr_pos
1034            .iter()
1035            .filter(|&&p| p != 0xFFFFFFFF)
1036            .count();
1037        let ibus_rdata_count = params
1038            .ibus_rdata_pos
1039            .iter()
1040            .filter(|&&p| p != 0xFFFFFFFF)
1041            .count();
1042        let dbus_adr_count = params
1043            .dbus_adr_pos
1044            .iter()
1045            .filter(|&&p| p != 0xFFFFFFFF)
1046            .count();
1047        clilog::info!(
1048            "WB trace: ibus_cyc={} ibus_stb={} ibus_adr={}/30 ibus_rdata={}/32",
1049            params.ibus_cyc_pos != 0xFFFFFFFF,
1050            params.ibus_stb_pos != 0xFFFFFFFF,
1051            ibus_adr_count,
1052            ibus_rdata_count
1053        );
1054        clilog::info!(
1055            "WB trace: dbus_cyc={} dbus_stb={} dbus_we={} dbus_adr={}/30",
1056            params.dbus_cyc_pos != 0xFFFFFFFF,
1057            params.dbus_stb_pos != 0xFFFFFFFF,
1058            params.dbus_we_pos != 0xFFFFFFFF,
1059            dbus_adr_count
1060        );
1061        clilog::info!(
1062            "WB trace: spiflash_ack={} sram_ack={} csr_ack={}",
1063            params.spiflash_ack_pos != 0xFFFFFFFF,
1064            params.sram_ack_pos != 0xFFFFFFFF,
1065            params.csr_ack_pos != 0xFFFFFFFF
1066        );
1067    } else {
1068        clilog::info!("WB trace: no bus signals found in netlist, tracing disabled");
1069    }
1070
1071    params
1072}

1074/// Build the bus-trace GPU params block and the parallel CPU decoder
1075/// lanes from the configured bus traces. The Vec index of each returned
1076/// lane equals the GPU `bus_id` packed into entry flags, so the drain
1077/// loop can route each beat to the right decoder.
1078///
1079/// Only APB3 is wired in Phase 1; AHB-Lite / AHB5 entries are skipped
1080/// with a warning (see `docs/plans/bus-transaction-tracing.md`).
1081#[cfg(any(feature = "metal", feature = "cuda", feature = "hip"))]
1082fn build_bus_trace_params(
1083    aig: &AIG,
1084    netlistdb: &NetlistDB,
1085    script: &FlattenedScriptV1,
1086    configs: &[crate::testbench::BusTraceConfig],
1087) -> (BusTraceParamsAll, Vec<BusTraceLane>) {
1088    // Resolve pin positions + build the CPU decoder lanes agnostically (shared
1089    // with the CPU backend), then pack the positions field-for-field into the
1090    // GPU `BusTraceParams`. The packing below is the *only* GPU-specific step;
1091    // the resolve/warn/lane logic lives in `build_bus_trace`, so the GPU struct
1092    // bytes are unchanged from the pre-extraction inline version.
1093    let (positions, lanes) = build_bus_trace(aig, netlistdb, script, configs);
1094
1095    let mut all = BusTraceParamsAll {
1096        n_buses: 0,
1097        _pad: [0; 3],
1098        buses: [BusTraceParams::default(); MAX_BUS_TRACES],
1099    };
1100    for pos in positions.iter() {
1101        // Only APB3 buses reach here (others are skipped in build_bus_trace),
1102        // so the GPU `protocol` field is always 0 == BUS_PROTO_APB3.
1103        debug_assert!(pos.protocol_apb3);
1104        let p = BusTraceParams {
1105            protocol: 0,
1106            addr_bits: pos.addr_bits as u32,
1107            data_bits: pos.data_bits as u32,

1187/// CUDA/HIP analog of `MetalBackend::build_flash_buffers` (cosim/metal.rs:956).
1188/// Builds the persistent `FlashState`, the const `FlashDinParams`/`FlashModelParams`,
1189/// and the 16 MiB firmware buffer as device-resident `UVec<u8>` (the event-buffer
1190/// FFI pattern), loading `config.flash.firmware` at `firmware_offset`. Init is
1191/// byte-identical to the Metal path (`data_width=1, prev_csn=1, model_prev_csn=1,
1192/// d_i=0x0F, in_reset=1`, rest 0; firmware area pre-filled 0xFF for erased flash).
1193/// Returns `(flash_state, flash_din_params, flash_model_params, flash_data)`.
1194#[cfg(any(feature = "cuda", feature = "hip"))]
1195pub(crate) fn build_flash_buffers_dev(
1196    config: &TestbenchConfig,
1197    script: &FlattenedScriptV1,
1198    state_size: usize,
1199    gpio_map: &GpioMapping,
1200    device: ulib::Device,
1201) -> (
1202    ulib::UVec<u8>,
1203    ulib::UVec<u8>,
1204    ulib::UVec<u8>,
1205    ulib::UVec<u8>,
1206) {
1207    // ADR 0013 plural QSPI memory (Stage B) — mirror MetalBackend::build_flash_buffers.
1208    const FLASH_SIZE_DEFAULT: usize = FLASH_DATA_SIZE;
1209    let qspi = config.effective_qspi_memory();
1210    assert!(
1211        qspi.len() <= MAX_QSPI_MEMS,
1212        "too many QSPI memories ({}); GPU backend supports at most {}",
1213        qspi.len(),
1214        MAX_QSPI_MEMS
1215    );
1216    let n = qspi.len();
1217
1218    // Per-instance backing-store sizes + byte offsets into the shared buffer.
1219    let sizes: Vec<usize> = qspi
1220        .iter()
1221        .map(|m| m.size_bytes.unwrap_or(FLASH_SIZE_DEFAULT))
1222        .collect();
1223    let mut offsets = vec![0u32; n];
1224    let mut total = 0usize;
1225    for i in 0..n {
1226        offsets[i] = total as u32;
1227        total += sizes[i];
1228    }
1229    let total = total.max(1);
1230
1231    // FlashState: one slot per instance (>=1 to avoid a 0-byte buffer).
1232    let n_slots = n.max(1);
1233    let mut states: Vec<FlashState> = vec![unsafe { std::mem::zeroed() }; n_slots];
1234    for fs in states.iter_mut() {
1235        fs.data_width = 1;
1236        fs.prev_csn = 1;
1237        fs.model_prev_csn = 1;
1238        fs.d_i = 0x0F;
1239        fs.in_reset = 1;
1240    }
1241    let flash_state = slice_to_dev(&states, device);
1242
1243    // FlashDinParamsAll (per-instance MISO input positions).
1244    let mut din: FlashDinParamsAll = unsafe { std::mem::zeroed() };
1245    din.n_flashes = n as u32;
1246    for (idx, m) in qspi.iter().enumerate() {
1247        let p = &mut din.flashes[idx];
1248        p.has_flash = 1;
1249        p.xmask_state_offset = script.xprop_state_offset;
1250        for i in 0..4 {
1251            p.d_in_pos[i] = gpio_map
1252                .input_bits
1253                .get(&(m.d0_gpio + i))
1254                .copied()
1255                .unwrap_or(0xFFFFFFFF);
1256        }
1257    }
1258    let flash_din_params = struct_to_dev(&din, device);
1259
1260    // FlashModelParamsAll (pin positions + per-instance backing slice).
1261    let mut model: FlashModelParamsAll = unsafe { std::mem::zeroed() };
1262    model.n_flashes = n as u32;
1263    for (idx, m) in qspi.iter().enumerate() {
1264        let p = &mut model.flashes[idx];
1265        p.state_size = state_size as u32;
1266        p.clk_out_pos = gpio_map.output_bits.get(&m.clk_gpio).copied().unwrap_or(0);
1267        p.csn_out_pos = gpio_map.output_bits.get(&m.csn_gpio).copied().unwrap_or(0);
1268        for i in 0..4 {
1269            p.d_out_pos[i] = gpio_map
1270                .output_bits
1271                .get(&(m.d0_gpio + i))
1272                .copied()
1273                .unwrap_or(0xFFFFFFFF);
1274        }
1275        p.flash_data_size = sizes[idx] as u32;
1276        p.data_offset = offsets[idx];
1277        // RAM-mode (QSPI PSRAM) config — sentinels/zero for plain flash.
1278        let ram = flash_ram_params(m);
1279        p.writable = ram.writable;
1280        p.enter_qpi_cmd = ram.enter_qpi_cmd;
1281        p.quad_write_cmd = ram.quad_write_cmd;
1282        p.qpi_read_dummy = ram.qpi_read_dummy;
1283    }
1284    let flash_model_params = struct_to_dev(&model, device);
1285
1286    // Concatenated backing store. Per-instance fill: 0xFF (erased flash) or, for
1287    // a writable QSPI PSRAM, 0x00 (power-on, cocotb `bytearray(size)`). Firmware
1288    // (if any) is overlaid at each instance's `firmware_offset` afterwards.
1289    let mut firmware = vec![0xFFu8; total];
1290    for (idx, m) in qspi.iter().enumerate() {
1291        if m.writable {
1292            let base = offsets[idx] as usize;
1293            firmware[base..base + sizes[idx]].fill(0x00);
1294        }
1295    }
1296    for (idx, m) in qspi.iter().enumerate() {
1297        if let Some(fw_path) = m.firmware.as_ref() {
1298            use std::io::Read;
1299            let mut file = std::fs::File::open(std::path::Path::new(fw_path))
1300                .expect("Failed to open firmware file");
1301            let mut data = Vec::new();
1302            file.read_to_end(&mut data)
1303                .expect("Failed to read firmware");
1304            let within = m.firmware_offset;
1305            assert!(
1306                within + data.len() <= sizes[idx],
1307                "Firmware too large for QSPI memory {idx} backing store"
1308            );
1309            let abs = offsets[idx] as usize + within;
1310            firmware[abs..abs + data.len()].copy_from_slice(&data);
1311            clilog::info!(
1312                "Loaded {} bytes firmware into QSPI memory {} at abs 0x{:X}",
1313                data.len(),
1314                idx,
1315                abs
1316            );
1317        }
1318    }
1319    let flash_data = host_bytes_to_dev(firmware, device);
1320
1321    (
1322        flash_state,
1323        flash_din_params,
1324        flash_model_params,
1325        flash_data,
1326    )
1327}

1424
1425/// Per-tick edge activity for each clock domain.
1426#[derive(Clone, Debug, PartialEq, Eq, Hash)]
1427struct TickEdges {
1428    /// For each domain index: (has_falling_edge, has_rising_edge).
1429    domain_edges: Vec<(bool, bool)>,
1430}
1431
1432/// Schedules clock edges across multiple domains at GCD granularity.
1433///
1434/// For N clock domains with half-periods H1, H2, ..., the GCD tick is
1435/// `gcd(H1, H2, ...)` and the schedule repeats every `lcm(H1, H2, ...) / gcd` ticks.
1436/// Each tick in the schedule records which domains have falling/rising edges.
1437struct MultiClockScheduler {
1438    /// GCD of all half-periods in picoseconds. One scheduler edge advances
1439    /// simulated time by `gcd_ps`.
1440    gcd_ps: u64,
1441    /// One entry per GCD tick in the LCM cycle.
1442    schedule: Vec<TickEdges>,
1443}
1444
1445/// Compute GCD of two numbers.
1446fn gcd(a: u64, b: u64) -> u64 {
1447    if b == 0 {
1448        a
1449    } else {
1450        gcd(b, a % b)
1451    }
1452}
1453
1454/// Compute LCM of two numbers.
1455fn lcm(a: u64, b: u64) -> u64 {
1456    a / gcd(a, b) * b
1457}
1458
1459/// A clock domain's timing parameters for scheduling.
1460struct ClockDomainTiming {
1461    /// Half-period in picoseconds (period_ps / 2).
1462    half_period_ps: u64,
1463    /// Phase offset in picoseconds.
1464    phase_offset_ps: u64,
1465    /// Index into GpioMapping::clock_domains.
1466    domain_index: usize,
1467}
1468
1469impl MultiClockScheduler {
1470    /// Build a scheduler from clock domain timings.
1471    ///
1472    /// For a single clock with no phase offset, produces a schedule of length 2
1473    /// alternating between falling (posedge_flag=0) and rising (posedge_flag=1)
1474    /// edges.
1475    fn new(timings: &[ClockDomainTiming]) -> Self {
1476        assert!(!timings.is_empty(), "At least one clock domain is required");
1477
1478        // Compute GCD of all half-periods and phase offsets
1479        let mut gcd_ps = timings[0].half_period_ps;
1480        for t in &timings[1..] {
1481            gcd_ps = gcd(gcd_ps, t.half_period_ps);
1482        }
1483        // Include phase offsets in GCD (non-zero offsets need finer granularity)
1484        for t in timings {
1485            if t.phase_offset_ps > 0 {
1486                gcd_ps = gcd(gcd_ps, t.phase_offset_ps);
1487            }
1488        }
1489
1490        // Compute LCM of all full periods (= 2 * half_period)
1491        // We need a full period to capture both falling and rising edges.
1492        let mut lcm_ps = timings[0].half_period_ps * 2;
1493        for t in &timings[1..] {
1494            lcm_ps = lcm(lcm_ps, t.half_period_ps * 2);
1495        }
1496
1497        let schedule_len = (lcm_ps / gcd_ps) as usize;
1498        assert!(
1499            schedule_len <= 1_000_000,
1500            "Multi-clock schedule too large ({} ticks). \
1501             Clock periods may not be commensurable at this resolution.",
1502            schedule_len
1503        );
1504
1505        let num_domains = timings
1506            .iter()
1507            .map(|t| t.domain_index + 1)
1508            .max()
1509            .unwrap_or(0);
1510        let mut schedule = Vec::with_capacity(schedule_len);
1511
1512        for tick in 0..schedule_len {
1513            let tick_ps = tick as u64 * gcd_ps;
1514            let mut domain_edges = vec![(false, false); num_domains];
1515
1516            for timing in timings.iter() {
1517                let hp = timing.half_period_ps;
1518                let offset = timing.phase_offset_ps;
1519                // Adjust tick_ps by phase offset
1520                if tick_ps < offset {
1521                    continue;
1522                }
1523                let adjusted = tick_ps - offset;
1524                if adjusted % hp != 0 {
1525                    continue;
1526                }
1527                let edge_count = adjusted / hp;
1528                // Index by domain_index (position in clock_domains) so
1529                // build_edge_ops can iterate clock_domains directly.
1530                if edge_count % 2 == 0 {
1531                    domain_edges[timing.domain_index].0 = true; // falling edge
1532                } else {
1533                    domain_edges[timing.domain_index].1 = true; // rising edge
1534                }
1535            }
1536
1537            schedule.push(TickEdges { domain_edges });
1538        }
1539
1540        clilog::info!(
1541            "MultiClockScheduler: gcd={}ps, schedule_len={} ticks, {} domains",
1542            gcd_ps,
1543            schedule_len,
1544            num_domains
1545        );
1546
1547        Self { gcd_ps, schedule }
1548    }
1549
1550    /// Build BitOps for a single scheduler edge.
1551    ///
1552    /// Handles all clock domains' edges at this tick uniformly: a domain with
1553    /// a falling edge sets its clk input low and asserts negedge_flag; a domain
1554    /// with a rising edge sets its clk input high and asserts posedge_flag.
1555    /// Domains with no edge at this tick leave their clk input untouched and
1556    /// deassert both flags.
1557    fn build_edge_ops(
1558        &self,
1559        tick_idx: usize,
1560        gpio_map: &GpioMapping,
1561        reset_gpio: usize,
1562        reset_val: u8,
1563        constant_inputs: &HashMap<String, u8>,
1564        constant_ports: &HashMap<String, u8>,
1565        model_driven_positions: &[u32],
1566    ) -> Vec<BitOp> {
1567        let tick = &self.schedule[tick_idx % self.schedule.len()];
1568        let mut ops = Vec::new();
1569
1570        // Reset
1571        ops.push(BitOp {
1572            position: gpio_map.input_bits[&reset_gpio],
1573            value: reset_val as u32,
1574        });
1575
1576        // Per-domain clock + edge flags
1577        for (i, domain) in gpio_map.clock_domains.iter().enumerate() {
1578            let (has_fall, has_rise) = tick.domain_edges[i];
1579
1580            // Set clock input. Falling and rising at the same scheduler tick
1581            // can't both be true for one domain (they're encoded as different
1582            // edge_count parities), so the order doesn't matter — but
1583            // defensively: rising wins, since posedge eval is what DFFs sample.
1584            if has_fall {
1585                if let Some(pos) = domain.clock_input_pos {
1586                    ops.push(BitOp {
1587                        position: pos,
1588                        value: 0,
1589                    });
1590                }
1591            }
1592            if has_rise {
1593                if let Some(pos) = domain.clock_input_pos {
1594                    ops.push(BitOp {
1595                        position: pos,
1596                        value: 1,
1597                    });
1598                }
1599            }
1600
1601            // Edge flags: posedge for domains rising at this edge, negedge for
1602            // falling, both deasserted otherwise.
1603            for &pos in &domain.posedge_flag_bits {
1604                ops.push(BitOp {
1605                    position: pos,
1606                    value: if has_rise { 1 } else { 0 },
1607                });
1608            }
1609            for &pos in &domain.negedge_flag_bits {
1610                ops.push(BitOp {
1611                    position: pos,
1612                    value: if has_fall { 1 } else { 0 },
1613                });
1614            }
1615        }
1616
1617        // Constant inputs
1618        for (gpio_str, val) in constant_inputs {
1619            if let Ok(gpio_idx) = gpio_str.parse::<usize>() {
1620                if let Some(&pos) = gpio_map.input_bits.get(&gpio_idx) {
1621                    ops.push(BitOp {
1622                        position: pos,
1623                        value: *val as u32,
1624                    });
1625                }
1626            }
1627        }
1628        // Named constant ports
1629        for (port_name, val) in constant_ports {

1654/// Direct AIG evaluator: evaluates all AIG pins by traversing the AND-gate
1655/// tree directly, bypassing the boomerang/FlattenedScript entirely.
1656/// Returns a map from AIG pin → value (0 or 1).
1657///
1658/// This is used to diagnose whether the AIG construction is correct
1659/// (if direct eval matches FlattenedScript → bug is in AIG; if they differ → bug in flattening).
1660fn eval_aig_direct(aig: &AIG, input_state: &[u32], script: &FlattenedScriptV1) -> Vec<u8> {
1661    let mut pin_values = vec![0u8; aig.num_aigpins + 1];
1662    // pin 0 = constant 0 (Tie0)
1663
1664    // Initialize all primary inputs from input_state using input_map
1665    for (&aigpin, &pos) in &script.input_map {
1666        let word = (pos / 32) as usize;
1667        let bit = pos & 31;
1668        if word < input_state.len() {
1669            pin_values[aigpin] = ((input_state[word] >> bit) & 1) as u8;
1670        }
1671    }
1672
1673    // Topological traversal using topo_traverse_generic
1674    // endpoints=None means traverse ALL AIG pins
1675    // is_primary_input = set of pins that are inputs (InputPort, InputClockFlag, DFF Q, SRAM)
1676    let primary_input_set: indexmap::IndexSet<usize> = script.input_map.keys().copied().collect();
1677    let order = aig.topo_traverse_generic(None, Some(&primary_input_set));
1678
1679    // Evaluate each AND gate in topological order
1680    for &pin in &order {
1681        if primary_input_set.contains(&pin) {
1682            // Already initialized from input_state
1683            continue;
1684        }
1685        if let DriverType::AndGate(a_iv, b_iv) = aig.drivers[pin] {
1686            let a_pin = a_iv >> 1;
1687            let a_inv = (a_iv & 1) as u8;
1688            let b_pin = b_iv >> 1;
1689            let b_inv = (b_iv & 1) as u8;
1690            let a_val = if a_pin == 0 {
1691                0
1692            } else {
1693                pin_values[a_pin] ^ a_inv
1694            };
1695            let b_val = if b_pin == 0 {
1696                0
1697            } else {
1698                pin_values[b_pin] ^ b_inv
1699            };
1700            pin_values[pin] = a_val & b_val;
1701        }
1702        // Non-AndGate, non-primary-input pins (shouldn't happen in traversal) stay 0
1703    }
1704
1705    pin_values
1706}

1708/// Compare direct AIG evaluation with FlattenedScript output at a given tick.
1709/// Prints diagnostic information about any mismatches.
1710fn compare_aig_vs_flattened(
1711    aig: &AIG,
1712    input_state: &[u32],
1713    output_state: &[u32],
1714    script: &FlattenedScriptV1,
1715    state_size: usize,
1716    tick: usize,
1717    netlistdb: Option<&NetlistDB>,
1718) {
1719    let pin_values = eval_aig_direct(aig, input_state, script);
1720
1721    // Compare DFF D inputs: direct AIG eval vs FlattenedScript output
1722    let mut dff_mismatch_count = 0;
1723    let mut dff_match_count = 0;
1724    let mut dff_en_active_count = 0;
1725    let mut dff_d_ones_direct = 0;
1726    let mut dff_d_ones_flat = 0;
1727    let mut first_mismatches: Vec<String> = Vec::new();
1728
1729    for (_cellid, dff) in aig.dffs.iter() {
1730        let d_pin = dff.d_iv >> 1;
1731        let d_inv = (dff.d_iv & 1) as u8;
1732        let en_pin = dff.en_iv >> 1;
1733        let en_inv = (dff.en_iv & 1) as u8;
1734
1735        // Direct AIG evaluation
1736        let d_val_direct = if d_pin == 0 {
1737            0
1738        } else {
1739            pin_values[d_pin] ^ d_inv
1740        };
1741        let en_val_direct = if en_pin == 0 {
1742            0
1743        } else {
1744            pin_values[en_pin] ^ en_inv
1745        };
1746
1747        if en_val_direct != 0 {
1748            dff_en_active_count += 1;
1749        }
1750        if d_val_direct != 0 {
1751            dff_d_ones_direct += 1;
1752        }
1753
1754        // FlattenedScript result: read from output_state
1755        if let Some(&pos) = script.output_map.get(&dff.d_iv) {
1756            let word = (pos / 32) as usize;
1757            let bit = pos & 31;
1758            let flat_val = if word < state_size {
1759                ((output_state[word] >> bit) & 1) as u8
1760            } else {
1761                0
1762            };
1763            if flat_val != 0 {
1764                dff_d_ones_flat += 1;
1765            }
1766
1767            if d_val_direct != flat_val {
1768                dff_mismatch_count += 1;
1769                if first_mismatches.len() < 10 {
1770                    first_mismatches.push(format!(
1771                        "DFF d_iv={} pos={} en={}: direct={} flat={} (d_pin={} en_pin={} en_val={})",
1772                        dff.d_iv, pos, dff.en_iv, d_val_direct, flat_val, d_pin, en_pin, en_val_direct
1773                    ));
1774                }
1775            } else {
1776                dff_match_count += 1;
1777            }
1778        }
1779    }
1780
1781    // Also compare primary outputs
1782    let mut po_mismatch_count = 0;
1783    for (&aigpin_iv, &pos) in &script.output_map {
1784        let pin = aigpin_iv >> 1;
1785        let inv = (aigpin_iv & 1) as u8;
1786        let direct_val = if pin == 0 { 0u8 } else { pin_values[pin] ^ inv };
1787
1788        let word = (pos / 32) as usize;
1789        let bit = pos & 31;
1790        let flat_val = if word < state_size {
1791            ((output_state[word] >> bit) & 1) as u8
1792        } else {
1793            0
1794        };
1795
1796        if direct_val != flat_val {
1797            po_mismatch_count += 1;
1798        }
1799    }
1800
1801    eprintln!("=== AIG vs FlattenedScript comparison at tick {} ===", tick);
1802    eprintln!(
1803        "  DFFs: {} match, {} MISMATCH (out of {} total)",
1804        dff_match_count,
1805        dff_mismatch_count,
1806        aig.dffs.len()
1807    );
1808    eprintln!(
1809        "  DFF en_active={}, d_ones_direct={}, d_ones_flat={}",
1810        dff_en_active_count, dff_d_ones_direct, dff_d_ones_flat
1811    );
1812    eprintln!("  All output_map entries: {} mismatches", po_mismatch_count);
1813    for m in &first_mismatches {
1814        eprintln!("  MISMATCH: {}", m);
1815    }
1816
1817    // Also check: how many DFF Q values are 1 in the input state?
1818    let mut q_ones = 0;
1819    let mut q_hash: u64 = 0;
1820    let mut q_one_entries: Vec<(usize, u32, String)> = Vec::new(); // (cellid, pos, name)
1821    for (&cellid, dff) in aig.dffs.iter() {
1822        if let Some(&pos) = script.input_map.get(&dff.q) {
1823            let word = (pos / 32) as usize;
1824            let bit = pos & 31;
1825            if word < input_state.len() && ((input_state[word] >> bit) & 1) != 0 {
1826                q_ones += 1;
1827                let name = if let Some(ndb) = netlistdb {
1828                    format!("{:?}", ndb.cellnames[cellid])
1829                } else {
1830                    format!("cell_{}", cellid)
1831                };
1832                q_one_entries.push((cellid, pos, name));
1833                q_hash = q_hash
1834                    .wrapping_mul(6364136223846793005)
1835                    .wrapping_add(pos as u64);
1836            }
1837        }
1838    }
1839    q_one_entries.sort_by_key(|&(cellid, _, _)| cellid);
1840    eprintln!(
1841        "  DFF Q values = 1 in input_state: {}/{} (hash=0x{:016X})",
1842        q_ones,
1843        aig.dffs.len(),
1844        q_hash
1845    );
1846    if q_ones <= 100 {
1847        for (cellid, pos, name) in &q_one_entries {
1848            eprintln!("  Q=1: cell {} pos={} = {}", cellid, pos, name);
1849        }
1850    }
1851}

2082/// Tracks a clock domain whose input pin is driven by a peripheral model
2083/// (e.g. JTAG TCK), so model transitions can be reflected into the schedule's
2084/// pre-computed posedge/negedge flags.
2085struct ModelDrivenClockState {
2086    clock_input_pos: u32,
2087    posedge_flag_positions: Vec<u32>,
2088    negedge_flag_positions: Vec<u32>,
2089    prev_value: u8,

2092/// After model overrides are applied, patch the edge flags in the current
2093/// tick's ops to reflect the model's actual clock transitions.
2094fn patch_model_clock_edges(
2095    backend: &mut dyn CosimBackend,
2096    clocks: &mut [ModelDrivenClockState],
2097    overrides: &crate::sim::models::ModelOverrides,
2098    sched_offset: usize,
2099) {
2100    if clocks.is_empty() {
2101        return;
2102    }
2103    let sched_idx = sched_offset % backend.edges_per_period();
2104    let ops = backend.edge_ops_mut(sched_idx);
2105
2106    for clock in clocks.iter_mut() {
2107        let new_val = overrides
2108            .get(&clock.clock_input_pos)
2109            .copied()
2110            .unwrap_or(clock.prev_value);
2111        let rising = clock.prev_value == 0 && new_val == 1;
2112        let falling = clock.prev_value == 1 && new_val == 0;
2113        clock.prev_value = new_val;
2114
2115        for op in ops.iter_mut() {
2116            for &pos in &clock.posedge_flag_positions {
2117                if op.position == pos {
2118                    op.value = if rising { 1 } else { 0 };
2119                }
2120            }
2121            for &pos in &clock.negedge_flag_positions {
2122                if op.position == pos {
2123                    op.value = if falling { 1 } else { 0 };
2124                }
2125            }
2126        }
2127    }
2128}

2130/// Backend-generic co-simulation driver (ADR 0017). Drives the entire run
2131/// through the [`CosimBackend`] trait — `B::new` constructs the backend, every
2132/// per-tick interaction is a trait method, and no concrete `MetalBackend` /
2133/// `metal::` token appears in this body. The public [`run_cosim`] shim picks
2134/// `MetalBackend`; a future CPU backend monomorphises the same code path.
2135fn run_cosim_generic<B: CosimBackend>(
2136    design: &mut LoadedDesign,
2137    config: &TestbenchConfig,
2138    opts: &CosimOpts,
2139    timing_constraints: &Option<Vec<u32>>,
2140) -> CosimResult {
2141    let script = &design.script;
2142    let aig = &design.aig;
2143    let netlistdb = &design.netlistdb;
2144    let num_blocks = script.num_blocks;
2145    let num_major_stages = script.num_major_stages;
2146    // When timing arrivals are enabled, each state slot includes the arrival section.
2147    let state_size = script.effective_state_size() as usize;
2148    let arrival_state_offset = script.arrival_state_offset;
2149
2150    // ── Build GPIO mapping ───────────────────────────────────────────────
2151
2152    let gpio_map = build_gpio_mapping(aig, netlistdb, script, config.port_mapping.as_ref());
2153
2154    // ── Agnostic flash const-params (P1.3b-i) ────────────────────────────
2155    //
2156    // The flash MISO bit positions and clk/csn output positions are
2157    // const-after-init in the backend's flash-buffer setup
2158    // (FlashModelParams / FlashDinParams). Recompute them here as
2159    // backend-agnostic locals so the cosim loop's flash diagnostics don't
2160    // read the GPU param buffers. Bit-identical to the buffer init formulas.
2161    // These backend-agnostic locals feed the cosim loop's flash *diagnostics*
2162    // (verbose logging + the `--check-with-cpu` inline reference, both
2163    // single-instance). They describe the primary (instance 0) QSPI memory; the
2164    // plural per-instance stepping lives inside `CpuBackend::run_edges`.
2165    let qspi_primary = config.effective_qspi_memory().into_iter().next();
2166    let flash_clk_out_pos: u32 = qspi_primary
2167        .as_ref()

4376// decoder (5b) and bus-trace beat extraction (5c) are stubbed (the `drain_*`
4377// methods return empty, `new` returns empty `bus_lanes`).
4378//
4379// Interior mutability: the trait's `run_edges(&self)` and `vcd_snapshot(&self)`
4380// take a shared ref (Metal's buffers are interior-mutable shared memory). The
4381// CpuBackend mirrors that with `UnsafeCell` over its `Vec`s — sound because
4382// cosim dispatch is strictly sequential (no concurrent access to a backend),
4383// and `state_mut(&mut self)` keeps the compile-time exclusivity the seam
4384// promises for the pre-loop stimulus deposits.
4385use std::cell::UnsafeCell;

4387/// Per-channel UART-TX decoder config, mirroring Metal `UartPerChannelConfig`
4388/// (`tx_out_pos`, `cycles_per_bit`) built identically to
4389/// `MetalBackend::build_io_buffers`. `cycles_per_bit` is already scaled to
4390/// scheduler edges (`cpb * sched_ticks_per_sys_clk_cycle`).
4391struct CpuUartConfig {
4392    tx_out_pos: u32,
4393    cycles_per_bit: u32,
4394}

4396/// Per-channel UART-TX decoder FSM state, mirroring Metal `UartDecoderState`.
4397/// Persistent across edges/batches (like the GPU `uart_state` buffer). Ported
4398/// verbatim from `gpu_io_step` (kernel_v1.metal:1189-1249): a 4-state decoder
4399/// (0=idle, 1=start, 2=data, 3=stop) advancing `current_cycle` by 1 per edge.
4400struct CpuUartDecoderState {
4401    state: u32,
4402    last_tx: u32,
4403    start_cycle: u32,
4404    bits_received: u32,
4405    value: u32,
4406    current_cycle: u32,
4407}

4414            last_tx: 1, // TX line idle high
4415            start_cycle: 0,
4416            bits_received: 0,
4417            value: 0,
4418            current_cycle: 0,

4423/// One CPU-side QSPI-memory oracle instance (Stage A plurality).
4424///
4425/// The single-flash path was a flat set of `flash_*` fields on `CpuBackend`;
4426/// making QSPI memory plural (`Vec<QspiMemoryConfig>`) means each configured
4427/// instance gets its own `CppSpiFlash` model (its own backing buffer), its own
4428/// resolved pin positions, and its own per-edge decode state. `run_edges` steps
4429/// every instance independently, routing each one's pins in isolation, so two
4430/// memories with different firmware read back independent data.
4431struct CpuFlashInstance {
4432    /// `CppSpiFlash` model with this instance's own backing buffer. `UnsafeCell`
4433    /// because `CppSpiFlash::step` is `&mut` but `run_edges` is `&self`
4434    /// (dispatch is sequential).
4435    model: UnsafeCell<CppSpiFlash>,
4436    /// Pin positions resolved in `new` (mirror `build_flash_buffers`):
4437    /// clk/csn/d_out are OUTPUT-slot bit positions, d_in are INPUT-slot (MISO).
4438    clk_out_pos: u32,
4439    csn_out_pos: u32,
4440    d_out_pos: [u32; 4],
4441    d_in_pos: [u32; 4],
4442    /// Current MISO nibble injected by `apply_flash_din` (mirrors `FlashState.d_i`).
4443    d_i: UnsafeCell<u8>,
4444    /// Delayed (previous-edge) output csn / d_out for the setup-delay dual-step
4445    /// (mirror `FlashState.prev_csn` / `prev_d_out`).
4446    prev_csn: UnsafeCell<u32>,
4447    prev_d_out: UnsafeCell<u8>,
4448}

4457    sram_xmask: UnsafeCell<Vec<u32>>,

4459    schedule: Vec<Vec<BitOp>>,
4460    edges_per_period: usize,
4461    gcd_ps: u64,
4462    /// Words per state slot (`effective_state_size()`).
4463    state_size: usize,
4464    /// True SRAM word count (`sram_storage_size`); `sram()` view length.
4465    sram_len: usize,
4466    num_blocks: usize,
4467    num_major_stages: usize,
4468    xprop_enabled: bool,

4470    /// else 0). Mirrors the GPU `state_prep` `xmask_state_offset` sentinel.
4471    xmask_state_offset: usize,
4472    /// Reg/IO words per slot (`reg_io_state_size`); the X-mask lane width.

4474    /// Flash reset line (logic fixtures have no flash; carried for the contract).

4477    /// (no lifetime entanglement with `LoadedDesign`).
4478    blocks_start: Vec<usize>,
4479    blocks_data: Vec<u32>,
4480    /// Per-edge `[input | output]` snapshots for the most recent `run_edges`,
4481    /// one slot per edge in the batch (N=1 per edge). `None` until enabled.

4531impl CosimBackend for CpuBackend {
4532    fn new(
4533        aig: &AIG,
4534        netlistdb: &NetlistDB,
4535        script: &FlattenedScriptV1,
4536        config: &TestbenchConfig,
4537        gpio_map: &GpioMapping,
4538        uart_configs: &[(String, usize, usize, u32)],
4539        sched_ticks_per_sys_clk_cycle: u64,
4540        state_size: usize,
4541        num_blocks: usize,
4542        num_major_stages: usize,
4543        _arrival_state_offset: u32,
4544        _timing_constraints: &Option<Vec<u32>>,
4545    ) -> (Self, Vec<BusTraceLane>) {
4546        // Phase 1 scope guards (plan Risks): the CPU stepper has no arrival
4547        // readback, and `simulate_block_v1` carries no SRAM X-mask, so SRAM
4548        // under xprop would read as always-known.
4549        assert!(
4550            !script.timing_arrivals_enabled,
4551            "CpuBackend: timed cosim not supported (Phase 1) — arrival readback \
4552             rides the GPU ring; use the Metal backend for --timing-vcd cosim."
4553        );
4554        assert!(
4555            !(script.xprop_enabled && script.sram_storage_size > 0),
4556            "CpuBackend: --xprop with SRAM not supported (Phase 1) — \
4557             cpu_reference::simulate_block_v1 has no SRAM X-mask, so SRAM cells \
4558             would read as always-known; use the Metal backend."
4559        );
4560
4561        // Design state: [input | output], zeroed, with the xprop X-mask seed
4562        // mirroring MetalBackend::build_state_buffers (both slots seeded so
4563        // edge-0 state_prep's output→input copy doesn't wipe the seed).
4564        let mut state = vec![0u32; 2 * state_size];
4565        if script.xprop_enabled {
4566            let rio = script.reg_io_state_size as usize;
4567            let xmask = crate::sim::vcd_io::xprop_xmask_template_cosim(script);
4568            state[rio..2 * rio].copy_from_slice(&xmask);
4569            state[state_size + rio..state_size + 2 * rio].copy_from_slice(&xmask);
4570            clilog::info!(
4571                "cosim X-propagation enabled (CpuBackend): {} reg/io words, \
4572                 X-mask seeded (both slots) for uninitialised state",
4573                rio
4574            );
4575        }
4576
4577        // SRAM backing store (sized max(1) like Metal; sliced to the real len).
4578        let sram_alloc_len = (script.sram_storage_size as usize).max(1);
4579        let mut sram = vec![0u32; sram_alloc_len];
4580
4581        // SRAM X-mask shadow (all-X when xprop, sized 1 dummy otherwise). The
4582        // (xprop && sram>0) assert above means xprop with real SRAM never
4583        // reaches here; this stays for layout parity.
4584        let sram_xmask_len = if script.xprop_enabled {
4585            (script.sram_storage_size as usize).max(1)
4586        } else {
4587            1
4588        };
4589        let mut sram_xmask = vec![
4590            if script.xprop_enabled {
4591                0xFFFF_FFFFu32
4592            } else {
4593                0u32
4594            };
4595            sram_xmask_len
4596        ];
4597
4598        // SRAM preload (issue #80) — same path as build_state_buffers. The
4599        // logic fixtures declare no `sram_init`, so this is untaken; wired for
4600        // correctness.
4601        if let Some(sram_init) = config.sram_init.as_ref() {
4602            let elf_path = std::path::Path::new(&sram_init.elf_path);
4603            let chunks = crate::sim::sram_preload::parse_elf_chunks(elf_path)
4604                .unwrap_or_else(|e| panic!("sram_init: {e}"));
4605            let (cellid, storage_offset) =
4606                crate::sim::sram_preload::resolve_single_sram(&script.sram_cell_storage_offsets)
4607                    .unwrap_or_else(|e| panic!("sram_init: {e}"));
4608            let total_bytes: usize = chunks.iter().map(|c| c.bytes.len()).sum();
4609            let mut xmask_slice = if script.xprop_enabled {
4610                Some(sram_xmask.as_mut_slice())
4611            } else {
4612                None
4613            };
4614            crate::sim::sram_preload::apply_chunks(
4615                &mut sram,
4616                storage_offset,
4617                &chunks,
4618                xmask_slice.as_deref_mut(),
4619            )
4620            .unwrap_or_else(|e| panic!("sram_init: {e}"));
4621            clilog::info!(
4622                "SRAM preload (CpuBackend): {} bytes from {} → cell {} at storage offset {}",
4623                total_bytes,
4624                elf_path.display(),
4625                cellid,
4626                storage_offset
4627            );
4628        }
4629
4630        // Per-channel UART-TX decoder config + FSM state (Phase 1 step 5b).
4631        // Built EXACTLY as MetalBackend::build_io_buffers builds UartParams:
4632        //   tx_out_pos   = gpio_map.output_bits[tx_gpio] (or 0 if absent)
4633        //   cycles_per_bit = cpb * sched_ticks_per_sys_clk_cycle (edges, not
4634        //                    clock cycles — the decoder counts scheduler edges)
4635        // The uart_configs tuple is (name, tx_gpio, _, cpb).
4636        let cpu_uart_configs: Vec<CpuUartConfig> = uart_configs
4637            .iter()
4638            .map(|(_, tx_gpio, _, cpb)| CpuUartConfig {
4639                tx_out_pos: gpio_map.output_bits.get(tx_gpio).copied().unwrap_or(0),
4640                cycles_per_bit: cpb * sched_ticks_per_sys_clk_cycle as u32,
4641            })
4642            .collect();
4643        let cpu_uart_states: Vec<CpuUartDecoderState> = (0..uart_configs.len())
4644            .map(|_| CpuUartDecoderState::new())
4645            .collect();
4646
4647        // Per-bus APB3 pin positions + CPU decoder lanes (Phase 1 step 5c),
4648        // built EXACTLY as MetalBackend's build_bus_trace_params (which packs
4649        // the same positions into the GPU struct) via the shared builder.
4650        let (bus_positions, bus_lanes) =
4651            build_bus_trace(aig, netlistdb, script, config.effective_bus_traces());
4652
4653        // QSPI-memory CPU oracle (Stage A: plural). Build one CppSpiFlash per
4654        // configured memory + resolve each one's GPIO positions exactly as
4655        // MetalBackend::build_flash_buffers does for the single instance. Each
4656        // instance's MISO (d_i) starts 0x0F (idle high) — matching the GPU
4657        // FlashState init and the patched CppSpiFlash p_d_i default — so
4658        // instance 0 stays byte-identical to the single-flash GPU path through
4659        // reset and the command/address phases. Instances >0 are independent
4660        // oracles with their own backing buffers and pins.
4661        let flashes: Vec<CpuFlashInstance> = config
4662            .effective_qspi_memory()
4663            .iter()
4664            .enumerate()
4665            .map(|(idx, flash_cfg)| {
4666                let mut fl = CppSpiFlash::new(flash_cfg.backing_size_bytes());
4667                // RAM (QSPI PSRAM) mode: enable the writable store + enter-QPI /
4668                // quad-write commands so the CPU oracle matches the GPU kernel.
4669                // Zero-fills the store (before any firmware overlay below).
4670                if flash_cfg.writable {
4671                    fl.set_ram_mode(
4672                        true,
4673                        flash_cfg.enter_qpi_cmd,
4674                        flash_cfg.quad_write_cmd,
4675                        flash_cfg.read_dummy_cycles.unwrap_or(6),
4676                    );
4677                }
4678                if let Some(ref fw) = flash_cfg.firmware {
4679                    let firmware_path = std::path::Path::new(fw);
4680                    match fl.load_firmware(firmware_path, flash_cfg.firmware_offset) {
4681                        Ok(size) => clilog::info!(
4682                            "CpuBackend qspi_memory[{}]: loaded {} bytes firmware at offset 0x{:X}",
4683                            idx,
4684                            size,
4685                            flash_cfg.firmware_offset
4686                        ),
4687                        Err(e) => {
4688                            panic!("CpuBackend qspi_memory[{idx}]: failed to load firmware: {e}")
4689                        }
4690                    }
4691                }
4692                let clk_out_pos = gpio_map
4693                    .output_bits
4694                    .get(&flash_cfg.clk_gpio)
4695                    .copied()
4696                    .unwrap_or(0);
4697                let csn_out_pos = gpio_map
4698                    .output_bits
4699                    .get(&flash_cfg.csn_gpio)
4700                    .copied()
4701                    .unwrap_or(0);
4702                let mut d_out_pos = [0xFFFFFFFFu32; 4];
4703                let mut d_in_pos = [0xFFFFFFFFu32; 4];
4704                for i in 0..4 {
4705                    d_out_pos[i] = gpio_map
4706                        .output_bits
4707                        .get(&(flash_cfg.d0_gpio + i))
4708                        .copied()
4709                        .unwrap_or(0xFFFFFFFF);
4710                    d_in_pos[i] = gpio_map
4711                        .input_bits
4712                        .get(&(flash_cfg.d0_gpio + i))
4713                        .copied()
4714                        .unwrap_or(0xFFFFFFFF);
4715                }
4716                CpuFlashInstance {
4717                    model: UnsafeCell::new(fl),
4718                    clk_out_pos,
4719                    csn_out_pos,
4720                    d_out_pos,
4721                    d_in_pos,
4722                    // Idle MISO high (mirror GPU FlashState.d_i / patched CppSpiFlash).
4723                    d_i: UnsafeCell::new(0x0F),
4724                    // CSN starts high (deselected); d_out starts low (mirror FlashState).
4725                    prev_csn: UnsafeCell::new(1),
4726                    prev_d_out: UnsafeCell::new(0),
4727                }
4728            })
4729            .collect();
4730
4731        let backend = CpuBackend {
4732            state: UnsafeCell::new(state),
4733            sram: UnsafeCell::new(sram),
4734            sram_xmask: UnsafeCell::new(sram_xmask),
4735            schedule: Vec::new(),
4736            edges_per_period: 0,
4737            gcd_ps: 0,
4738            state_size,
4739            sram_len: script.sram_storage_size as usize,
4740            num_blocks,
4741            num_major_stages,
4742            xprop_enabled: script.xprop_enabled,
4743            xmask_state_offset: 0,
4744            reg_io_state_size: script.reg_io_state_size as usize,
4745            flash_in_reset: false,
4746            blocks_start: script.blocks_start.iter().copied().collect(),
4747            blocks_data: script.blocks_data.iter().copied().collect(),
4748            vcd_ring: UnsafeCell::new(None),
4749            enable_vcd: false,
4750            uart_configs: cpu_uart_configs,
4751            uart_states: UnsafeCell::new(cpu_uart_states),
4752            uart_decoded: UnsafeCell::new(Vec::new()),
4753            bus_positions,
4754            bus_prev_gate: UnsafeCell::new(0),
4755            bus_current_tick: UnsafeCell::new(0),
4756            bus_beats: UnsafeCell::new(Vec::new()),
4757            flashes,
4758        };
4759        (backend, bus_lanes)
4760    }

4868        // (mirrors the GPU ring drain in MetalBackend::drain_bus_beats; here the
4869        // beats are already in RawBeat form, captured in run_edges).
4870        std::mem::take(self.bus_beats.get_mut())
4871    }
4872
4873    fn run_edges(&self, batch: usize, schedule_offset: usize) -> u64 {
4874        let state_size = self.state_size;
4875        let state = self.state_inner_mut();
4876        // SAFETY: sequential dispatch — sram/sram_xmask/vcd_ring are only
4877        // touched here and in &mut-self methods, never concurrently.
4878        let sram = unsafe { &mut *self.sram.get() };
4879        let sram_xmask = unsafe { &mut *self.sram_xmask.get() };
4880        let ring = unsafe { &mut *self.vcd_ring.get() };
4881        // UART-TX decoder state + accumulator (Phase 1 step 5b). The FSM state
4882        // persists across edges/batches (mirrors the GPU `uart_state` buffer);
4883        // completed bytes accumulate until `drain_uart_tx`.
4884        let uart_states = unsafe { &mut *self.uart_states.get() };
4885        let uart_decoded = unsafe { &mut *self.uart_decoded.get() };
4886        // Bus-trace FSM state + accumulator (Phase 1 step 5c). `prev_gate` /
4887        // `current_tick` persist across edges/batches (mirror the GPU
4888        // `BusTraceChannel` header); beats accumulate until `drain_bus_beats`.
4889        let bus_prev_gate = unsafe { &mut *self.bus_prev_gate.get() };
4890        let bus_current_tick = unsafe { &mut *self.bus_current_tick.get() };
4891        let bus_beats = unsafe { &mut *self.bus_beats.get() };
4892
4893        if let Some(r) = ring.as_mut() {
4894            r.clear();
4895        }
4896
4897        let xmask_off = self.xmask_state_offset;
4898        let rio = self.reg_io_state_size;
4899
4900        for e in 0..batch {
4901            let sched_idx = (schedule_offset + e) % self.edges_per_period;
4902
4903            // ── CPU state_prep (mirrors kernel_v1.metal `state_prep`) ──────
4904            // Step 1: copy output slot → input slot.
4905            state.copy_within(state_size..2 * state_size, 0);
4906            // Step 2: apply this edge's BitOps to the input slot; driving a
4907            // bit clears its X-mask (xmask_off != 0 ⇒ xprop on).
4908            for op in &self.schedule[sched_idx] {
4909                let word_idx = (op.position >> 5) as usize;
4910                let bit_mask = 1u32 << (op.position & 31);
4911                if op.value != 0 {
4912                    state[word_idx] |= bit_mask;
4913                } else {
4914                    state[word_idx] &= !bit_mask;
4915                }
4916                if xmask_off != 0 {
4917                    state[xmask_off + word_idx] &= !bit_mask;
4918                }
4919            }
4920            // ── CPU apply_flash_din (mirror gpu_apply_flash_din, shader:904) ─
4921            // Inject each memory's current MISO nibble into ITS input-state
4922            // d_in bits BEFORE simulate, clearing their X-mask (driven ⇒ known,
4923            // #95 ph3). Every QSPI instance drives its own lanes independently.
4924            for inst in &self.flashes {
4925                // SAFETY: sequential dispatch — no concurrent borrow.
4926                let d_i = unsafe { *inst.d_i.get() };
4927                for i in 0..4usize {
4928                    let pos = inst.d_in_pos[i];
4929                    if pos == 0xFFFFFFFF {
4930                        continue;
4931                    }
4932                    let word_idx = (pos >> 5) as usize;
4933                    let bit_mask = 1u32 << (pos & 31);
4934                    if (d_i >> i) & 1 != 0 {
4935                        state[word_idx] |= bit_mask;
4936                    } else {
4937                        state[word_idx] &= !bit_mask;
4938                    }
4939                    if xmask_off != 0 {
4940                        state[xmask_off + word_idx] &= !bit_mask;
4941                    }
4942                }
4943            }
4944
4945            // ── Simulate every partition for this edge ─────────────────────
4946            for stage_i in 0..self.num_major_stages {
4947                for block_i in 0..self.num_blocks {
4948                    let start = self.blocks_start[stage_i * self.num_blocks + block_i];
4949                    let end = self.blocks_start[stage_i * self.num_blocks + block_i + 1];
4950                    if start == end {
4951                        continue;
4952                    }
4953                    let script_slice = &self.blocks_data[start..end];
4954                    let (input_half, output_half) = state.split_at_mut(state_size);
4955                    if self.xprop_enabled {
4956                        // Value + X-mask lanes are packed in the same slot at
4957                        // [0..rio] and [xmask_off..xmask_off+rio]. The xprop
4958                        // executor takes them as separate slices (it indexes
4959                        // xmasks from 0, not from xmask_off).
4960                        let (in_val, in_x) = input_half.split_at(xmask_off);
4961                        let (out_val, out_x) = output_half.split_at_mut(xmask_off);
4962                        crate::sim::cpu_reference::simulate_block_v1_xprop(
4963                            script_slice,
4964                            in_val,
4965                            out_val,
4966                            &in_x[..rio],
4967                            &mut out_x[..rio],
4968                            sram,
4969                            sram_xmask,
4970                            false,
4971                        );
4972                    } else {
4973                        crate::sim::cpu_reference::simulate_block_v1(
4974                            script_slice,
4975                            input_half,
4976                            output_half,
4977                            sram,
4978                            false,
4979                        );
4980                    }
4981                }
4982            }
4983
4984            // ── CPU flash_model_step (mirror gpu_flash_model_step, shader:943) ─
4985            // For EACH memory: read its clk/csn/d_out from the OUTPUT slot, then
4986            // dual-step ITS CppSpiFlash with delayed CSN (setup-delay model) to
4987            // compute the next MISO d_i. Runs after simulate (output fresh) on
4988            // every edge, mirroring the Metal `encode_flash_model_step` cadence
4989            // (before io_step / UART). Each instance keeps its own delayed
4990            // csn/d_out so the memories advance fully independently.
4991            for inst in &self.flashes {
4992                let read_out = |pos: u32| -> u32 {
4993                    if pos == 0xFFFFFFFF {
4994                        0
4995                    } else {
4996                        (state[state_size + (pos >> 5) as usize] >> (pos & 31)) & 1
4997                    }
4998                };
4999                let clk = read_out(inst.clk_out_pos);
5000                let csn = read_out(inst.csn_out_pos);
5001                let mut d_out = 0u8;
5002                for i in 0..4usize {
5003                    d_out |= (read_out(inst.d_out_pos[i]) as u8) << i;
5004                }
5005                // SAFETY: sequential dispatch — these cells are only touched here.
5006                let d_i_cell = unsafe { &mut *inst.d_i.get() };
5007                let prev_csn = unsafe { &mut *inst.prev_csn.get() };
5008                let prev_d_out = unsafe { &mut *inst.prev_d_out.get() };
5009                if self.flash_in_reset {
5010                    // Reset branch (shader:974): force MISO high, update the
5011                    // delayed values, do NOT step the model.
5012                    *d_i_cell = 0x0F;
5013                    *prev_csn = csn;
5014                    *prev_d_out = d_out;
5015                } else {
5016                    let fl = unsafe { &mut *inst.model.get() };
5017                    // Dual-step with delayed CSN (matches the GPU setup-delay
5018                    // model): step 1 = delayed csn + delayed d_out, step 2 =
5019                    // delayed csn + current d_out. CppSpiFlash::step does one
5020                    // eval()+commit() (CXXRTL agent.step semantics).
5021                    let pcsn = *prev_csn != 0;
5022                    fl.step(clk != 0, pcsn, *prev_d_out);
5023                    let d_i = fl.step(clk != 0, pcsn, d_out);
5024                    *d_i_cell = d_i;
5025                    *prev_csn = csn; // current output csn → next edge's delayed csn
5026                    *prev_d_out = d_out;
5027                }
5028            }
5029
5030            // ── UART-TX decoder (ported from gpu_io_step, kernel_v1.metal ──
5031            // 1189-1249). Runs AFTER `simulate` — same point the Metal
5032            // `encode_io_step` runs, with the output slot fresh — on EVERY
5033            // edge from tick 0 (including reset edges), `current_cycle` += 1
5034            // per edge, matching the GPU. READ_OUT_BIT reads the OUTPUT slot:
5035            // states[state_size + (pos>>5)] >> (pos&31) & 1; pos==u32::MAX → 0.
5036            for (ui, cfg) in self.uart_configs.iter().enumerate() {
5037                let cycles_per_bit = cfg.cycles_per_bit;
5038                let tx = if cfg.tx_out_pos != u32::MAX {
5039                    (state[state_size + (cfg.tx_out_pos >> 5) as usize] >> (cfg.tx_out_pos & 31))
5040                        & 1
5041                } else {
5042                    0
5043                };
5044
5045                let us = &mut uart_states[ui];
5046                let cycle = us.current_cycle;
5047                let mut st = us.state;
5048                let last_tx = us.last_tx;
5049                let mut start_cycle = us.start_cycle;
5050                let mut bits_received = us.bits_received;
5051                let mut value = us.value;
5052
5053                if st == 0 {
5054                    if last_tx == 1 && tx == 0 {
5055                        st = 1;
5056                        start_cycle = cycle;
5057                    }
5058                } else if st == 1 {
5059                    if cycle >= start_cycle + cycles_per_bit / 2 {
5060                        if tx == 0 {
5061                            st = 2;
5062                            start_cycle += cycles_per_bit;
5063                            bits_received = 0;
5064                            value = 0;
5065                        } else {
5066                            st = 0;
5067                        }
5068                    }
5069                } else if st == 2 {
5070                    let bit_center =
5071                        start_cycle + bits_received * cycles_per_bit + cycles_per_bit / 2;
5072                    if cycle >= bit_center {
5073                        value |= tx << bits_received;
5074                        if bits_received >= 7 {
5075                            st = 3;
5076                            start_cycle += 8 * cycles_per_bit;
5077                        } else {
5078                            bits_received += 1;
5079                        }
5080                    }
5081                } else if st == 3 {
5082                    if cycle >= start_cycle + cycles_per_bit / 2 {
5083                        if tx == 1 {
5084                            uart_decoded.push((ui, (value & 0xFF) as u8));
5085                        }
5086                        st = 0;
5087                    }
5088                }
5089
5090                us.state = st;
5091                us.last_tx = tx;
5092                us.start_cycle = start_cycle;
5093                us.bits_received = bits_received;
5094                us.value = value;
5095                us.current_cycle = cycle + 1;
5096            }
5097
5098            // ── Bus transaction trace (APB3, ported from gpu_io_step,
5099            // kernel_v1.metal:1305-1352). Runs AFTER `simulate`, same point as
5100            // the GPU block, reading the OUTPUT slot. Per bus: gate = sel & en &
5101            // rdy (rdy=1 when pready absent); on a rising edge of the gate emit
5102            // one RawBeat stamped with `current_tick`. `prev_gate` /
5103            // `current_tick` persist; `current_tick` += 1 once per edge from 0,
5104            // regardless of whether any bus fired. READ_OUT_BIT reads the OUTPUT
5105            // slot; pos==u32::MAX → 0 (sentinel short-circuit).
5106            if !self.bus_positions.is_empty() {
5107                let btick = *bus_current_tick;
5108                let prev = *bus_prev_gate;
5109                let mut new_gate = 0u32;
5110                let read_out = |pos: u32| -> u32 {
5111                    if pos != u32::MAX {
5112                        (state[state_size + (pos >> 5) as usize] >> (pos & 31)) & 1
5113                    } else {
5114                        0
5115                    }
5116                };
5117                for (b, bp) in self.bus_positions.iter().enumerate() {
5118                    let gate = if bp.protocol_apb3 {
5119                        let sel = read_out(bp.sel_pos);
5120                        let en = read_out(bp.enable_pos);
5121                        let rdy = if bp.ready_pos == 0xFFFFFFFF {
5122                            1
5123                        } else {
5124                            read_out(bp.ready_pos)
5125                        };
5126                        sel & en & rdy
5127                    } else {
5128                        0
5129                    };
5130                    new_gate |= gate << b;
5131
5132                    let rising = gate != 0 && ((prev >> b) & 1) == 0;
5133                    if rising {
5134                        let write = read_out(bp.write_pos);
5135                        let err = read_out(bp.resp_pos);
5136                        let mut addr = 0u32;
5137                        for i in 0..bp.addr_bits.min(BUS_TRACE_MAX_ADR_BITS) {
5138                            addr |= read_out(bp.addr_pos[i]) << i;
5139                        }
5140                        let mut wdata = 0u32;
5141                        let mut rdata = 0u32;
5142                        for i in 0..bp.data_bits.min(BUS_TRACE_MAX_DAT_BITS) {
5143                            wdata |= read_out(bp.wdata_pos[i]) << i;
5144                            rdata |= read_out(bp.rdata_pos[i]) << i;
5145                        }
5146                        // Pack the same flags the GPU writes, then unpack into a
5147                        // RawBeat identically to MetalBackend::drain_bus_beats.
5148                        let flags = (write & 1) | ((err & 1) << 1) | ((b as u32) << 8);
5149                        bus_beats.push(crate::sim::models::bus_trace::RawBeat {
5150                            tick: btick as u64,
5151                            bus_id: (flags >> 8) & 0xFF,
5152                            write: (flags & 1) != 0,
5153                            err: (flags >> 1) & 1 != 0,
5154                            addr: addr as u64,
5155                            wdata: wdata as u64,
5156                            rdata: rdata as u64,
5157                        });
5158                    }
5159                }
5160                *bus_prev_gate = new_gate;
5161                *bus_current_tick = btick + 1;
5162            }
5163
5164            // ── VCD snapshot: full [input | output] slot for this edge ─────
5165            if self.enable_vcd {
5166                if let Some(r) = ring.as_mut() {
5167                    r.push(state.clone());
5168                }
5169            }
5170        }
5171
5172        // CpuBackend runs synchronously; the token is unused (wait is a no-op).
5173        0
5174    }

5177        // Synchronous CPU backend — run_edges already completed.
5178    }
