97/// Max buses a single cosim run may trace (bounds both backends' per-bus arrays).
98pub(crate) const MAX_BUS_TRACES: usize = 4;
99/// Max address-bus width captured per beat.
100pub(crate) const BUS_TRACE_MAX_ADR_BITS: usize = 32;
101/// Max data-bus width captured per beat.
102pub(crate) const BUS_TRACE_MAX_DAT_BITS: usize = 32;

104/// Backend-agnostic per-bus pin positions resolved from the netlist. Plain
105/// Rust (no `repr(C)`, no `metal::`): the Metal backend packs these field-for-
106/// field into the GPU `BusTraceParams`; the CPU backend reads them directly in
107/// its bus-trace FSM. `0xFFFFFFFF` is the "unresolved / absent" sentinel,
108/// matching the GPU `READ_OUT_BIT` short-circuit.
109pub(crate) struct BusTracePositions {
110    /// True iff this bus decodes APB3 (the only protocol gated in Phase 1).
111    pub protocol_apb3: bool,
112    pub addr_bits: usize,
113    pub data_bits: usize,
114    pub sel_pos: u32,
115    pub enable_pos: u32,
116    pub ready_pos: u32,
117    pub write_pos: u32,
118    pub resp_pos: u32,
119    pub addr_pos: [u32; BUS_TRACE_MAX_ADR_BITS],
120    pub wdata_pos: [u32; BUS_TRACE_MAX_DAT_BITS],
121    pub rdata_pos: [u32; BUS_TRACE_MAX_DAT_BITS],
122}
123
124impl Default for BusTracePositions {
125    fn default() -> Self {
126        BusTracePositions {
127            protocol_apb3: true, // APB3 == GPU `protocol: 0`
128            addr_bits: 0,
129            data_bits: 0,
130            sel_pos: 0xFFFFFFFF,
131            enable_pos: 0xFFFFFFFF,
132            ready_pos: 0xFFFFFFFF,
133            write_pos: 0xFFFFFFFF,
134            resp_pos: 0xFFFFFFFF,
135            addr_pos: [0xFFFFFFFF; BUS_TRACE_MAX_ADR_BITS],
136            wdata_pos: [0xFFFFFFFF; BUS_TRACE_MAX_DAT_BITS],
137            rdata_pos: [0xFFFFFFFF; BUS_TRACE_MAX_DAT_BITS],
138        }
139    }

142/// Resolve every configured bus's pin positions + build its CPU-side decoder
143/// lane. Backend-agnostic (uses the agnostic `resolve_to_state_pos`); shared by
144/// the Metal backend (which packs the positions into the GPU `BusTraceParams`)
145/// and the CPU backend (which runs the beat FSM directly). Protocol-gated to
146/// APB3 — other protocols are skipped with a warning (Phase 2). The returned
147/// `Vec<BusTracePositions>` and `Vec<BusTraceLane>` are index-aligned: vec index
148/// == GPU `bus_id` (packed into the beat `flags>>8`).
149pub(crate) fn build_bus_trace(
150    aig: &AIG,
151    netlistdb: &NetlistDB,
152    script: &FlattenedScriptV1,
153    configs: &[crate::testbench::BusTraceConfig],
154) -> (Vec<BusTracePositions>, Vec<BusTraceLane>) {
155    use crate::sim::models::bus_trace::{pin_basename, BusTraceDecoder};
156    use crate::sim::trace_signals::resolve_to_state_pos;
157    use crate::testbench::BusProtocol;
158
159    let mut positions: Vec<BusTracePositions> = Vec::new();
160    let mut lanes: Vec<BusTraceLane> = Vec::new();
161
162    let resolve =
163        |name: &str| resolve_to_state_pos(aig, netlistdb, script, name).unwrap_or(0xFFFFFFFF);
164
165    for cfg in configs.iter() {
166        if positions.len() >= MAX_BUS_TRACES {
167            clilog::warn!(
168                "bus-trace: more than {} buses configured; `{}` and later ignored",
169                MAX_BUS_TRACES,
170                cfg.name
171            );
172            break;
173        }
174        match cfg.protocol {
175            BusProtocol::Apb3 => {}
176            other => {
177                clilog::warn!(
178                    "bus-trace `{}`: protocol {:?} not yet implemented (Phase 2); skipping",
179                    cfg.name,
180                    other
181                );
182                continue;
183            }
184        }
185
186        let addr_bits = cfg.addr_bits.min(BUS_TRACE_MAX_ADR_BITS);
187        let data_bits = cfg.data_bits.min(BUS_TRACE_MAX_DAT_BITS);
188        if cfg.addr_bits > BUS_TRACE_MAX_ADR_BITS || cfg.data_bits > BUS_TRACE_MAX_DAT_BITS {
189            clilog::warn!(
190                "bus-trace `{}`: addr/data width capped at {}/{} bits (Phase 1)",
191                cfg.name,
192                BUS_TRACE_MAX_ADR_BITS,
193                BUS_TRACE_MAX_DAT_BITS
194            );
195        }
196
197        let mut p = BusTracePositions {
198            protocol_apb3: true,
199            addr_bits,
200            data_bits,
201            sel_pos: resolve(&pin_basename(cfg, "psel")),
202            enable_pos: resolve(&pin_basename(cfg, "penable")),
203            ready_pos: resolve(&pin_basename(cfg, "pready")),
204            write_pos: resolve(&pin_basename(cfg, "pwrite")),
205            resp_pos: resolve(&pin_basename(cfg, "pslverr")),
206            ..Default::default()
207        };
208        let abase = pin_basename(cfg, "paddr");
209        for b in 0..addr_bits {
210            p.addr_pos[b] = resolve(&format!("{abase}[{b}]"));
211        }
212        let wbase = pin_basename(cfg, "pwdata");
213        let rbase = pin_basename(cfg, "prdata");
214        for b in 0..data_bits {
215            p.wdata_pos[b] = resolve(&format!("{wbase}[{b}]"));
216            p.rdata_pos[b] = resolve(&format!("{rbase}[{b}]"));
217        }
218
219        if p.sel_pos == 0xFFFFFFFF || p.enable_pos == 0xFFFFFFFF {
220            clilog::warn!(
221                "bus-trace `{}`: psel/penable did not resolve (prefix `{}`) — \
222                 this bus will not capture. Check the prefix / `signals` overrides.",
223                cfg.name,
224                cfg.prefix
225            );
226        } else {
227            let n_addr = p.addr_pos.iter().filter(|&&x| x != 0xFFFFFFFF).count();
228            clilog::info!(
229                "bus-trace `{}` (APB3): psel/penable resolved, addr {}/{} bits, \
230                 pready={} pslverr={}",
231                cfg.name,
232                n_addr,
233                addr_bits,
234                p.ready_pos != 0xFFFFFFFF,
235                p.resp_pos != 0xFFFFFFFF
236            );
237        }
238
239        positions.push(p);
240        lanes.push(BusTraceLane {
241            name: cfg.name.clone(),
242            decoder: BusTraceDecoder::new(cfg.protocol),
243        });
244    }
245
246    (positions, lanes)
247}

250/// Per-clock-domain flag bit positions in the packed state buffer.
251///
252/// Each clock domain (identified by its netlistdb pin ID) has its own set of
253/// posedge/negedge flag bits that gate DFF writeout for that domain.
254struct ClockDomainFlags {
255    /// Netlistdb pin ID for this clock (key from `clock_pin2aigpins`).
256    #[allow(dead_code)]
257    clock_pinid: usize,
258    /// Human-readable name from pin (e.g. "io$clk$i").
259    name: String,
260    /// State positions for this domain's posedge flags.
261    posedge_flag_bits: Vec<u32>,
262    /// State positions for this domain's negedge flags.
263    negedge_flag_bits: Vec<u32>,
264    /// State position of the clock input port itself (if it's a primary input).
265    clock_input_pos: Option<u32>,
266    /// GPIO index if this clock is driven from a GPIO.
267    clock_gpio: Option<usize>,
268}

270/// Maps GPIO pin indices to bit positions in the packed u32 state buffer.
271pub(crate) struct GpioMapping {
272    /// gpio_in[idx] → (aigpin, state bit position)
273    input_bits: HashMap<usize, u32>,
274    /// gpio_out[idx] → state bit position in output_map
275    output_bits: HashMap<usize, u32>,
276    /// Per-clock-domain flag bit positions (grouped by netlistdb pin ID).
277    clock_domains: Vec<ClockDomainFlags>,
278    /// Named input port → state bit position (for non-GPIO ports like por_l, resetb_h)
279    named_input_bits: HashMap<String, u32>,
280}

282/// Set a single bit in a packed u32 state buffer.
283#[inline]
284fn set_bit(state: &mut [u32], pos: u32, val: u8) {
285    let word = &mut state[(pos >> 5) as usize];
286    let mask = 1u32 << (pos & 31);
287    if val != 0 {
288        *word |= mask;
289    } else {
290        *word &= !mask;
291    }
292}
293
294/// Clear a single bit in a packed u32 state buffer.
295#[inline]
296fn clear_bit(state: &mut [u32], pos: u32) {
297    state[(pos >> 5) as usize] &= !(1u32 << (pos & 31));
298}

300/// Convert a 0/1 bit value to a VCD scalar value.
301#[inline]
302fn bit_to_vcd_value(bit: u8) -> vcd_ng::Value {
303    if bit != 0 {
304        vcd_ng::Value::V1
305    } else {
306        vcd_ng::Value::V0
307    }

310/// Build GPIO-to-state-buffer mapping from AIG + FlattenedScript.
311///
312/// When `port_mapping` is provided, maps GPIO indices to port names explicitly
313/// (for designs like ChipFlow that use named ports instead of gpio_in[N]/gpio_out[N]).
314/// Falls back to parsing gpio_in[N]/gpio_out[N] from pin names when no mapping given.
315pub(crate) fn build_gpio_mapping(
316    aig: &AIG,
317    netlistdb: &NetlistDB,
318    script: &FlattenedScriptV1,
319    port_mapping: Option<&PortMapping>,
320) -> GpioMapping {
321    let mut input_bits: HashMap<usize, u32> = HashMap::new();
322    let mut output_bits: HashMap<usize, u32> = HashMap::new();
323    let mut named_input_bits: HashMap<String, u32> = HashMap::new();
324
325    // Group clock flags by pinid → ClockDomainFlags.
326    // BTreeMap gives deterministic iteration order (ascending pinid).
327    let mut clock_domain_map: std::collections::BTreeMap<usize, (Vec<u32>, Vec<u32>)> =
328        std::collections::BTreeMap::new();
329
330    // Build reverse lookup: port_name → gpio_index for port_mapping mode
331    let input_name_to_gpio: HashMap<String, usize> = port_mapping
332        .map(|pm| {
333            pm.inputs
334                .iter()
335                .filter_map(|(k, v)| k.parse::<usize>().ok().map(|idx| (v.clone(), idx)))
336                .collect()
337        })
338        .unwrap_or_default();
339    let output_name_to_gpio: HashMap<String, usize> = port_mapping
340        .map(|pm| {
341            pm.outputs
342                .iter()
343                .filter_map(|(k, v)| k.parse::<usize>().ok().map(|idx| (v.clone(), idx)))
344                .collect()
345        })
346        .unwrap_or_default();
347
348    // Map input ports → state buffer positions
349    for (aigpin_idx, driv) in aig.drivers.iter().enumerate() {
350        match driv {
351            DriverType::InputPort(pinid) => {
352                let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
353                // Try port_mapping first, then fall back to gpio_in[N] parsing
354                let gpio_idx = input_name_to_gpio
355                    .get(&pin_name)
356                    .copied()
357                    .or_else(|| parse_gpio_index(&pin_name, "gpio_in"));
358                if let Some(gpio_idx) = gpio_idx {
359                    if let Some(&pos) = script.input_map.get(&aigpin_idx) {
360                        input_bits.insert(gpio_idx, pos);
361                        clilog::debug!(
362                            "input[{}] = pin '{}' → state pos {}",
363                            gpio_idx,
364                            pin_name,
365                            pos
366                        );
367                    }
368                }
369                // Also store by name for constant_ports
370                if let Some(&pos) = script.input_map.get(&aigpin_idx) {
371                    named_input_bits.insert(pin_name.clone(), pos);
372                }
373            }
374            DriverType::InputClockFlag(pinid, is_negedge) => {
375                if let Some(&pos) = script.input_map.get(&aigpin_idx) {
376                    let entry = clock_domain_map
377                        .entry(*pinid)
378                        .or_insert_with(|| (Vec::new(), Vec::new()));
379                    if *is_negedge == 0 {
380                        entry.0.push(pos);
381                    } else {
382                        entry.1.push(pos);
383                    }
384                    let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
385                    clilog::debug!(
386                        "ClockFlag aigpin={} pin={} (pinid={}) negedge={} pos={}",
387                        aigpin_idx,
388                        pin_name,
389                        pinid,
390                        is_negedge,
391                        pos
392                    );
393                }
394            }
395            _ => {}
396        }
397    }
398
399    // Reverse-lookup map for extra observables: name → aigpin. Used
400    // below to resolve top-level `inout` ports — those have
401    // Direction::Unknown so they don't appear in output_map under
402    // their own name, but the AIG-side bidir-pad observability code
403    // (see `aig.rs::resolve_bi_24t_observable_base`) registers the
404    // core-drive-out aigpin under the synthetic `<port>__out` name.
405    // Looking that up gives peripheral observers (UART RX decoder,
406    // GPIO output watchers) the right signal. See #79.
407    let mut extra_observable_by_name: HashMap<String, usize> = HashMap::new();
408    for (&aigpin, names) in aig.extra_observable_names.iter() {
409        for name in names {
410            extra_observable_by_name.insert(name.clone(), aigpin);
411        }
412    }
413
414    // Map output ports → state buffer positions
415    for i in netlistdb.cell2pin.iter_set(0) {
416        let pin_name = netlistdb.pinnames[i].dbg_fmt_pin();
417        // Try port_mapping first, then fall back to gpio_out[N] parsing
418        let gpio_idx = output_name_to_gpio
419            .get(&pin_name)
420            .copied()
421            .or_else(|| parse_gpio_index(&pin_name, "gpio_out"));
422        if let Some(gpio_idx) = gpio_idx {
423            let mut aigpin_iv = aig.pin2aigpin_iv[i];
424            // Top-level inout: pin2aigpin_iv on a wafer.space-style
425            // `bi_24t` pad resolves to the input-side `Y` aigpin
426            // (PAD→core), which is non-MAX but absent from
427            // output_map. The OUTPUT direction's aigpin (core `A`→PAD)
428            // is registered under `<pin>__out` in
429            // extra_observable_names; falling back unconditionally
430            // (when found) overrides the input-side resolution with
431            // the actual output driver. Safe when the synthetic name
432            // is absent: aigpin_iv stays as-is and we fall through to
433            // the existing diagnostic.
434            if netlistdb.pindirect[i] == Direction::Unknown {
435                let synth_name = format!("{pin_name}__out");
436                if let Some(&aigpin) = extra_observable_by_name.get(&synth_name) {
437                    aigpin_iv = aigpin;
438                    clilog::debug!(
439                        "output[{}] inout pin '{}' resolved via bidir-pad \
440                         observable '{}' → aigpin {}",
441                        gpio_idx,
442                        pin_name,
443                        synth_name,
444                        aigpin
445                    );
446                }
447            }
448            if aigpin_iv == usize::MAX {
449                clilog::info!(
450                    "output[{}] pin '{}' has no AIG connection (usize::MAX)",
451                    gpio_idx,
452                    pin_name
453                );
454                continue;
455            }
456            if aigpin_iv <= 1 {
457                clilog::info!(
458                    "output[{}] pin '{}' is constant (aigpin_iv={})",
459                    gpio_idx,
460                    pin_name,
461                    aigpin_iv
462                );
463                continue;
464            }
465            if let Some(&pos) = script.output_map.get(&aigpin_iv) {
466                output_bits.insert(gpio_idx, pos);
467                clilog::debug!(
468                    "output[{}] = pin '{}' → state pos {}",
469                    gpio_idx,
470                    pin_name,
471                    pos
472                );
473            } else if let Some(&pos) = script.output_map.get(&(aigpin_iv ^ 1)) {
474                output_bits.insert(gpio_idx, pos);
475                clilog::debug!(
476                    "output[{}] = pin '{}' → state pos {} (flipped inv)",
477                    gpio_idx,
478                    pin_name,
479                    pos
480                );
481            } else {
482                clilog::info!(
483                    "output[{}] pin '{}' aigpin_iv={} (aigpin={}) not in output_map (dir={:?})",
484                    gpio_idx,
485                    pin_name,
486                    aigpin_iv,
487                    aigpin_iv >> 1,
488                    netlistdb.pindirect[i]
489                );
490            }
491        }
492    }
493
494    // Build ClockDomainFlags from grouped clock flags
495    // Also resolve GPIO index for each clock domain by looking up the clock pin
496    // in the input_bits mapping.
497    let mut clock_domains: Vec<ClockDomainFlags> = Vec::new();
498    // Build reverse: pinid → (pin_name, gpio_idx, state_pos) from input_bits
499    let mut pinid_to_input_info: HashMap<usize, (String, Option<usize>, Option<u32>)> =
500        HashMap::new();
501    for (aigpin_idx, driv) in aig.drivers.iter().enumerate() {
502        if let DriverType::InputPort(pinid) = driv {
503            let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
504            let gpio_idx = input_name_to_gpio
505                .get(&pin_name)
506                .copied()
507                .or_else(|| parse_gpio_index(&pin_name, "gpio_in"));
508            let pos = script.input_map.get(&aigpin_idx).copied();
509            pinid_to_input_info.insert(*pinid, (pin_name, gpio_idx, pos));
510        }
511    }
512
513    for (pinid, (posedge_bits, negedge_bits)) in &clock_domain_map {
514        let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
515        // Try to find the GPIO index and state position for this clock pin
516        let (clock_gpio, clock_input_pos) =
517            if let Some((_, gpio_idx, pos)) = pinid_to_input_info.get(pinid) {
518                (*gpio_idx, *pos)
519            } else {
520                (None, None)
521            };
522        clock_domains.push(ClockDomainFlags {
523            clock_pinid: *pinid,
524            name: pin_name.clone(),
525            posedge_flag_bits: posedge_bits.clone(),
526            negedge_flag_bits: negedge_bits.clone(),
527            clock_input_pos,
528            clock_gpio,
529        });
530        clilog::info!(
531            "Clock domain '{}' (pinid={}): {} posedge flags, {} negedge flags, gpio={:?}",
532            pin_name,
533            pinid,
534            posedge_bits.len(),
535            negedge_bits.len(),
536            clock_gpio
537        );
538    }
539
540    let total_posedge: usize = clock_domains
541        .iter()
542        .map(|d| d.posedge_flag_bits.len())
543        .sum();
544    let total_negedge: usize = clock_domains
545        .iter()
546        .map(|d| d.negedge_flag_bits.len())
547        .sum();
548    clilog::info!(
549        "GPIO mapping: {} inputs, {} outputs, {} clock domains ({} posedge flags, {} negedge flags)",
550        input_bits.len(),
551        output_bits.len(),
552        clock_domains.len(),
553        total_posedge,
554        total_negedge
555    );
556
557    clilog::info!("Named input ports: {} mapped", named_input_bits.len());
558
559    GpioMapping {
560        input_bits,
561        output_bits,
562        clock_domains,
563        named_input_bits,
564    }
565}

567/// Resolve an internal signal name to its output state bit position.
568///
569/// Searches DFF Q output pins (which appear as cell output pins driving named nets)
570/// for names containing `pattern`. Returns the state buffer bit position, or 0xFFFFFFFF.
571fn resolve_signal_pos(
572    aig: &AIG,
573    netlistdb: &NetlistDB,
574    script: &FlattenedScriptV1,
575    pattern: &str,
576) -> u32 {
577    // Strategy: scan ALL cell pins looking for names matching pattern.
578    // DFF Q outputs are cell output pins whose names match the net they drive.
579    for cellid in 0..netlistdb.num_cells {
580        for pinid in netlistdb.cell2pin.iter_set(cellid) {

582            if !pin_name.contains(pattern) {

584            }
585            // Only care about output pins (Q of DFFs) — check for output_map presence
586            let aigpin_iv = aig.pin2aigpin_iv[pinid];
587            if aigpin_iv == usize::MAX || aigpin_iv <= 1 {
588                continue;
589            }
590            if let Some(&pos) = script.output_map.get(&aigpin_iv) {
591                return pos;
592            }
593            if let Some(&pos) = script.output_map.get(&(aigpin_iv ^ 1)) {
594                return pos;
595            }
596        }
597    }
598    0xFFFFFFFF
599}

658/// Per-tick edge activity for each clock domain.
659#[derive(Clone, Debug, PartialEq, Eq, Hash)]
660struct TickEdges {
661    /// For each domain index: (has_falling_edge, has_rising_edge).
662    domain_edges: Vec<(bool, bool)>,
663}
664
665/// Schedules clock edges across multiple domains at GCD granularity.
666///
667/// For N clock domains with half-periods H1, H2, ..., the GCD tick is
668/// `gcd(H1, H2, ...)` and the schedule repeats every `lcm(H1, H2, ...) / gcd` ticks.
669/// Each tick in the schedule records which domains have falling/rising edges.
670struct MultiClockScheduler {
671    /// GCD of all half-periods in picoseconds. One scheduler edge advances
672    /// simulated time by `gcd_ps`.
673    gcd_ps: u64,
674    /// One entry per GCD tick in the LCM cycle.
675    schedule: Vec<TickEdges>,
676}
677
678/// Compute GCD of two numbers.
679fn gcd(a: u64, b: u64) -> u64 {
680    if b == 0 {
681        a
682    } else {
683        gcd(b, a % b)
684    }
685}
686
687/// Compute LCM of two numbers.
688fn lcm(a: u64, b: u64) -> u64 {
689    a / gcd(a, b) * b
690}
691
692/// A clock domain's timing parameters for scheduling.
693struct ClockDomainTiming {
694    /// Half-period in picoseconds (period_ps / 2).
695    half_period_ps: u64,
696    /// Phase offset in picoseconds.
697    phase_offset_ps: u64,
698    /// Index into GpioMapping::clock_domains.
699    domain_index: usize,
700}
701
702impl MultiClockScheduler {
703    /// Build a scheduler from clock domain timings.
704    ///
705    /// For a single clock with no phase offset, produces a schedule of length 2
706    /// alternating between falling (posedge_flag=0) and rising (posedge_flag=1)
707    /// edges.
708    fn new(timings: &[ClockDomainTiming]) -> Self {
709        assert!(!timings.is_empty(), "At least one clock domain is required");
710
711        // Compute GCD of all half-periods and phase offsets
712        let mut gcd_ps = timings[0].half_period_ps;
713        for t in &timings[1..] {
714            gcd_ps = gcd(gcd_ps, t.half_period_ps);
715        }
716        // Include phase offsets in GCD (non-zero offsets need finer granularity)
717        for t in timings {
718            if t.phase_offset_ps > 0 {
719                gcd_ps = gcd(gcd_ps, t.phase_offset_ps);
720            }
721        }
722
723        // Compute LCM of all full periods (= 2 * half_period)
724        // We need a full period to capture both falling and rising edges.
725        let mut lcm_ps = timings[0].half_period_ps * 2;
726        for t in &timings[1..] {
727            lcm_ps = lcm(lcm_ps, t.half_period_ps * 2);
728        }
729
730        let schedule_len = (lcm_ps / gcd_ps) as usize;
731        assert!(
732            schedule_len <= 1_000_000,
733            "Multi-clock schedule too large ({} ticks). \
734             Clock periods may not be commensurable at this resolution.",
735            schedule_len
736        );
737
738        let num_domains = timings.iter().map(|t| t.domain_index + 1).max().unwrap_or(0);
739        let mut schedule = Vec::with_capacity(schedule_len);
740
741        for tick in 0..schedule_len {
742            let tick_ps = tick as u64 * gcd_ps;
743            let mut domain_edges = vec![(false, false); num_domains];
744
745            for timing in timings.iter() {
746                let hp = timing.half_period_ps;
747                let offset = timing.phase_offset_ps;
748                // Adjust tick_ps by phase offset
749                if tick_ps < offset {
750                    continue;
751                }
752                let adjusted = tick_ps - offset;
753                if adjusted % hp != 0 {
754                    continue;
755                }
756                let edge_count = adjusted / hp;
757                // Index by domain_index (position in clock_domains) so
758                // build_edge_ops can iterate clock_domains directly.
759                if edge_count % 2 == 0 {
760                    domain_edges[timing.domain_index].0 = true; // falling edge
761                } else {
762                    domain_edges[timing.domain_index].1 = true; // rising edge
763                }
764            }
765
766            schedule.push(TickEdges { domain_edges });
767        }
768
769        clilog::info!(
770            "MultiClockScheduler: gcd={}ps, schedule_len={} ticks, {} domains",
771            gcd_ps,
772            schedule_len,
773            num_domains
774        );
775
776        Self { gcd_ps, schedule }
777    }
778
779    /// Build BitOps for a single scheduler edge.
780    ///
781    /// Handles all clock domains' edges at this tick uniformly: a domain with
782    /// a falling edge sets its clk input low and asserts negedge_flag; a domain
783    /// with a rising edge sets its clk input high and asserts posedge_flag.
784    /// Domains with no edge at this tick leave their clk input untouched and
785    /// deassert both flags.
786    fn build_edge_ops(
787        &self,
788        tick_idx: usize,
789        gpio_map: &GpioMapping,
790        reset_gpio: usize,
791        reset_val: u8,
792        constant_inputs: &HashMap<String, u8>,
793        constant_ports: &HashMap<String, u8>,
794        model_driven_positions: &[u32],
795    ) -> Vec<BitOp> {
796        let tick = &self.schedule[tick_idx % self.schedule.len()];
797        let mut ops = Vec::new();
798
799        // Reset
800        ops.push(BitOp {
801            position: gpio_map.input_bits[&reset_gpio],
802            value: reset_val as u32,
803        });
804
805        // Per-domain clock + edge flags
806        for (i, domain) in gpio_map.clock_domains.iter().enumerate() {
807            let (has_fall, has_rise) = tick.domain_edges[i];
808
809            // Set clock input. Falling and rising at the same scheduler tick
810            // can't both be true for one domain (they're encoded as different
811            // edge_count parities), so the order doesn't matter — but
812            // defensively: rising wins, since posedge eval is what DFFs sample.
813            if has_fall {
814                if let Some(pos) = domain.clock_input_pos {
815                    ops.push(BitOp {
816                        position: pos,
817                        value: 0,
818                    });
819                }
820            }
821            if has_rise {
822                if let Some(pos) = domain.clock_input_pos {
823                    ops.push(BitOp {
824                        position: pos,
825                        value: 1,
826                    });
827                }
828            }
829
830            // Edge flags: posedge for domains rising at this edge, negedge for
831            // falling, both deasserted otherwise.
832            for &pos in &domain.posedge_flag_bits {
833                ops.push(BitOp {
834                    position: pos,
835                    value: if has_rise { 1 } else { 0 },
836                });
837            }
838            for &pos in &domain.negedge_flag_bits {
839                ops.push(BitOp {
840                    position: pos,
841                    value: if has_fall { 1 } else { 0 },
842                });
843            }
844        }
845
846        // Constant inputs
847        for (gpio_str, val) in constant_inputs {
848            if let Ok(gpio_idx) = gpio_str.parse::<usize>() {
849                if let Some(&pos) = gpio_map.input_bits.get(&gpio_idx) {
850                    ops.push(BitOp {
851                        position: pos,
852                        value: *val as u32,
853                    });
854                }
855            }
856        }
857        // Named constant ports
858        for (port_name, val) in constant_ports {
859            if let Some(&pos) = gpio_map.named_input_bits.get(port_name) {
860                ops.push(BitOp {
861                    position: pos,
862                    value: *val as u32,
863                });
864            }
865        }
866
867        // Placeholders for peripheral-model-driven inputs. Initial value 0;
868        // updated each batch in `patch_model_driven_in_ops` to reflect the
869        // current model state.
870        for &pos in model_driven_positions {
871            ops.push(BitOp {
872                position: pos,
873                value: 0,
874            });
875        }
876
877        ops
878    }
879}

883/// Direct AIG evaluator: evaluates all AIG pins by traversing the AND-gate
884/// tree directly, bypassing the boomerang/FlattenedScript entirely.
885/// Returns a map from AIG pin → value (0 or 1).
886///
887/// This is used to diagnose whether the AIG construction is correct
888/// (if direct eval matches FlattenedScript → bug is in AIG; if they differ → bug in flattening).
889fn eval_aig_direct(aig: &AIG, input_state: &[u32], script: &FlattenedScriptV1) -> Vec<u8> {
890    let mut pin_values = vec![0u8; aig.num_aigpins + 1];
891    // pin 0 = constant 0 (Tie0)
892
893    // Initialize all primary inputs from input_state using input_map
894    for (&aigpin, &pos) in &script.input_map {
895        let word = (pos / 32) as usize;
896        let bit = pos & 31;
897        if word < input_state.len() {
898            pin_values[aigpin] = ((input_state[word] >> bit) & 1) as u8;
899        }
900    }
901
902    // Topological traversal using topo_traverse_generic
903    // endpoints=None means traverse ALL AIG pins
904    // is_primary_input = set of pins that are inputs (InputPort, InputClockFlag, DFF Q, SRAM)
905    let primary_input_set: indexmap::IndexSet<usize> = script.input_map.keys().copied().collect();
906    let order = aig.topo_traverse_generic(None, Some(&primary_input_set));
907
908    // Evaluate each AND gate in topological order
909    for &pin in &order {
910        if primary_input_set.contains(&pin) {
911            // Already initialized from input_state
912            continue;
913        }
914        if let DriverType::AndGate(a_iv, b_iv) = aig.drivers[pin] {
915            let a_pin = a_iv >> 1;
916            let a_inv = (a_iv & 1) as u8;
917            let b_pin = b_iv >> 1;
918            let b_inv = (b_iv & 1) as u8;
919            let a_val = if a_pin == 0 {
920                0
921            } else {
922                pin_values[a_pin] ^ a_inv
923            };
924            let b_val = if b_pin == 0 {
925                0
926            } else {
927                pin_values[b_pin] ^ b_inv
928            };
929            pin_values[pin] = a_val & b_val;
930        }
931        // Non-AndGate, non-primary-input pins (shouldn't happen in traversal) stay 0
932    }
933
934    pin_values
935}

937/// Compare direct AIG evaluation with FlattenedScript output at a given tick.
938/// Prints diagnostic information about any mismatches.
939fn compare_aig_vs_flattened(
940    aig: &AIG,
941    input_state: &[u32],
942    output_state: &[u32],
943    script: &FlattenedScriptV1,
944    state_size: usize,
945    tick: usize,
946    netlistdb: Option<&NetlistDB>,
947) {
948    let pin_values = eval_aig_direct(aig, input_state, script);
949
950    // Compare DFF D inputs: direct AIG eval vs FlattenedScript output
951    let mut dff_mismatch_count = 0;
952    let mut dff_match_count = 0;
953    let mut dff_en_active_count = 0;
954    let mut dff_d_ones_direct = 0;
955    let mut dff_d_ones_flat = 0;
956    let mut first_mismatches: Vec<String> = Vec::new();
957
958    for (_cellid, dff) in aig.dffs.iter() {
959        let d_pin = dff.d_iv >> 1;
960        let d_inv = (dff.d_iv & 1) as u8;
961        let en_pin = dff.en_iv >> 1;
962        let en_inv = (dff.en_iv & 1) as u8;
963
964        // Direct AIG evaluation
965        let d_val_direct = if d_pin == 0 {
966            0
967        } else {
968            pin_values[d_pin] ^ d_inv
969        };
970        let en_val_direct = if en_pin == 0 {
971            0
972        } else {
973            pin_values[en_pin] ^ en_inv
974        };
975
976        if en_val_direct != 0 {
977            dff_en_active_count += 1;
978        }
979        if d_val_direct != 0 {
980            dff_d_ones_direct += 1;
981        }
982
983        // FlattenedScript result: read from output_state
984        if let Some(&pos) = script.output_map.get(&dff.d_iv) {
985            let word = (pos / 32) as usize;
986            let bit = pos & 31;
987            let flat_val = if word < state_size {
988                ((output_state[word] >> bit) & 1) as u8
989            } else {
990                0
991            };
992            if flat_val != 0 {
993                dff_d_ones_flat += 1;
994            }
995
996            if d_val_direct != flat_val {
997                dff_mismatch_count += 1;
998                if first_mismatches.len() < 10 {
999                    first_mismatches.push(format!(
1000                        "DFF d_iv={} pos={} en={}: direct={} flat={} (d_pin={} en_pin={} en_val={})",
1001                        dff.d_iv, pos, dff.en_iv, d_val_direct, flat_val, d_pin, en_pin, en_val_direct
1002                    ));
1003                }
1004            } else {
1005                dff_match_count += 1;
1006            }
1007        }
1008    }
1009
1010    // Also compare primary outputs
1011    let mut po_mismatch_count = 0;
1012    for (&aigpin_iv, &pos) in &script.output_map {
1013        let pin = aigpin_iv >> 1;
1014        let inv = (aigpin_iv & 1) as u8;
1015        let direct_val = if pin == 0 { 0u8 } else { pin_values[pin] ^ inv };
1016
1017        let word = (pos / 32) as usize;
1018        let bit = pos & 31;
1019        let flat_val = if word < state_size {
1020            ((output_state[word] >> bit) & 1) as u8
1021        } else {
1022            0
1023        };
1024
1025        if direct_val != flat_val {
1026            po_mismatch_count += 1;
1027        }
1028    }
1029
1030    eprintln!("=== AIG vs FlattenedScript comparison at tick {} ===", tick);
1031    eprintln!(
1032        "  DFFs: {} match, {} MISMATCH (out of {} total)",
1033        dff_match_count,
1034        dff_mismatch_count,
1035        aig.dffs.len()
1036    );
1037    eprintln!(
1038        "  DFF en_active={}, d_ones_direct={}, d_ones_flat={}",
1039        dff_en_active_count, dff_d_ones_direct, dff_d_ones_flat
1040    );
1041    eprintln!("  All output_map entries: {} mismatches", po_mismatch_count);
1042    for m in &first_mismatches {
1043        eprintln!("  MISMATCH: {}", m);
1044    }
1045
1046    // Also check: how many DFF Q values are 1 in the input state?
1047    let mut q_ones = 0;
1048    let mut q_hash: u64 = 0;
1049    let mut q_one_entries: Vec<(usize, u32, String)> = Vec::new(); // (cellid, pos, name)
1050    for (&cellid, dff) in aig.dffs.iter() {
1051        if let Some(&pos) = script.input_map.get(&dff.q) {
1052            let word = (pos / 32) as usize;
1053            let bit = pos & 31;
1054            if word < input_state.len() && ((input_state[word] >> bit) & 1) != 0 {
1055                q_ones += 1;
1056                let name = if let Some(ndb) = netlistdb {
1057                    format!("{:?}", ndb.cellnames[cellid])
1058                } else {
1059                    format!("cell_{}", cellid)
1060                };
1061                q_one_entries.push((cellid, pos, name));
1062                q_hash = q_hash
1063                    .wrapping_mul(6364136223846793005)
1064                    .wrapping_add(pos as u64);
1065            }
1066        }
1067    }
1068    q_one_entries.sort_by_key(|&(cellid, _, _)| cellid);
1069    eprintln!(
1070        "  DFF Q values = 1 in input_state: {}/{} (hash=0x{:016X})",
1071        q_ones,
1072        aig.dffs.len(),
1073        q_hash
1074    );
1075    if q_ones <= 100 {
1076        for (cellid, pos, name) in &q_one_entries {
1077            eprintln!("  Q=1: cell {} pos={} = {}", cellid, pos, name);
1078        }
1079    }
1080}

1265// ── Free-function ops patchers ─────────────────────────────────────────────
1266//
1267// These mutate per-edge ops between command-buffer completions, through the
1268// backend's `edge_ops_mut` seam (zero-copy over shared GPU memory on Metal).
1269// They take the backend by shared ref and are called sequentially before
1270// `run_edges`, so no long-lived borrow is held — this is the "closure-borrow"
1271// resolution noted in ADR 0017. `&dyn CosimBackend` keeps them backend-neutral
1272// for Phase 1.
1273
1274/// Set the reset input bit's value in every edge's ops.
1275fn patch_reset_in_ops(backend: &mut dyn CosimBackend, reset_pos: u32, reset_val: u8) {
1276    for sched_idx in 0..backend.edges_per_period() {
1277        for op in backend.edge_ops_mut(sched_idx).iter_mut() {
1278            if op.position == reset_pos {
1279                op.value = reset_val as u32;
1280            }
1281        }
1282    }
1283}

1285/// Sync model-driven input bits into every edge's ops (called when a
1286/// peripheral model's state changes).
1287fn patch_model_driven_in_ops(
1288    backend: &mut dyn CosimBackend,
1289    overrides: &crate::sim::models::ModelOverrides,
1290) {
1291    if overrides.is_empty() {
1292        return;
1293    }
1294    for sched_idx in 0..backend.edges_per_period() {
1295        for op in backend.edge_ops_mut(sched_idx).iter_mut() {
1296            if let Some(&val) = overrides.get(&op.position) {
1297                op.value = val as u32;
1298            }
1299        }
1300    }

1303/// Tracks a clock domain whose input pin is driven by a peripheral model
1304/// (e.g. JTAG TCK), so model transitions can be reflected into the schedule's
1305/// pre-computed posedge/negedge flags.
1306struct ModelDrivenClockState {
1307    clock_input_pos: u32,
1308    posedge_flag_positions: Vec<u32>,
1309    negedge_flag_positions: Vec<u32>,
1310    prev_value: u8,
1311}
1312
1313/// After model overrides are applied, patch the edge flags in the current
1314/// tick's ops to reflect the model's actual clock transitions.
1315fn patch_model_clock_edges(
1316    backend: &mut dyn CosimBackend,
1317    clocks: &mut [ModelDrivenClockState],
1318    overrides: &crate::sim::models::ModelOverrides,
1319    sched_offset: usize,
1320) {
1321    if clocks.is_empty() {
1322        return;
1323    }
1324    let sched_idx = sched_offset % backend.edges_per_period();
1325    let ops = backend.edge_ops_mut(sched_idx);
1326
1327    for clock in clocks.iter_mut() {
1328        let new_val = overrides
1329            .get(&clock.clock_input_pos)
1330            .copied()
1331            .unwrap_or(clock.prev_value);
1332        let rising = clock.prev_value == 0 && new_val == 1;
1333        let falling = clock.prev_value == 1 && new_val == 0;
1334        clock.prev_value = new_val;
1335
1336        for op in ops.iter_mut() {
1337            for &pos in &clock.posedge_flag_positions {
1338                if op.position == pos {
1339                    op.value = if rising { 1 } else { 0 };
1340                }
1341            }
1342            for &pos in &clock.negedge_flag_positions {
1343                if op.position == pos {
1344                    op.value = if falling { 1 } else { 0 };
1345                }
1346            }
1347        }
1348    }
1349}

1790    //

1793    // ops (clk + posedge/negedge flags for any domains active at this tick)

1796    // (one falling, one rising); for multi-domain, schedule_len edges per

1799    // edge. Handed to the backend *once* via `init_schedule` (after backend
1800    // construction below); the backend materialises its native per-edge

1822    // Edge-granularity conversion: config fields `reset_cycles` and
1823    // `num_cycles` are user-facing in sys_clk cycles, but internal counters
1824    // (the main loop's `tick` variable, scheduler offsets, UART decoder
1825    // current_cycle) advance per scheduler edge (one GCD tick).
1826    //
1827    // sched_ticks_per_sys_clk_cycle = how many dense scheduler ticks (each
1828    // `gcd_ps` long) fall in one sys_clk period. NOT the number of sys_clk
1829    // *transitions* (which is always 2). It is 2 only when `gcd_ps` equals
1830    // sys_clk's half-period — i.e. single-clock, or multi-clock whose other
1831    // domains' half-periods and phase offsets are all integer multiples of
1832    // it. With non-commensurate periods or phase offsets, `gcd_ps` shrinks
1833    // below the half-period (`MultiClockScheduler::new`, where gcd folds in
1834    // every half-period and offset) and this is >2. It is the user-cycles →
1835    // internal-tick conversion factor (the codebase calls GCD ticks "edges",

1875    // ── Stimulus deposits (backend-agnostic) ─────────────────────────────

1880    // the xprop X-mask seed + SRAM preload, and the flash/IO buffer setup

1886        // Register value-injection (issue #108, ADR 0016). `reg_init` deposits
1887        // a definite value into chosen registers at tick 0 with `$deposit`
1888        // semantics — the seed clears the power-up X-mask, then the design
1889        // drives the register normally. The register sibling of `sram_init`;
1890        // the fix path for X-poisoned unreset CDC launch registers (#102). NOT
1891        // `force`: applied once here, not re-driven each edge.

1893        // A DFF Q is sequential state, not a primary input in the per-edge
1894        // BitOp schedule, so a deposit must survive `state_prep`'s

1897        // the source of truth the first `simulate` reads; the input slot
1898        // covers any pre-copy read.

1906                    // its bare name; a wider one resolves each bit `name[b]` (a
1907                    // register's bits need not be contiguous after synthesis).

1909                        entry.name.clone()
1910                    } else {

1914                        aig, netlistdb, script, &bit_name,
1915                    )

1917                        panic!(
1918                            "reg_init: cannot resolve register '{}' to a DFF state slot \
1919                             (not a register, or name not found in netlist)",
1920                            bit_name
1921                        )
1922                    });
1923                    let val = if b < 64 { ((entry.value >> b) & 1) as u8 } else { 0 };
1924                    // Deposit the value into both slots' value sections.
1925                    set_bit(&mut states[..state_size], pos, val);
1926                    set_bit(&mut states[state_size..2 * state_size], pos, val);
1927                    // Clear the X-mask in both slots so the deposit reads known.
1928                    // The X-mask section starts `rio` words into each slot.
1929                    if script.xprop_enabled {
1930                        clear_bit(&mut states[rio..state_size], pos);
1931                        clear_bit(&mut states[state_size + rio..2 * state_size], pos);
1932                    }
1933                    deposited_bits += 1;
1934                }
1935            }
1936            clilog::info!(
1937                "reg_init: deposited {} register bit(s) from {} config entries (cleared power-up X)",
1938                deposited_bits,
1939                config.reg_init.len()
1940            );
1941        }
1942
1943        // Initialize: set reset active
1944        let reset_val = if config.reset_active_high { 1u8 } else { 0u8 };
1945        set_bit(
1946            &mut states[..state_size],
1947            gpio_map.input_bits[&reset_gpio],
1948            reset_val,
1949        );
1950        clilog::info!(
1951            "Initial state: reset GPIO {} = {} (active)",
1952            reset_gpio,
1953            reset_val
1954        );

1963                    val,
1964                    pos
1965                );

1968            }
1969        }

1975                &mut states[..state_size],

3401                    .iter()

3445// ── CpuBackend (non-gated CPU reference backend, Phase 1 step 5a) ───────────
3446//
3447// The CPU reference backend (ADR 0017 Layer 2): all design state lives in plain
3448// `Vec<u32>`s, and `run_edges` steps the design through
3449// `cpu_reference::simulate_block_v1[_xprop]` — the same partition executor the
3450// `--check-with-cpu` path uses. It is the cosim oracle: byte-identical to the
3451// Metal golden on the logic fixtures (no timing, no SRAM-xprop — both asserted
3452// off in `new`). Phase 1 step 5a implements the LOGIC subset; the UART-TX
3453// decoder (5b) and bus-trace beat extraction (5c) are stubbed (the `drain_*`
3454// methods return empty, `new` returns empty `bus_lanes`).
3455//
3456// Interior mutability: the trait's `run_edges(&self)` and `vcd_snapshot(&self)`
3457// take a shared ref (Metal's buffers are interior-mutable shared memory). The
3458// CpuBackend mirrors that with `UnsafeCell` over its `Vec`s — sound because
3459// cosim dispatch is strictly sequential (no concurrent access to a backend),
3460// and `state_mut(&mut self)` keeps the compile-time exclusivity the seam
3461// promises for the pre-loop stimulus deposits.

3819    fn run_edges(&self, batch: usize, schedule_offset: usize) -> u64 {
3820        let state_size = self.state_size;
3821        let state = self.state_inner_mut();
3822        // SAFETY: sequential dispatch — sram/sram_xmask/vcd_ring are only
3823        // touched here and in &mut-self methods, never concurrently.
3824        let sram = unsafe { &mut *self.sram.get() };
3825        let sram_xmask = unsafe { &mut *self.sram_xmask.get() };
3826        let ring = unsafe { &mut *self.vcd_ring.get() };
3827        // UART-TX decoder state + accumulator (Phase 1 step 5b). The FSM state
3828        // persists across edges/batches (mirrors the GPU `uart_state` buffer);
3829        // completed bytes accumulate until `drain_uart_tx`.
3830        let uart_states = unsafe { &mut *self.uart_states.get() };
3831        let uart_decoded = unsafe { &mut *self.uart_decoded.get() };
3832        // Bus-trace FSM state + accumulator (Phase 1 step 5c). `prev_gate` /
3833        // `current_tick` persist across edges/batches (mirror the GPU
3834        // `BusTraceChannel` header); beats accumulate until `drain_bus_beats`.
3835        let bus_prev_gate = unsafe { &mut *self.bus_prev_gate.get() };
3836        let bus_current_tick = unsafe { &mut *self.bus_current_tick.get() };
3837        let bus_beats = unsafe { &mut *self.bus_beats.get() };
3838
3839        if let Some(r) = ring.as_mut() {
3840            r.clear();
3841        }
3842
3843        let xmask_off = self.xmask_state_offset;
3844        let rio = self.reg_io_state_size;
3845
3846        for e in 0..batch {
3847            let sched_idx = (schedule_offset + e) % self.edges_per_period;
3848
3849            // ── CPU state_prep (mirrors kernel_v1.metal `state_prep`) ──────
3850            // Step 1: copy output slot → input slot.
3851            state.copy_within(state_size..2 * state_size, 0);
3852            // Step 2: apply this edge's BitOps to the input slot; driving a
3853            // bit clears its X-mask (xmask_off != 0 ⇒ xprop on).
3854            for op in &self.schedule[sched_idx] {
3855                let word_idx = (op.position >> 5) as usize;
3856                let bit_mask = 1u32 << (op.position & 31);
3857                if op.value != 0 {
3858                    state[word_idx] |= bit_mask;
3859                } else {
3860                    state[word_idx] &= !bit_mask;
3861                }
3862                if xmask_off != 0 {
3863                    state[xmask_off + word_idx] &= !bit_mask;
3864                }
3865            }
3866            // (No flash_din injection: logic fixtures have no SPI flash. 5a.)
3867
3868            // ── Simulate every partition for this edge ─────────────────────
3869            for stage_i in 0..self.num_major_stages {
3870                for block_i in 0..self.num_blocks {
3871                    let start = self.blocks_start[stage_i * self.num_blocks + block_i];
3872                    let end = self.blocks_start[stage_i * self.num_blocks + block_i + 1];
3873                    if start == end {
3874                        continue;
3875                    }
3876                    let script_slice = &self.blocks_data[start..end];
3877                    let (input_half, output_half) = state.split_at_mut(state_size);
3878                    if self.xprop_enabled {
3879                        // Value + X-mask lanes are packed in the same slot at
3880                        // [0..rio] and [xmask_off..xmask_off+rio]. The xprop
3881                        // executor takes them as separate slices (it indexes
3882                        // xmasks from 0, not from xmask_off).
3883                        let (in_val, in_x) = input_half.split_at(xmask_off);
3884                        let (out_val, out_x) = output_half.split_at_mut(xmask_off);
3885                        crate::sim::cpu_reference::simulate_block_v1_xprop(
3886                            script_slice,
3887                            in_val,
3888                            out_val,
3889                            &in_x[..rio],
3890                            &mut out_x[..rio],
3891                            sram,
3892                            sram_xmask,
3893                            false,
3894                        );
3895                    } else {
3896                        crate::sim::cpu_reference::simulate_block_v1(
3897                            script_slice,
3898                            input_half,
3899                            output_half,
3900                            sram,
3901                            false,
3902                        );
3903                    }
3904                }
3905            }
3906
3907            // ── UART-TX decoder (ported from gpu_io_step, kernel_v1.metal ──
3908            // 1189-1249). Runs AFTER `simulate` — same point the Metal
3909            // `encode_io_step` runs, with the output slot fresh — on EVERY
3910            // edge from tick 0 (including reset edges), `current_cycle` += 1
3911            // per edge, matching the GPU. READ_OUT_BIT reads the OUTPUT slot:
3912            // states[state_size + (pos>>5)] >> (pos&31) & 1; pos==u32::MAX → 0.
3913            for (ui, cfg) in self.uart_configs.iter().enumerate() {
3914                let cycles_per_bit = cfg.cycles_per_bit;
3915                let tx = if cfg.tx_out_pos != u32::MAX {
3916                    (state[state_size + (cfg.tx_out_pos >> 5) as usize]
3917                        >> (cfg.tx_out_pos & 31))
3918                        & 1
3919                } else {
3920                    0
3921                };
3922
3923                let us = &mut uart_states[ui];
3924                let cycle = us.current_cycle;
3925                let mut st = us.state;
3926                let last_tx = us.last_tx;
3927                let mut start_cycle = us.start_cycle;
3928                let mut bits_received = us.bits_received;
3929                let mut value = us.value;
3930
3931                if st == 0 {
3932                    if last_tx == 1 && tx == 0 {
3933                        st = 1;
3934                        start_cycle = cycle;
3935                    }
3936                } else if st == 1 {
3937                    if cycle >= start_cycle + cycles_per_bit / 2 {
3938                        if tx == 0 {
3939                            st = 2;
3940                            start_cycle += cycles_per_bit;
3941                            bits_received = 0;
3942                            value = 0;
3943                        } else {
3944                            st = 0;
3945                        }
3946                    }
3947                } else if st == 2 {
3948                    let bit_center =
3949                        start_cycle + bits_received * cycles_per_bit + cycles_per_bit / 2;
3950                    if cycle >= bit_center {
3951                        value |= tx << bits_received;
3952                        if bits_received >= 7 {
3953                            st = 3;
3954                            start_cycle += 8 * cycles_per_bit;
3955                        } else {
3956                            bits_received += 1;
3957                        }
3958                    }
3959                } else if st == 3 {
3960                    if cycle >= start_cycle + cycles_per_bit / 2 {
3961                        if tx == 1 {
3962                            uart_decoded.push((ui, (value & 0xFF) as u8));
3963                        }
3964                        st = 0;
3965                    }
3966                }
3967
3968                us.state = st;
3969                us.last_tx = tx;
3970                us.start_cycle = start_cycle;
3971                us.bits_received = bits_received;
3972                us.value = value;
3973                us.current_cycle = cycle + 1;
3974            }
3975
3976            // ── Bus transaction trace (APB3, ported from gpu_io_step,
3977            // kernel_v1.metal:1305-1352). Runs AFTER `simulate`, same point as
3978            // the GPU block, reading the OUTPUT slot. Per bus: gate = sel & en &
3979            // rdy (rdy=1 when pready absent); on a rising edge of the gate emit
3980            // one RawBeat stamped with `current_tick`. `prev_gate` /
3981            // `current_tick` persist; `current_tick` += 1 once per edge from 0,
3982            // regardless of whether any bus fired. READ_OUT_BIT reads the OUTPUT
3983            // slot; pos==u32::MAX → 0 (sentinel short-circuit).
3984            if !self.bus_positions.is_empty() {
3985                let btick = *bus_current_tick;
3986                let prev = *bus_prev_gate;
3987                let mut new_gate = 0u32;
3988                let read_out = |pos: u32| -> u32 {
3989                    if pos != u32::MAX {
3990                        (state[state_size + (pos >> 5) as usize] >> (pos & 31)) & 1
3991                    } else {
3992                        0
3993                    }
3994                };
3995                for (b, bp) in self.bus_positions.iter().enumerate() {
3996                    let gate = if bp.protocol_apb3 {
3997                        let sel = read_out(bp.sel_pos);
3998                        let en = read_out(bp.enable_pos);
3999                        let rdy = if bp.ready_pos == 0xFFFFFFFF {
4000                            1
4001                        } else {
4002                            read_out(bp.ready_pos)
4003                        };
4004                        sel & en & rdy
4005                    } else {
4006                        0
4007                    };
4008                    new_gate |= gate << b;
4009
4010                    let rising = gate != 0 && ((prev >> b) & 1) == 0;
4011                    if rising {
4012                        let write = read_out(bp.write_pos);
4013                        let err = read_out(bp.resp_pos);
4014                        let mut addr = 0u32;
4015                        for i in 0..bp.addr_bits.min(BUS_TRACE_MAX_ADR_BITS) {
4016                            addr |= read_out(bp.addr_pos[i]) << i;
4017                        }
4018                        let mut wdata = 0u32;
4019                        let mut rdata = 0u32;
4020                        for i in 0..bp.data_bits.min(BUS_TRACE_MAX_DAT_BITS) {
4021                            wdata |= read_out(bp.wdata_pos[i]) << i;
4022                            rdata |= read_out(bp.rdata_pos[i]) << i;
4023                        }
4024                        // Pack the same flags the GPU writes, then unpack into a
4025                        // RawBeat identically to MetalBackend::drain_bus_beats.
4026                        let flags = (write & 1) | ((err & 1) << 1) | ((b as u32) << 8);
4027                        bus_beats.push(crate::sim::models::bus_trace::RawBeat {
4028                            tick: btick as u64,
4029                            bus_id: (flags >> 8) & 0xFF,
4030                            write: (flags & 1) != 0,
4031                            err: (flags >> 1) & 1 != 0,
4032                            addr: addr as u64,
4033                            wdata: wdata as u64,
4034                            rdata: rdata as u64,
4035                        });
4036                    }
4037                }
4038                *bus_prev_gate = new_gate;
4039                *bus_current_tick = btick + 1;
4040            }
4041
4042            // ── VCD snapshot: full [input | output] slot for this edge ─────
4043            if self.enable_vcd {
4044                if let Some(r) = ring.as_mut() {
4045                    r.push(state.clone());
4046                }
4047            }
4048        }
4049
4050        // CpuBackend runs synchronously; the token is unused (wait is a no-op).
4051        0
4052    }

4055        // Synchronous CPU backend — run_edges already completed.
4056    }
