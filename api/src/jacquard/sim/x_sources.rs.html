2//! Static enumeration of a design's X-sources for `cosim --xprop` debugging.
3//!
4//! Under `cosim --xprop` an X value originates at a known X-source — an
5//! unreset DFF Q (undefined at power-up), an SRAM read port (memory undefined
6//! until written), or a primary input the testbench leaves undriven — and
7//! propagates forward through combinational logic. Finding *why* a given
8//! signal reads X is otherwise a manual trace→guess→re-run loop.
9//!
10//! This module computes the X-source set statically from the netlist (+ the
11//! cosim config, for the undriven-input class) so the `jacquard xsources`
12//! subcommand can dump it as JSON. `netlist-graph xroots` then intersects a
13//! signal's backward cone with this set to answer "which X-sources make
14//! signal S read X?" — see issue #98 and `docs/x-debugging.md`.
15//!
16//! Everything here is plain CPU logic (no GPU/Metal feature): X-sources are a
17//! static property of the netlist + config, independent of simulation.
18

27/// Schema version of the emitted manifest. Additive-only: new fields and new
28/// `XSourceKind` variants may be added without a major bump.
29///
30/// 1.1: added the optional per-source `src` (RTL `file:line` provenance, WS-B).
31pub const X_SOURCE_SCHEMA_VERSION: &str = "1.1";
32
33/// Classification of an X-source.
34#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
35#[serde(rename_all = "kebab-case")]
36pub enum XSourceKind {
37    /// DFF Q with no reset/set pin connected — X at power-up and never reset,
38    /// so a persistent root unless its data cone resolves to a known value.
39    UnresetDff,
40    /// DFF Q *with* a reset/set pin — X at power-up but defined once reset
41    /// asserts. Reported so a backward query can see it, but it is usually not
42    /// the culprit when reset has been applied.
43    ResetDff,
44    /// SRAM read-data port — X until the addressed cell has been written.
45    SramRead,
46    /// Primary input the cosim config leaves undriven — no clock/reset/
47    /// constant/peripheral drives it, so it reads X every tick.
48    UndrivenInput,
49}
50
51/// One X-source: the net that carries the X, its classification, and (for
52/// DFF/SRAM) the driving cell instance.
53#[derive(Debug, Clone, Serialize)]
54pub struct XSource {
55    /// Hierarchical net name, in the same convention `--trace-signals`
56    /// resolves and `netlist-graph` matches against.
57    pub net: String,
58    pub kind: XSourceKind,
59    /// Driving cell instance name (DFF/SRAM sources only).
60    #[serde(skip_serializing_if = "Option::is_none")]
61    pub cell: Option<String>,
62    /// RTL source location (`file:line`) of the driving cell, from Yosys
63    /// `(* src *)` provenance (WS-B). Present for DFF/SRAM sources when the
64    /// netlist carried provenance; `None` for undriven inputs (no cell) and for
65    /// `-noattr`/non-provenance netlists. Lets a query report `spiflash.v:88`
66    /// instead of the flattened cell name.
67    #[serde(skip_serializing_if = "Option::is_none")]
68    pub src: Option<String>,
69}
70
71/// The RTL source location of a netlist cell, from its `(* src *)` provenance
72/// (`NetlistDB.cell_src`, WS-B B1), as an owned `String` for serialization.
73fn cell_src_string(netlistdb: &NetlistDB, cell_id: usize) -> Option<String> {
74    netlistdb
75        .cell_src
76        .get(cell_id)
77        .and_then(|o| o.as_ref())
78        .map(|s| s.to_string())
79}

82#[derive(Debug, Clone, Serialize)]
83pub struct XSourceManifest {

85    /// The netlist path the manifest was computed from.

93/// Pin-name leaves that mark a DFF reset or set input. Matched case-
94/// insensitively against the cell's pin port names across sky130 / gf180mcu /
95/// generic libraries.
96const RESET_SET_PIN_LEAVES: &[&str] = &[
97    "reset_b", "set_b", "resetn", "setn", "reset", "set", "clrz", "clr", "clrn", "pre", "pren",
98    "rn", "sn", "rb", "sb", "r", "s",
99];
100
101/// Parse a GPIO index from a port name like `gpio_in[5]` or `gpio_in:5`.
102///
103/// Shared by cosim's GPIO mapping (`build_gpio_mapping`) and `xsources`'
104/// driven-input set so the two stay consistent — see issue #98.
105pub fn parse_gpio_index(pin_name: &str, prefix: &str) -> Option<usize> {
106    // Try "gpio_in[N]" format.
107    if let Some(start) = pin_name.find(&format!("{}[", prefix)) {
108        let after = &pin_name[start + prefix.len() + 1..];
109        if let Some(end) = after.find(']') {
110            return after[..end].parse().ok();
111        }
112    }
113    // Try "gpio_in:N" format.
114    if let Some(start) = pin_name.find(&format!("{}:", prefix)) {
115        let after = &pin_name[start + prefix.len() + 1..];
116        let num_str: String = after.chars().take_while(|c| c.is_ascii_digit()).collect();
117        return num_str.parse().ok();
118    }
119    None
120}

122/// Build aigpin → representative netlistdb net-id. A net's driver pin and all
123/// its load pins share the same aigpin, so any pin on the net resolves the
124/// name; we keep the first seen.
125fn build_aigpin_to_net(netlistdb: &NetlistDB, aig: &AIG) -> HashMap<usize, usize> {
126    let mut map: HashMap<usize, usize> = HashMap::new();
127    for pinid in 0..netlistdb.num_pins {
128        let aigpin = aig.pin2aigpin_iv[pinid] >> 1;
129        let netid = netlistdb.pin2net[pinid];
130        map.entry(aigpin).or_insert(netid);
131    }
132    map
133}

135/// Name an aigpin by its net (falling back to a synthetic label if unmapped).
136fn net_name_of(aigpin: usize, aigpin2net: &HashMap<usize, usize>, netlistdb: &NetlistDB) -> String {
137    match aigpin2net.get(&aigpin) {
138        Some(&netid) => netlistdb.netnames[netid].dbg_fmt_pin(),
139        None => format!("<aigpin {aigpin}>"),
140    }
141}

143/// Name a cell's primary output net (the net driven by its first output pin).
144///
145/// Used to name DFF Q and SRAM read-data sources directly from the netlist,
146/// which is robust where an AIG-pin → net lookup is not: a reset DFF's
147/// `dff.q` aigpin is the post-reset-mux internal node, not the node bound to
148/// the netlist Q pin, so it has no entry in the aigpin→net map.
149fn cell_output_net_name(netlistdb: &NetlistDB, cell_id: usize) -> Option<String> {
150    for pinid in netlistdb.cell2pin.iter_set(cell_id) {
151        if netlistdb.pindirect[pinid] == Direction::O {
152            let netid = netlistdb.pin2net[pinid];
153            return Some(netlistdb.netnames[netid].dbg_fmt_pin());
154        }
155    }
156    None
157}

159/// Whether a DFF cell instance has a reset or set pin wired up.
160fn dff_has_reset(netlistdb: &NetlistDB, cell_id: usize) -> bool {
161    for pinid in netlistdb.cell2pin.iter_set(cell_id) {
162        // pinnames[pinid].1 is the port leaf name (e.g. "RESET_B").
163        let leaf = netlistdb.pinnames[pinid].1.to_ascii_lowercase();
164        if RESET_SET_PIN_LEAVES.contains(&leaf.as_str()) {
165            return true;
166        }
167    }
168    false
169}

171/// The set of GPIO indices and explicit port names the config drives.
172///
173/// Mirrors the driver wiring `cosim` performs: clock(s), reset, constant
174/// inputs/ports, and every peripheral's pins. A primary input outside this set
175/// is left undriven and reads X under `--xprop`. Keep in sync with the cosim
176/// driving sites in `cosim_metal.rs` (reset/constant application, `build_edge_ops`,
177/// `set_flash_din`, and each `PeripheralModel`'s driven pins) — a peripheral
178/// that starts driving an input there must be reflected here, or that input is
179/// mis-reported as an undriven X-source.
180fn driven_gpios_and_ports(config: &TestbenchConfig) -> (HashSet<usize>, HashSet<String>) {
181    let mut gpios: HashSet<usize> = HashSet::new();
182    let mut ports: HashSet<String> = HashSet::new();
183
184    gpios.insert(config.reset_gpio);
185    for clk in config.effective_clocks() {
186        gpios.insert(clk.gpio);
187    }
188    for key in config.constant_inputs.keys() {
189        if let Ok(idx) = key.parse::<usize>() {
190            gpios.insert(idx);
191        }
192    }
193    for name in config.constant_ports.keys() {
194        ports.insert(name.clone());
195    }
196    for flash in config.effective_qspi_memory() {
197        gpios.extend([flash.clk_gpio, flash.csn_gpio]);
198        // `set_flash_din` drives 4 data lanes (d0..d0+3) for dual/quad SPI.
199        gpios.extend((0..4).map(|i| flash.d0_gpio + i));
200    }
201    for uart in config.effective_uarts() {
202        // rx is a design input the model drives; tx is a design output —
203        // harmless to include, it is not a primary input.
204        gpios.insert(uart.rx_gpio);
205        gpios.insert(uart.tx_gpio);
206    }
207    for gpio in &config.gpios {
208        gpios.extend(gpio.pins.iter().copied());
209    }
210    if let Some(jtag) = &config.jtag {
211        gpios.extend([jtag.tck_gpio, jtag.tms_gpio, jtag.tdi_gpio]);
212        if let Some(trst) = jtag.trst_gpio {
213            gpios.insert(trst);
214        }
215    }
216    (gpios, ports)
217}

219/// Compute the X-source manifest for a design.
220///
221/// `config` is required only for the `undriven-input` class; without it those
222/// sources are omitted and `undriven_inputs_classified` is false.
223pub fn compute_x_source_manifest(
224    netlistdb: &NetlistDB,
225    aig: &AIG,
226    config: Option<&TestbenchConfig>,
227    netlist_path: &str,
228) -> XSourceManifest {
229    let aigpin2net = build_aigpin_to_net(netlistdb, aig);
230    let mut x_sources: Vec<XSource> = Vec::new();
231
232    // DFF Q outputs (cell_id → DFF). Name via the netlist Q output pin
233    // (robust for reset DFFs, whose `dff.q` aigpin is the post-mux node).
234    for &cell_id in aig.dffs.keys() {
235        let net = cell_output_net_name(netlistdb, cell_id)
236            .unwrap_or_else(|| format!("<dff cell {cell_id}>"));
237        let kind = if dff_has_reset(netlistdb, cell_id) {
238            XSourceKind::ResetDff
239        } else {
240            XSourceKind::UnresetDff
241        };
242        x_sources.push(XSource {
243            net,
244            kind,
245            cell: Some(netlistdb.cellnames[cell_id].dbg_fmt_hier()),
246            src: cell_src_string(netlistdb, cell_id),
247        });
248    }
249
250    // SRAM read-data ports (cell_id → RAMBlock). Slot 0 = unused, skip.
251    for (&cell_id, sram) in &aig.srams {
252        let cell = netlistdb.cellnames[cell_id].dbg_fmt_hier();
253        for &rd_pin in &sram.port_r_rd_data {
254            if rd_pin == 0 {
255                continue;
256            }
257            x_sources.push(XSource {
258                net: net_name_of(rd_pin, &aigpin2net, netlistdb),
259                kind: XSourceKind::SramRead,
260                cell: Some(cell.clone()),
261                src: cell_src_string(netlistdb, cell_id),
262            });
263        }
264    }
265
266    // Undriven primary inputs — needs the config to know the driven set.
267    let undriven_inputs_classified = config.is_some();
268    if let Some(config) = config {
269        let (driven_gpios, driven_ports) = driven_gpios_and_ports(config);
270        let input_name_to_gpio: HashMap<String, usize> = config
271            .port_mapping
272            .as_ref()
273            .map(|pm| {
274                pm.inputs
275                    .iter()
276                    .filter_map(|(k, v)| k.parse::<usize>().ok().map(|idx| (v.clone(), idx)))
277                    .collect()
278            })
279            .unwrap_or_default();
280
281        for (aigpin, driv) in aig.drivers.iter().enumerate() {
282            let DriverType::InputPort(pinid) = driv else {
283                continue;
284            };
285            let pin_name = netlistdb.pinnames[*pinid].dbg_fmt_pin();
286            let gpio_idx = input_name_to_gpio
287                .get(&pin_name)
288                .copied()
289                .or_else(|| parse_gpio_index(&pin_name, "gpio_in"));
290            let driven = gpio_idx.is_some_and(|idx| driven_gpios.contains(&idx))
291                || driven_ports.contains(&pin_name);
292            if !driven {
293                x_sources.push(XSource {
294                    net: net_name_of(aigpin, &aigpin2net, netlistdb),
295                    kind: XSourceKind::UndrivenInput,
296                    cell: None,
297                    // Undriven inputs are ports, not cells — no `(* src *)`.
298                    src: None,
299                });
300            }
301        }
302    }
303
304    XSourceManifest {
305        schema_version: X_SOURCE_SCHEMA_VERSION.to_string(),
306        netlist: netlist_path.to_string(),
307        undriven_inputs_classified,
308        x_sources,
309    }
310}

319    const DEMO_NETLIST: &str = concat!(

328    fn demo_manifest(with_config: bool) -> XSourceManifest {
329        let (netlistdb, aig) =
330            build_netlist_and_aig(Path::new(DEMO_NETLIST), None, &[], None, None);
331        let config: Option<TestbenchConfig> = if with_config {
332            let f = std::fs::File::open(DEMO_CONFIG).unwrap();
333            Some(serde_json::from_reader(std::io::BufReader::new(f)).unwrap())
334        } else {
335            None
336        };
337        compute_x_source_manifest(&netlistdb, &aig, config.as_ref(), DEMO_NETLIST)
338    }

340    fn kind_of<'a>(m: &'a XSourceManifest, net: &str) -> Option<&'a XSourceKind> {
341        m.x_sources.iter().find(|s| s.net == net).map(|s| &s.kind)
342    }
343
344    /// Look up a source by its driving cell instance. Cell names are stable;
345    /// net names go through netlistdb canonicalisation (assign-merge), so the
346    /// counter flops' Q nets are emitted as `_10_[*]` not `unreset_count[*]`.
347    fn kind_of_cell<'a>(m: &'a XSourceManifest, cell: &str) -> Option<&'a XSourceKind> {
348        m.x_sources
349            .iter()
350            .find(|s| s.cell.as_deref() == Some(cell))
351            .map(|s| &s.kind)
352    }

354    #[test]
355    fn dff_sources_classified_by_reset_connection() {
356        let m = demo_manifest(false);
357        // Self-holding register with no reset → unreset (named output port).
358        assert_eq!(kind_of(&m, "q_unreset"), Some(&XSourceKind::UnresetDff));
359        // The counter flops (cells _30_/_31_/_32_) have no reset either.
360        assert_eq!(kind_of_cell(&m, "_30_"), Some(&XSourceKind::UnresetDff));
361        assert_eq!(kind_of_cell(&m, "_32_"), Some(&XSourceKind::UnresetDff));
362        // DFFSR-based regs carry a reset/set pin → reset-dff.
363        assert_eq!(kind_of(&m, "q_reset"), Some(&XSourceKind::ResetDff));
364        assert_eq!(kind_of(&m, "q_undriven_reg"), Some(&XSourceKind::ResetDff));
365    }

367    #[test]

369        // Reset DFFs must resolve to real net names, not `<aigpin N>` /
370        // `<dff cell N>` fallbacks (regression for the post-reset-mux aigpin).
371        let m = demo_manifest(false);
372        for s in &m.x_sources {
373            assert!(
374                !s.net.starts_with('<'),

382        let m = demo_manifest(false);
383        assert!(!m.undriven_inputs_classified);
384        assert!(
385            !m.x_sources
386                .iter()
387                .any(|s| s.kind == XSourceKind::UndrivenInput),

392    #[test]
393    fn config_classifies_undriven_but_not_driven_inputs() {
394        let m = demo_manifest(true);
395        assert!(m.undriven_inputs_classified);
396        // The genuinely-unconnected input reads X.
397        assert_eq!(
398            kind_of(&m, "unconnected"),
399            Some(&XSourceKind::UndrivenInput)
400        );
401        // clk / rst_n are driven (clock / reset gpio via port_mapping), so
402        // they must NOT be reported as undriven X-sources.
403        assert!(kind_of(&m, "clk").is_none());
404        assert!(kind_of(&m, "rst_n").is_none());
405    }

412    #[test]
413    fn x_source_carries_src_provenance() {
414        // WS-B B3: an annotated netlist's X-source reports its RTL `src`.
415        let (netlistdb, aig) =
416            build_netlist_and_aig(Path::new(ANNOTATED_NETLIST), None, &[], None, None);
417        let m = compute_x_source_manifest(&netlistdb, &aig, None, ANNOTATED_NETLIST);
418
419        let s = m
420            .x_sources
421            .iter()
422            .find(|s| s.cell.as_deref() == Some("reg0"))
423            .expect("no reg0 x-source in manifest");
424        assert_eq!(s.kind, XSourceKind::UnresetDff);
425        assert_eq!(s.src.as_deref(), Some("xprov.v:9.3-9.44"));
426        // ...and it round-trips into the emitted JSON.
427        let json = serde_json::to_string(&m).unwrap();
428        assert!(json.contains("\"src\":\"xprov.v:9.3-9.44\""));
429
430        // A source without provenance omits `src` from the JSON entirely
431        // (skip_serializing_if), so un-annotated `-noattr` manifests are
432        // byte-identical to before this field existed.
433        let bare = XSource {
434            net: "n".to_string(),
435            kind: XSourceKind::UndrivenInput,
436            cell: None,
437            src: None,
438        };
439        assert!(!serde_json::to_string(&bare).unwrap().contains("src"));
440    }
