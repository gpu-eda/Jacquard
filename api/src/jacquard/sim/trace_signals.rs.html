2//! User-controlled signal tracing for in-design observability.
3//!
4//! Resolves a list of hierarchical signal names (one per line, file-
5//! supplied) into AIG pins and registers each into
6//! `aig.extra_observable_names`, piggybacking on the same VCD-emission
7//! path that the GF180MCU `bi_24t` bidir-pad split already uses
8//! (`emit_extra_observables` in `src/sim/vcd_io.rs`). Each traced
9//! signal becomes a primary output, gets a state-buffer slot at
10//! partition time, and lands in the output VCD alongside top-level
11//! IO. Works uniformly for sequential (DFF Q) and combinational nets
12//! — anything that has a name in netlistdb's `netname2id` is
13//! traceable.
14//!
15//! Trace list registration happens at AIG construction time (before
16//! partitioning), so the file must be supplied via the
17//! `--trace-signals` CLI flag rather than a runtime env var.
18//!
19//! File format: one signal per line. Blank lines and lines whose
20//! first non-whitespace character is `#` are skipped. Hierarchical
21//! names use `.` as a separator; bit indices use trailing `[N]`.
22//!
23//! ```text
24//! # JTAG-DM internal state
25//! chip_core.dm.haltreq_q[0]
26//! chip_core.dm.haltreq_q[1]
27//!
28//! # Yosys-internal nets — same syntax works
29//! chip_core.sram_u._00147_
30//! ```
31//!
32//! Unresolved names are warned, not fatal — the rest of the trace
33//! list still registers. A trailing summary line reports how many
34//! signals registered vs. how many were dropped, so a bad name list
35//! surfaces clearly at startup.
36

45/// Parsed shape of a single signal name from the trace file.
46/// `hier` is the cell-instance hierarchy (everything before the leaf
47/// segment); `leaf` is the final segment name; `bit` is an optional
48/// trailing `[N]` bus index.
49#[derive(Debug, Clone, PartialEq, Eq)]
50pub struct ParsedSignalName {
51    pub hier: HierName,
52    pub leaf: CompactString,
53    pub bit: Option<isize>,
54}
55
56#[derive(Debug)]
57pub enum TraceError {
58    Io {
59        path: std::path::PathBuf,
60        err: std::io::Error,
61    },
62    InvalidName {
63        line: usize,
64        raw: String,
65        reason: String,
66    },
67}
68
69impl std::fmt::Display for TraceError {
70    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
71        match self {
72            Self::Io { path, err } => write!(f, "trace file {}: {}", path.display(), err),
73            Self::InvalidName { line, raw, reason } => {
74                write!(f, "trace file line {line}: `{raw}` — {reason}")
75            }
76        }
77    }
78}

82/// Read a trace file, returning the list of non-empty, non-comment
83/// signal name strings (raw, not yet parsed). Caller invokes
84/// `parse_signal_name` to convert each into a `ParsedSignalName`.
85pub fn read_trace_file(path: &Path) -> Result<Vec<String>, TraceError> {
86    let contents = std::fs::read_to_string(path).map_err(|err| TraceError::Io {
87        path: path.to_path_buf(),
88        err,
89    })?;
90    let mut names = Vec::new();
91    for line in contents.lines() {
92        let trimmed = line.trim();
93        if trimmed.is_empty() || trimmed.starts_with('#') {
94            continue;
95        }
96        names.push(trimmed.to_string());
97    }
98    Ok(names)
99}

101/// Parse a signal name into one or more candidate
102/// `ParsedSignalName` shapes for netlistdb lookup. Dotted names are
103/// ambiguous in synthesized netlists: Yosys often emits flattened
104/// designs where the entire hierarchy is collapsed into a single
105/// escaped identifier (`\soc.sram.wb_bus__ack`) with dots as
106/// content rather than separators, while other tools preserve the
107/// structural hierarchy and the dots are real splits. Rather than
108/// guess, try both: candidate-0 is the flat-leaf interpretation
109/// (entire name as leaf, empty hier — matches Yosys-style flattened
110/// nets); candidate-1 is the dot-split interpretation (last segment
111/// as leaf, everything before as hier).
112///
113/// Trailing `[N]` bit index is stripped first and applied to both
114/// candidates. Leading `\` is stripped (Verilog escaped-identifier
115/// syntax); we treat the name without the backslash since netlistdb
116/// stores names canonically.
117///
118/// Examples:
119/// - `"clk"` → one candidate: flat leaf `"clk"`, no hier
120/// - `"top.dm.haltreq_q[0]"` → two candidates:
121///   - flat: leaf `"top.dm.haltreq_q"`, bit `Some(0)` (Yosys-flattened)
122///   - dotted: hier `["top","dm"]`, leaf `"haltreq_q"`, bit `Some(0)`
123/// - `"\soc.bus__addr[3]"` → flat: leaf `"soc.bus__addr"`, bit `Some(3)`
124pub fn parse_signal_name(raw: &str) -> Result<Vec<ParsedSignalName>, String> {
125    let raw = raw.trim();
126    if raw.is_empty() {
127        return Err("empty name".to_string());
128    }
129
130    // Verilog escaped-identifier syntax: leading `\` + trailing
131    // space delimits a name with arbitrary characters. Strip both;
132    // netlistdb stores canonical forms.
133    let raw = raw.strip_prefix('\\').unwrap_or(raw);
134    let raw = raw.trim_end();
135
136    // Split off trailing [N] bit index if present. Brackets only
137    // valid at the very end of the name; intermediate brackets
138    // would be part of a (rare) escaped-identifier and aren't
139    // supported.
140    let (head, bit) = if let Some(open) = raw.rfind('[') {
141        if !raw.ends_with(']') {
142            return Err(format!("unmatched `[` in `{raw}`"));
143        }
144        let idx_str = &raw[open + 1..raw.len() - 1];
145        let bit_val: isize = idx_str
146            .parse()
147            .map_err(|e| format!("bit index `{idx_str}` not an integer: {e}"))?;
148        (&raw[..open], Some(bit_val))
149    } else {
150        (raw, None)
151    };
152
153    if head.is_empty() {
154        return Err("empty leaf segment".to_string());
155    }
156
157    let mut candidates = Vec::new();
158
159    // Candidate 0: flat-leaf with explicit bit index (Yosys-
160    // flattened design with dotted escaped-identifier net names,
161    // where the bus index is a structured field in netlistdb).
162    // Always emit; common for synthesized netlists with real bus
163    // declarations.
164    candidates.push(ParsedSignalName {
165        hier: HierName::empty(),
166        leaf: head.into(),
167        bit,
168    });
169
170    // Candidate 1: flat-leaf where the bracket-index is part of
171    // the leaf name itself (Yosys-flattened design where bus
172    // expansion produced 32 scalar wires named
173    // `\foo.bar[0]`...`\foo.bar[31]`, each registered in
174    // netlistdb as `(empty_hier, "foo.bar[N]", None)`). Emit only
175    // when `bit` was actually stripped — otherwise candidate 0
176    // already covers this shape.
177    if bit.is_some() {
178        candidates.push(ParsedSignalName {
179            hier: HierName::empty(),
180            leaf: raw.into(),
181            bit: None,
182        });
183    }
184
185    // Candidate 2: structural-hierarchy split on `.`. Emit only
186    // when the name contains a dot (otherwise candidate 0 covers
187    // it). Skips empty segments to avoid degenerate matches on
188    // names with leading/trailing/consecutive dots.
189    if head.contains('.') {
190        let mut parts: Vec<&str> = head.split('.').collect();
191        let leaf = parts.pop().unwrap();
192        if !leaf.is_empty() && !parts.iter().any(|p| p.is_empty()) {
193            candidates.push(ParsedSignalName {
194                hier: HierName::from_topdown_hier_iter(parts.iter().copied()),
195                leaf: leaf.into(),
196                bit,
197            });
198        }
199    }
200
201    Ok(candidates)
202}

204/// Look up each parsed signal name in netlistdb, register the
205/// resolved aigpin (iv) into `aig.extra_observable_names` and
206/// `aig.primary_outputs`. Returns `(registered, dropped)`.
207/// Unresolved names emit a `clilog::warn!` and increment `dropped`
208/// but don't abort the registration loop — the rest of the list
209/// still lands. Names whose resolved aigpin is the constant-0/1
210/// (`iv <= 1`) are dropped with a warning since they don't need
211/// runtime observation.
212pub fn register_trace_signals(
213    aig: &mut AIG,
214    netlistdb: &NetlistDB,
215    names: &[String],
216) -> (usize, usize) {
217    let mut registered = 0usize;
218    let mut dropped = 0usize;
219
220    for raw in names {
221        let candidates = match parse_signal_name(raw) {
222            Ok(c) => c,
223            Err(e) => {
224                clilog::warn!("--trace-signals: `{raw}` — parse error: {e}");
225                dropped += 1;
226                continue;
227            }
228        };
229
230        // Try each candidate against netname2id; the first hit
231        // wins. Both flat-leaf and dotted-hierarchy shapes are
232        // attempted so the same syntax works for Yosys-flattened
233        // and structurally-hierarchical netlists.
234        let netid = candidates.iter().find_map(|c| {
235            let key = (c.hier.clone(), c.leaf.clone(), c.bit);
236            netlistdb.netname2id.get(&key).copied()
237        });
238        let Some(netid) = netid else {
239            clilog::warn!(
240                "--trace-signals: `{raw}` — not found in netlistdb \
241                 (tried {} candidate(s))",
242                candidates.len()
243            );
244            dropped += 1;
245            continue;
246        };
247
248        // Any pin on the net carries the same aigpin iv (all pins
249        // share the net's driver). Take the first.
250        let Some(pinid) = netlistdb.net2pin.iter_set(netid).next() else {
251            clilog::warn!("--trace-signals: `{raw}` — net has no pins");
252            dropped += 1;
253            continue;
254        };
255
256        let iv = aig.pin2aigpin_iv[pinid];
257        if iv == usize::MAX {
258            clilog::warn!(
259                "--trace-signals: `{raw}` — pin {} has no AIG mapping \
260                 (likely a non-logic pin or stripped cone)",
261                pinid
262            );
263            dropped += 1;
264            continue;
265        }
266        if iv <= 1 {
267            clilog::warn!(
268                "--trace-signals: `{raw}` — resolves to constant {} \
269                 (iv={}); skipping runtime observation",
270                if iv == 0 { 0 } else { 1 },
271                iv
272            );
273            dropped += 1;
274            continue;
275        }
276
277        aig.primary_outputs.insert(iv);
278        aig.extra_observable_names
279            .entry(iv)
280            .or_default()
281            .push(raw.clone());
282        // Surface RTL source provenance (WS-B B3) when the netlist carried it:
283        // the traced net's aigpin resolves back to `(* src *)` file:line(s).
284        let src = aig.aigpin_src_locations(iv >> 1, netlistdb);
285        if !src.is_empty() {
286            clilog::info!("--trace-signals: `{raw}` → RTL {}", src.join(", "));
287        }
288        registered += 1;
289    }
290
291    (registered, dropped)
292}

294/// Resolve a hierarchical signal name to its bit position in the GPU
295/// output state buffer, using the same multi-candidate matching as
296/// [`register_trace_signals`] (so Yosys-flattened / scalar-expanded /
297/// structural naming all work) but reading `script.output_map` instead
298/// of registering an observable.
299///
300/// The signal must already be present in `output_map` — i.e. it is a
301/// primary output or was registered as observable before partitioning
302/// (see `DesignArgs::extra_observable_signals`). Returns `None` if the
303/// name can't be parsed, isn't in netlistdb, has no AIG mapping, or
304/// didn't land in the output state (e.g. tied to a constant).
305///
306/// Used by the cosim bus-trace param builder to bind each protocol pin
307/// to a state position the GPU kernel can read.
308/// Parse `name` and resolve it to a netlistdb net id, scanning the
309/// multi-candidate interpretations (`parse_signal_name`) until one hits
310/// `netname2id`. Shared by the state-position resolvers below. Returns `None`
311/// if the name can't be parsed or matches no net.
312fn resolve_net_id(netlistdb: &NetlistDB, name: &str) -> Option<usize> {
313    let candidates = parse_signal_name(name).ok()?;
314    candidates.iter().find_map(|c| {
315        let key = (c.hier.clone(), c.leaf.clone(), c.bit);
316        netlistdb.netname2id.get(&key).copied()
317    })
318}

320pub fn resolve_to_state_pos(
321    aig: &AIG,
322    netlistdb: &NetlistDB,
323    script: &FlattenedScriptV1,
324    name: &str,
325) -> Option<u32> {
326    let netid = resolve_net_id(netlistdb, name)?;
327    let pinid = netlistdb.net2pin.iter_set(netid).next()?;
328    let iv = aig.pin2aigpin_iv[pinid];
329    if iv == usize::MAX {
330        return None;
331    }
332    // The net's driver may be stored under either polarity in the
333    // output map; try the pin's polarity first, then its inversion.
334    if let Some(&pos) = script.output_map.get(&iv) {
335        return Some(pos);
336    }
337    if let Some(&pos) = script.output_map.get(&(iv ^ 1)) {
338        return Some(pos);
339    }
340    None
341}

343/// Resolve a hierarchical register (DFF) name to its bit position in the
344/// **input** state slot — the DFF Q slot a power-up value must be deposited
345/// into for `reg_init` value-injection (issue #108).
346///
347/// Distinct from [`resolve_to_state_pos`], which reads `output_map` (keyed by
348/// DFF *D* inputs) for bus-trace pin binding. A register *name* resolves to
349/// the net driven by the DFF's *Q*, whose aigpin lives in `input_map`.
350/// Returns `None` if the name doesn't resolve to a DFF Q present in the
351/// flattened script (a pure-comb net, a constant, or an unknown name).
352pub fn resolve_to_input_state_pos(
353    aig: &AIG,
354    netlistdb: &NetlistDB,
355    script: &FlattenedScriptV1,
356    name: &str,
357) -> Option<u32> {
358    let netid = resolve_net_id(netlistdb, name)?;
359    // Scan every pin on the net, not just the first: the DFF Q *driver* pin
360    // is the one keyed in `input_map`, but it need not be the net's first pin
361    // — load pins (e.g. adder inputs consuming the register) share the net.
362    // `input_map` is keyed by the **bare** aigpin (`dff.q`, from
363    // `add_aigpin`), whereas `pin2aigpin_iv` is iv-encoded
364    // (`aigpin << 1 | invert`); recover the bare aigpin with `>> 1`, which
365    // also drops the per-pin invert so both polarities collapse to one key.
366    for pinid in netlistdb.net2pin.iter_set(netid) {
367        let iv = aig.pin2aigpin_iv[pinid];
368        if iv == usize::MAX {
369            continue;
370        }
371        if let Some(&pos) = script.input_map.get(&(iv >> 1)) {
372            return Some(pos);
373        }
374    }
375    None
376}
