22/// Parameters for loading a design.
23pub struct DesignArgs {
24    pub netlist_verilog: PathBuf,
25    pub top_module: Option<String>,
26    pub level_split: Vec<usize>,
27    pub num_blocks: usize,
28    pub json_path: Option<PathBuf>,
29    pub sdf: Option<PathBuf>,
30    pub sdf_corner: String,
31    pub sdf_debug: bool,
32    /// Clock period in picoseconds for SDF timing. Defaults to 25000 if not set.
33    pub clock_period_ps: Option<u64>,
34    /// Enable selective X-propagation.
35    pub xprop: bool,
36    /// Treat undriven primary inputs as X sources (#95 phase 3). Set by the
37    /// reactive `cosim` path, where a primary input the testbench leaves
38    /// undriven reads X; its forward cone must be X-capable for that X to
39    /// propagate. OFF for `sim` (inputs always come from the VCD = known), so
40    /// the static path pays no extra X-aware cost. Only meaningful with `xprop`.
41    pub xprop_undriven_inputs: bool,
42    /// Path to Liberty library file for timing data (pre-layout, no SDF needed).
43    pub liberty: Option<PathBuf>,
44    /// Path to a Jacquard timing-IR (.jtir) file. Mutually exclusive with `sdf`.
45    pub timing_ir: Option<PathBuf>,
46    /// Optional `--timing-corner <NAME>`. When set, the IR's corner of
47    /// that name is selected; when unset, corner index 0 (the first
48    /// declared corner) is used.
49    pub timing_corner: Option<String>,
50    /// User-supplied Verilog cell library files. Each is parsed at
51    /// startup via `sverilogparse` and made available to the netlist
52    /// reader through `ChainedPinProvider`. Sibling
53    /// `<file>.cells.toml` manifests are autoloaded for per-cell
54    /// `kind` annotations. See ADR 0010 and
55    /// `docs/plans/declarative-cell-metadata.md`.
56    pub cell_library: Vec<PathBuf>,
57    /// Optional `--trace-signals <PATH>` — file with one
58    /// hierarchical signal name per line. Resolved against
59    /// `netlistdb.netname2id` after AIG construction; each resolved
60    /// aigpin is registered into `aig.extra_observable_names` so the
61    /// existing VCD-emission path surfaces it as a top-level wire.
62    /// See `src/sim/trace_signals.rs` for the file format and
63    /// resolution semantics.
64    pub trace_signals: Option<PathBuf>,
65    /// Optional `--cell-descriptor <PATH>` — a generated cell-model-IR
66    /// descriptor (ADR 0019). When set, GF180 combinational cells the
67    /// descriptor models are built from its pre-decomposed AIG instead of the
68    /// hardcoded vendored `functional.v` path. Absent = today's behaviour.
69    pub cell_descriptor: Option<PathBuf>,
70    /// Optional `--bundled-descriptor <NAME>` — select a build-time-embedded
71    /// cell-model-IR descriptor (ADR 0019 D7) by name (e.g. `gf180mcu_7t`)
72    /// instead of pointing at an on-disk file. `cell_descriptor` (an explicit
73    /// file) takes precedence when both are set. Absent = today's behaviour.
74    pub bundled_descriptor: Option<String>,
75    /// Additional hierarchical signal names to register as observable,
76    /// supplied programmatically rather than from a file. Any feature
77    /// that needs specific nets to have state-buffer slots before
78    /// partitioning should populate this (the bus-trace feature uses it
79    /// to surface a configured bus's pins). Registered through the same
80    /// path as `trace_signals`.
81    pub extra_observable_signals: Vec<String>,
82    /// Optional `--corner <NAME>` — the ADR 0019 D5 PVT corner to read L4
83    /// timing from in the descriptor-driven no-SDF pre-layout timing path.
84    /// Names a corner in the resolved descriptor's corner set; absent uses the
85    /// descriptor's `default_corner`. Distinct from `timing_corner`, which
86    /// selects a corner in a per-design timing IR (ADR 0002).
87    pub corner: Option<String>,
88    /// `--timed` requested pre-layout timing without an SDF/IR. Triggers the
89    /// no-SDF timing path (ADR 0019 D5) to source L4 from the cell-model-IR
90    /// descriptor even when `--liberty` is absent. When false, that path fires
91    /// only if `--liberty` was passed (legacy opt-in).
92    pub timed: bool,
93}

104/// Build just the `NetlistDB` and `AIG` from a netlist — the feature-
105/// independent front half of [`load_design`], without staging, partitioning,
106/// or GPU script generation. Used by `load_design` and by the `jacquard
107/// xsources` subcommand, which needs only the static AIG to enumerate
108/// X-sources (issue #98).
109pub fn build_netlist_and_aig(
110    netlist_verilog: &Path,
111    top_module: Option<&str>,
112    cell_library: &[PathBuf],
113    cell_descriptor: Option<&Path>,
114    bundled_descriptor: Option<&str>,
115) -> (NetlistDB, AIG) {
116    // Detect cell library
117    let lib = detect_library_from_file(netlist_verilog).expect("Failed to read netlist file");
118    clilog::info!("Detected cell library: {}", lib);
119
120    if lib == CellLibrary::Mixed {
121        panic!("Mixed cells from multiple PDKs in netlist not supported");
122    }
123
124    // Load any user-supplied runtime cell libraries (`--cell-library`).
125    // Empty when no flags were passed; in that case the chained
126    // provider degenerates to its built-in fallback.
127    let runtime_lib = if cell_library.is_empty() {
128        RuntimeCellLibrary::default()
129    } else {
130        let loaded = RuntimeCellLibrary::from_files(cell_library)
131            .unwrap_or_else(|e| panic!("loading --cell-library: {e}"));
132        clilog::info!(
133            "Loaded {} cell-library pin entries, {} kind annotations from --cell-library",
134            loaded.pin_entry_count(),
135            loaded.kind_entry_count()
136        );
137        loaded
138    };
139
140    let netlistdb = match lib {
141        CellLibrary::SKY130 => {
142            let provider = ChainedPinProvider {
143                runtime: &runtime_lib,
144                fallback: &SKY130LeafPins,
145            };
146            NetlistDB::from_sverilog_file(netlist_verilog, top_module, &provider)
147                .expect("cannot build netlist")
148        }
149        // GF180MCU (7t5v0 + 9t5v0). Combinational + sequential paths
150        // (DFFs, latches, scan, clock-gating) are all wired through
151        // `AIG::from_netlistdb` → `gf180mcu_postprocess`.
152        CellLibrary::GF180MCU => {
153            let provider = ChainedPinProvider {
154                runtime: &runtime_lib,
155                fallback: &crate::gf180mcu::GF180MCULeafPins,
156            };
157            NetlistDB::from_sverilog_file(netlist_verilog, top_module, &provider)
158                .expect("cannot build netlist")
159        }
160        CellLibrary::AIGPDK | CellLibrary::Mixed => {
161            let provider = ChainedPinProvider {
162                runtime: &runtime_lib,
163                fallback: &AIGPDKLeafPins(),
164            };
165            NetlistDB::from_sverilog_file(netlist_verilog, top_module, &provider)
166                .expect("cannot build netlist")
167        }
168    };
169
170    // Runtime cell library is consulted both for pin tables (above,
171    // via ChainedPinProvider) and for `kind` metadata (here, by
172    // AIG::from_netlistdb_with_cells). For `kind = "ram"` cells
173    // outside the vendored PDKs, AIG construction allocates opaque
174    // RAMBlock slots whose outputs are X-source. See ADR 0010.
175    let cell_lib_ref = if runtime_lib.is_empty() {
176        None
177    } else {
178        Some(&runtime_lib)
179    };
180
181    // ADR 0019: optional cell-model-IR descriptor. Precedence (D8 + C3.2):
182    //   1. explicit `--cell-descriptor <path>` file (incl. proprietary),
183    //   2. explicit `--bundled-descriptor <name>` selector,
184    //   3. auto-match the netlist's cell-name prefixes against the bundled
185    //      descriptors' declared D8 `library.prefixes` (the C3.2 default —
186    //      a 9t netlist now auto-selects the 9t descriptor, fixing #130),
187    //   4. legacy per-PDK path (`None`) when nothing matches — today's
188    //      behaviour, kept so non-GF180 designs stay green (C3.3 cutover).
189    // When supplied, GF180 combinational cells the descriptor models are
190    // spliced from its pre-decomposed AIG instead of the hardcoded
191    // `functional.v` path. The `.cells.toml` hand-override (ADR 0010) is a
192    // separate runtime-library layer (`cell_lib_ref`) and composes orthogonally.
193    let descriptor = crate::bundled_descriptors::resolve(cell_descriptor, bundled_descriptor)
194        .or_else(|| {
195            crate::bundled_descriptors::auto_select(netlistdb.celltypes.iter().map(|s| s.as_str()))
196                .unwrap_or_else(|e| panic!("{e}"))
197        })
198        // ADR 0019 C3.3c: no prefix-keyed descriptor matched → the AIGPDK
199        // default fallback (mirrors `pdk::resolve_library`, where AIGPDK is the
200        // default verdict). Harmless for unknown netlists: only cells whose
201        // type the AIGPDK descriptor models are spliced; the rest fall through
202        // to the legacy path.
203        .or_else(crate::bundled_descriptors::default_fallback);
204    if let Some(ir) = &descriptor {
205        clilog::info!(
206            "Loaded cell-model-IR descriptor '{}' ({} cells)",
207            ir.library.name,
208            ir.cells.len()
209        );
210    }
211    let aig = AIG::from_netlistdb_with_cells_and_descriptor(
212        &netlistdb,
213        cell_lib_ref,
214        descriptor.as_ref(),
215    );
216    (netlistdb, aig)
217}

223/// Overrides and options for the `sim`/`cosim` input classifier
224/// ([`resolve_netlist_input`]).
225pub struct InputDispatch {
226    /// `--rtl`: force the synthesis path regardless of detection.
227    pub force_rtl: bool,
228    /// `--netlist`: force direct-structural loading, skipping detection.
229    pub force_netlist: bool,
230    /// `--emit-synth <path>`: dump the intermediate gate-level netlist.
231    pub emit_synth: Option<PathBuf>,
232    /// Explicit `yosys.wasm` path for synthesis (else env / discovery).
233    pub yosys_wasm: Option<PathBuf>,
234    /// Top module passed to synthesis (from `--top-module`).
235    pub top_module: Option<String>,
236    /// True when the user supplied `--cell-descriptor`, `--cell-library`, or
237    /// `--bundled-descriptor` — i.e. they have already described a non-built-in
238    /// PDK, so unrecognized cells are expected input, not an error.
239    pub has_cell_hint: bool,
240}

242/// Classify a `sim`/`cosim` netlist input (ADR 0021 §1) and return the path to
243/// actually load: the original file for a gate-level netlist, or a freshly
244/// synthesized `.gv` for behavioral RTL.
245///
246/// Dispatch:
247/// - `--netlist` → load `design` directly (skip detection).
248/// - `--rtl` → synthesize `design`, load the result.
249/// - otherwise: attempt a structural parse (`sverilogparse`). If it parses,
250///   the input is gate-level — every instantiated leaf cell is tested against
251///   the built-in stdcell recognizers. All recognized (or the user supplied a
252///   cell hint) → load directly; any unrecognized → error toward
253///   `--cell-descriptor` (it is already a netlist, not synthesized). If the
254///   structural parse fails, the input is behavioral → synthesize; a synthesis
255///   failure surfaces *both* the structural and Yosys diagnostics.
256pub fn resolve_netlist_input(design: &Path, d: &InputDispatch) -> anyhow::Result<PathBuf> {
257    use anyhow::bail;
258
259    if d.force_netlist && d.force_rtl {
260        bail!("--rtl and --netlist are mutually exclusive");
261    }
262    if d.force_netlist {
263        clilog::info!(
264            "{}: --netlist → loading as gate-level directly",
265            design.display()
266        );
267        return Ok(design.to_path_buf());
268    }
269    if d.force_rtl {
270        clilog::info!("{}: --rtl → synthesizing", design.display());
271        return synth_input(design, d);
272    }
273
274    match sverilogparse::SVerilog::parse_file(design) {
275        Ok(sv) => {
276            let unrecognized = unrecognized_cells(&sv);
277            if unrecognized.is_empty() || d.has_cell_hint {
278                clilog::info!(
279                    "{}: gate-level netlist → simulating directly",
280                    design.display()
281                );
282                Ok(design.to_path_buf())
283            } else {
284                bail!(
285                    "{}: gate-level netlist with unrecognized cell type(s): {}. \
286                     This is already a netlist (not behavioral RTL), so it is not \
287                     synthesized. Pass `--cell-descriptor <path>` (ADR 0019) to \
288                     describe the cell library, or `--cell-library <file.v>` for \
289                     pin definitions. Use `--rtl` to force synthesis instead.",
290                    design.display(),
291                    unrecognized.join(", ")
292                );
293            }
294        }
295        Err(parse_err) => {
296            clilog::info!(
297                "{}: not structural Verilog → treating as behavioral RTL",
298                design.display()
299            );
300            synth_input(design, d).map_err(|synth_err| {
301                anyhow::anyhow!(
302                    "{design}: could not load as a gate-level netlist, and \
303                     synthesizing it as behavioral RTL also failed.\n  \
304                     - structural parse error: {parse_err}\n  \
305                     - synthesis error: {synth_err:#}",
306                    design = design.display()
307                )
308            })
309        }
310    }
311}

313/// Instantiated leaf-cell types not matched by any built-in stdcell recognizer.
314/// Submodule instantiations (types that are themselves defined modules in the
315/// same file) are excluded — only leaf cells are classified.
316fn unrecognized_cells(sv: &sverilogparse::SVerilog) -> Vec<String> {
317    use std::collections::BTreeSet;
318    let defined: BTreeSet<&str> = sv.modules.iter().map(|(name, _)| name.as_str()).collect();
319    let mut out: BTreeSet<String> = BTreeSet::new();
320    for (_, m) in &sv.modules {
321        for cell in &m.cells {
322            let ty = cell.macro_name.as_str();
323            if defined.contains(ty) {
324                continue; // hierarchical submodule, not a leaf cell
325            }
326            if !is_recognized_cell(ty) {
327                out.insert(ty.to_string());
328            }
329        }
330    }
331    out.into_iter().collect()
332}

334/// True if `celltype` belongs to a built-in stdcell library (SKY130, GF180MCU,
335/// or AIGPDK) — the families [`detect_library`] dispatches on.
336fn is_recognized_cell(celltype: &str) -> bool {
337    crate::sky130::is_sky130_cell(celltype)
338        || crate::gf180mcu::is_gf180mcu_cell(celltype)
339        || crate::sky130::is_aigpdk_cell(celltype)
340}

342/// Synthesize `design` via the embedded Yosys engine, or (without the `synth`
343/// feature) return an actionable message pointing at a synth-enabled build.
344#[allow(unused_variables)]
345fn synth_input(design: &Path, d: &InputDispatch) -> anyhow::Result<PathBuf> {
346    #[cfg(feature = "synth")]
347    {
348        let opts = crate::synth::SynthOptions {
349            top_module: d.top_module.clone(),
350            yosys_wasm: d.yosys_wasm.clone(),
351            keep_assertions: true,
352            emit_synth: d.emit_synth.clone(),
353        };
354        let out = crate::synth::synthesize(design, &opts)?;
355        clilog::info!(
356            "{}: behavioral RTL → synthesized [YoWASP Yosys, functional QoR] → {}",
357            design.display(),
358            out.display()
359        );
360        Ok(out)
361    }
362    #[cfg(not(feature = "synth"))]
363    {
364        anyhow::bail!(
365            "{}: input looks like behavioral RTL, but this jacquard binary was \
366             built without synthesis support. Rebuild with `--features synth` \
367             (or use a release binary, which bundles it) to synthesize RTL \
368             on-the-fly, or pass a pre-synthesized gate-level netlist. If this \
369             really is a gate-level netlist that failed to parse, fix the syntax \
370             or pass `--netlist` to force direct loading.",
371            design.display()
372        )
373    }
374}

376/// Load a design through the full pipeline: netlist → AIG → staged → script.
377///
378/// Detects cell library (AIGPDK, SKY130, or GF180MCU), loads display info
379/// from JSON, builds the flattened script, and optionally loads SDF timing
380/// data.
381pub fn load_design(args: &DesignArgs) -> LoadedDesign {
382    let (netlistdb, mut aig) = build_netlist_and_aig(
383        &args.netlist_verilog,
384        args.top_module.as_deref(),
385        &args.cell_library,
386        args.cell_descriptor.as_deref(),
387        args.bundled_descriptor.as_deref(),
388    );
389
390    // Register user-supplied trace signals (--trace-signals). Must
391    // run before staging/partitioning so the resolved aigpins land
392    // in primary_outputs and get state-buffer slots. Unresolved
393    // names log warnings but don't abort.
394    if let Some(trace_path) = args.trace_signals.as_ref() {
395        match crate::sim::trace_signals::read_trace_file(trace_path) {
396            Ok(names) => {
397                let (registered, dropped) =
398                    crate::sim::trace_signals::register_trace_signals(&mut aig, &netlistdb, &names);
399                clilog::info!(
400                    "--trace-signals: registered {} signal(s), dropped {} (file: {})",
401                    registered,
402                    dropped,
403                    trace_path.display()
404                );
405            }
406            Err(e) => {
407                clilog::warn!("--trace-signals: {}; skipping", e);
408            }
409        }
410    }
411
412    // Register programmatic extra observables (e.g. bus-trace pins).
413    // Same ordering constraint as --trace-signals: must run before
414    // staging so the resolved aigpins get state-buffer slots.
415    if !args.extra_observable_signals.is_empty() {
416        let (registered, dropped) = crate::sim::trace_signals::register_trace_signals(
417            &mut aig,
418            &netlistdb,
419            &args.extra_observable_signals,
420        );
421        clilog::info!(
422            "bus-trace observables: registered {} signal(s), dropped {}",
423            registered,
424            dropped
425        );
426    }
427
428    // Load display format info from JSON if available
429    let json_path = args
430        .json_path
431        .clone()
432        .unwrap_or_else(|| args.netlist_verilog.with_extension("json"));
433    if json_path.exists() {
434        match extract_display_info_from_json(&json_path) {
435            Ok(display_info) => {
436                if !display_info.is_empty() {
437                    clilog::info!(
438                        "Loaded {} display format strings from {:?}",
439                        display_info.len(),
440                        json_path
441                    );
442                    aig.populate_display_info(&display_info);
443                }
444            }
445            Err(e) => {
446                clilog::warn!("Could not load display info from JSON: {}", e);
447            }
448        }
449    }
450
451    let stageds = build_staged_aigs(&aig, &args.level_split);
452
453    let parts_in_stages: Vec<Vec<Partition>> = generate_partitions(&aig, &stageds, 0);
454    clilog::info!(
455        "# of effective partitions in each stage: {:?}",
456        parts_in_stages
457            .iter()
458            .map(|ps| ps.len())
459            .collect::<Vec<_>>()
460    );
461
462    let mut input_layout = Vec::new();
463    for (i, driv) in aig.drivers.iter().enumerate() {
464        if let DriverType::InputPort(_) | DriverType::InputClockFlag(_, _) = driv {
465            input_layout.push(i);
466        }
467    }
468
469    let staged_refs: Vec<_> = stageds.iter().map(|(_, _, staged)| staged).collect();
470    let parts_refs: Vec<_> = parts_in_stages.iter().map(|ps| ps.as_slice()).collect();
471
472    let mut script = if args.xprop {
473        let (x_capable, stats) = aig.compute_x_capable_pins(args.xprop_undriven_inputs);
474        clilog::info!(
475            "X-propagation: {}/{} pins ({:.1}%) X-capable, {} fixpoint iterations",
476            stats.num_x_capable_pins,
477            stats.total_pins,
478            if stats.total_pins > 0 {
479                stats.num_x_capable_pins as f64 / stats.total_pins as f64 * 100.0
480            } else {
481                0.0
482            },
483            stats.fixpoint_iterations
484        );
485        FlattenedScriptV1::from_with_xprop(
486            &aig,
487            &staged_refs,
488            &parts_refs,
489            args.num_blocks,
490            input_layout,
491            &x_capable,
492        )
493    } else {
494        FlattenedScriptV1::from(
495            &aig,
496            &staged_refs,
497            &parts_refs,
498            args.num_blocks,
499            input_layout,
500        )
501    };
502
503    if args.xprop {
504        let x_parts = script.partition_x_capable.iter().filter(|&&x| x).count();
505        let total_parts = script.partition_x_capable.len();
506        clilog::info!(
507            "X-propagation: {}/{} partitions X-aware",
508            x_parts,
509            total_parts
510        );
511    }
512
513    // Load timing IR if provided (preferred path, per docs/adr/0006-sdf-preprocessing-model.md).
514    if let Some(ref ir_path) = args.timing_ir {
515        load_timing_ir(
516            &mut script,
517            &aig,
518            &netlistdb,
519            ir_path,
520            args.clock_period_ps,
521            args.liberty.as_deref(),
522            args.timing_corner.as_deref(),
523            args.cell_descriptor.as_deref(),
524            args.bundled_descriptor.as_deref(),
525            args.corner.as_deref(),
526            args.sdf_debug,
527        );
528    } else if let Some(ref sdf_path) = args.sdf {
529        // Per ADR 0006 § Amendment: --sdf is processed by subprocessing
530        // `opensta-to-ir` (and through it OpenSTA). This is the shipping
531        // mechanism, not an interim bridge. The hand-rolled `load_sdf`
532        // path is gone; only `cosim_metal.rs` still ships an unrelated
533        // pre-converted-IR-only path.
534        load_sdf_via_opensta_to_ir(
535            &mut script,
536            &aig,
537            &netlistdb,
538            sdf_path,
539            &args.netlist_verilog,
540            args.liberty.as_deref(),
541            args.top_module.as_deref(),
542            args.clock_period_ps,
543            args.timing_corner.as_deref(),
544            args.cell_descriptor.as_deref(),
545            args.bundled_descriptor.as_deref(),
546            args.corner.as_deref(),
547            args.sdf_debug,
548        );
549    }
550
551    // Pre-layout (no-SDF, no-IR) timing. ADR 0019 D5: source per-cell-type L4
552    // from the cell-model-IR descriptor (explicit > bundled > netlist-prefix
553    // auto-select), with `--liberty` as an explicit `.lib` override and AIGPDK
554    // as the default — one implementation via `resolve_timing_library`. The
555    // trigger stays opt-in so functional runs pay no timing cost: fire when
556    // `--liberty` was passed (legacy) or `--timed` requested it (the D5 case —
557    // timing from the embedded descriptor with no `--liberty`).
558    if args.sdf.is_none() && args.timing_ir.is_none() && (args.liberty.is_some() || args.timed) {
559        let lib = crate::liberty_parser::resolve_timing_library(
560            netlistdb.celltypes.iter().map(|s| s.as_str()),
561            args.cell_descriptor.as_deref(),
562            args.bundled_descriptor.as_deref(),
563            args.corner.as_deref(),
564            args.liberty.as_deref(),
565        );
566        let clock_ps = args.clock_period_ps.unwrap_or(25000);
567        clilog::info!(
568            "Loading pre-layout timing '{}' (clock_period={}ps)",
569            lib.name,
570            clock_ps
571        );
572        script.load_timing(&aig, &netlistdb, &lib, clock_ps);
573        script.inject_timing_to_script();
574    }
575
576    // Print script hash
577    use std::collections::hash_map::DefaultHasher;
578    use std::hash::{Hash, Hasher};
579    let mut s = DefaultHasher::new();
580    script.blocks_data.hash(&mut s);
581    println!("Script hash: {}", s.finish());
582
583    LoadedDesign {
584        netlistdb,
585        aig,
586        script,
587        json_path,
588    }

591/// Convert SDF → IR via `opensta-to-ir` (subprocesses OpenSTA), then
592/// consume the IR through `load_timing_from_ir`.
593///
594/// Subprocess wrapper per ADR 0006 § Amendment. Requires Liberty +
595/// Verilog (OpenSTA links the design from these). Per WS-RH.1, missing
596/// or out-of-date OpenSTA — and missing Liberty when `--sdf` was
597/// requested — are hard errors, not silent fall-throughs.
598#[allow(clippy::too_many_arguments)]
599pub fn load_sdf_via_opensta_to_ir(
600    script: &mut FlattenedScriptV1,
601    aig: &AIG,
602    netlistdb: &NetlistDB,
603    sdf_path: &Path,
604    verilog_path: &Path,
605    liberty_path: Option<&Path>,
606    top_module: Option<&str>,
607    clock_period_ps: Option<u64>,
608    corner_name: Option<&str>,
609    cell_descriptor: Option<&Path>,
610    bundled_descriptor: Option<&str>,
611    l4_corner: Option<&str>,
612    debug: bool,
613) {
614    let liberty_path = liberty_path.unwrap_or_else(|| {
615        panic!(
616            "--sdf requires --liberty <PATH>. OpenSTA needs the Liberty library to link the \
617             design. Alternatively, pre-convert the SDF with `opensta-to-ir` and pass the \
618             result via --timing-ir."
619        );
620    });
621
622    let clock_ps = clock_period_ps.unwrap_or(25000);
623
624    let located = opensta_to_ir::opensta::locate_and_check(None).unwrap_or_else(|e| {
625        panic!("--sdf requires OpenSTA: {e}");
626    });
627    if located.version > opensta_to_ir::opensta::MAX_TESTED_OPENSTA_VERSION {
628        clilog::warn!(
629            "Detected OpenSTA v{}, newer than the latest tested version v{}. \
630             Please report any timing discrepancies.",
631            located.version,
632            opensta_to_ir::opensta::MAX_TESTED_OPENSTA_VERSION
633        );
634    }
635
636    let top = top_module.unwrap_or_else(|| {
637        verilog_path
638            .file_stem()
639            .and_then(|s| s.to_str())
640            .unwrap_or("top")
641    });
642
643    let corners = vec![opensta_to_ir::opensta::CornerSpec::single_default(vec![
644        liberty_path.to_path_buf(),
645    ])];
646    let verilog_paths = vec![verilog_path.to_path_buf()];
647    let invocation = opensta_to_ir::opensta::Invocation {
648        corners: &corners,
649        verilog: &verilog_paths,
650        sdf: Some(sdf_path),
651        spef: None,
652        sdc: None,
653        top,
654        generator_version: env!("CARGO_PKG_VERSION"),
655    };
656
657    clilog::info!(
658        "Loading SDF via opensta-to-ir (OpenSTA v{}): {:?} (clock_period={}ps)",
659        located.version,
660        sdf_path,
661        clock_ps
662    );
663
664    let doc = opensta_to_ir::opensta::run(&located.binary, &invocation, false, false)
665        .unwrap_or_else(|e| panic!("opensta-to-ir failed processing {sdf_path:?}: {e}"));
666
667    let (ir_buf, _stats) = opensta_to_ir::builder::build_ir(&doc, env!("CARGO_PKG_VERSION"));
668    let ir_file = crate::sim::timing_ir_loader::TimingIrFile::from_bytes(ir_buf)
669        .unwrap_or_else(|e| panic!("opensta-to-ir produced invalid IR: {e}"));
670
671    // ADR 0019 D5: the per-cell-type fallback for instances the SDF/IR does not
672    // annotate comes from the cell-model-IR descriptor's L4 (explicit > bundled
673    // > netlist-prefix auto-select), with the (OpenSTA-required) `--liberty` as
674    // the override. The per-instance IR remains PRIMARY (ADR 0002); this only
675    // fills cells absent from it.
676    let liberty_fallback = Some(crate::liberty_parser::resolve_timing_library(
677        netlistdb.celltypes.iter().map(|s| s.as_str()),
678        cell_descriptor,
679        bundled_descriptor,
680        l4_corner,
681        Some(liberty_path),
682    ));
683
684    let ir = ir_file.view();
685    let corner_index = crate::flatten::resolve_corner_index(&ir, corner_name)
686        .unwrap_or_else(|e| panic!("--timing-corner: {e}"));
687    script.load_timing_from_ir(
688        aig,
689        netlistdb,
690        &ir,
691        clock_ps,
692        corner_index,
693        liberty_fallback.as_ref(),
694        debug,
695    );
696    script.inject_timing_to_script();
697}

699/// Load a Jacquard timing-IR (.jtir) file into a script.
700///
701/// Mirrors `load_sdf` but consumes the IR via `TimingIrFile`. Optionally
702/// uses a Liberty library as fallback for cells absent from the IR.
703#[allow(clippy::too_many_arguments)]
704pub fn load_timing_ir(
705    script: &mut FlattenedScriptV1,
706    aig: &AIG,
707    netlistdb: &NetlistDB,
708    ir_path: &Path,
709    clock_period_ps: Option<u64>,
710    liberty_fallback_path: Option<&Path>,
711    corner_name: Option<&str>,
712    cell_descriptor: Option<&Path>,
713    bundled_descriptor: Option<&str>,
714    l4_corner: Option<&str>,
715    debug: bool,
716) {
717    let clock_ps = clock_period_ps.unwrap_or(25000);
718    clilog::info!(
719        "Loading timing IR: {:?} (clock_period={}ps)",
720        ir_path,
721        clock_ps
722    );
723
724    let ir_file = match crate::sim::timing_ir_loader::TimingIrFile::from_path(ir_path) {
725        Ok(f) => f,
726        Err(e) => {
727            clilog::warn!("Failed to load timing IR: {}", e);
728            return;
729        }
730    };
731
732    // ADR 0019 D5: cells absent from the timing IR fall back to the cell-model-
733    // IR descriptor's L4 (explicit > bundled > netlist-prefix auto-select), with
734    // `--liberty` as the override. The per-instance IR stays PRIMARY (ADR 0002).
735    let liberty_fallback = Some(crate::liberty_parser::resolve_timing_library(
736        netlistdb.celltypes.iter().map(|s| s.as_str()),
737        cell_descriptor,
738        bundled_descriptor,
739        l4_corner,
740        liberty_fallback_path,
741    ));
742
743    let ir = ir_file.view();
744    let corner_index = crate::flatten::resolve_corner_index(&ir, corner_name)
745        .unwrap_or_else(|e| panic!("--timing-corner: {e}"));
746    script.load_timing_from_ir(
747        aig,
748        netlistdb,
749        &ir,
750        clock_ps,
751        corner_index,
752        liberty_fallback.as_ref(),
753        debug,
754    );
755    script.inject_timing_to_script();
756}

758/// Build timing constraint buffer for GPU-side setup/hold checking.
759///
760/// Returns `Some((clock_ps, constraint_buffer))` if timing is enabled,
761/// where `constraint_buffer` = `[clock_ps, constraints[0], constraints[1], ...]`.
762pub fn build_timing_constraints(script: &FlattenedScriptV1) -> Option<Vec<u32>> {
763    if script.timing_enabled && !script.dff_constraints.is_empty() {
764        let (clock_ps, constraints) = script.build_timing_constraint_buffer();
765        let non_zero = constraints.iter().filter(|&&v| v != 0).count();
766        clilog::info!(
767            "Timing constraints: {} words, {} with DFF constraints, clock_period={}ps",
768            constraints.len(),
769            non_zero,
770            clock_ps
771        );
772        let mut buf = Vec::with_capacity(1 + constraints.len());
773        buf.push(clock_ps);
774        buf.extend_from_slice(&constraints);
775        Some(buf)
776    } else {
777        None
778    }
779}

781/// Invoke the mt-kahypar partitioner.
782fn run_par(hg: &RCHyperGraph, num_parts: usize) -> Vec<Vec<usize>> {
783    clilog::debug!("invoking partitioner (#parts {})", num_parts);
784    // mt-kahypar requires k >= 2, handle k=1 manually
785    if num_parts == 1 {
786        return vec![(0..hg.num_vertices()).collect()];
787    }
788
789    let parts_ids = hg.partition(num_parts);
790    let mut parts = vec![vec![]; num_parts];
791    for (i, part_id) in parts_ids.into_iter().enumerate() {
792        parts[part_id].push(i);
793    }
794    parts
795}

797/// Generate partitions from the AIG and staged AIGs.
798///
799/// Iteratively partitions endpoints using mt-kahypar hypergraph partitioning.
800/// `max_stage_degrad` controls how many degradation layers are allowed
801/// during partition merging (0 = no degradation).
802fn generate_partitions(
803    aig: &AIG,
804    stageds: &[(usize, usize, StagedAIG)],
805    max_stage_degrad: usize,
806) -> Vec<Vec<Partition>> {
807    stageds
808        .iter()
809        .map(|&(l, r, ref staged)| {
810            clilog::info!(
811                "interactive partitioning stage {}-{}",
812                l,
813                match r {
814                    usize::MAX => "max".to_string(),
815                    r => format!("{}", r),
816                }
817            );
818
819            let mut parts_good: Vec<(Vec<usize>, Partition)> = Vec::new();
820            let mut unrealized_endpoints =
821                (0..staged.num_endpoint_groups()).collect::<Vec<_>>();
822            let mut division = 600;
823
824            while !unrealized_endpoints.is_empty() {
825                division = (division / 2).max(1);
826                let num_parts = unrealized_endpoints.len().div_ceil(division);
827                clilog::info!(
828                    "current: {} endpoints, try {} parts",
829                    unrealized_endpoints.len(),
830                    num_parts
831                );
832                let staged_ur = staged.to_endpoint_subset(&unrealized_endpoints);
833                let hg_ur = RCHyperGraph::from_staged_aig(aig, &staged_ur);
834                let mut parts_indices = run_par(&hg_ur, num_parts);
835                for idcs in &mut parts_indices {
836                    for i in idcs {
837                        *i = unrealized_endpoints[*i];
838                    }
839                }
840                let parts_try = parts_indices
841                    .par_iter()
842                    .map(|endpts| Partition::build_one(aig, staged, endpts))
843                    .collect::<Vec<_>>();
844                let mut new_unrealized_endpoints = Vec::new();
845                for (idx, part_opt) in parts_indices.into_iter().zip(parts_try.into_iter()) {
846                    match part_opt {
847                        Some(part) => {
848                            parts_good.push((idx, part));
849                        }
850                        None => {
851                            if idx.len() == 1 {
852                                panic!("A single endpoint still cannot map, you need to increase level cut granularity.");
853                            }
854                            for endpt_i in idx {
855                                new_unrealized_endpoints.push(endpt_i);
856                            }
857                        }
858                    }
859                }
860                new_unrealized_endpoints.sort_unstable();
861                unrealized_endpoints = new_unrealized_endpoints;
862            }
863
864            clilog::info!(
865                "interactive partition completed: {} in total. merging started.",
866                parts_good.len()
867            );
868
869            let (parts_indices_good, prebuilt): (Vec<_>, Vec<_>) =
870                parts_good.into_iter().unzip();
871            let mut effective_parts = process_partitions(
872                aig,
873                staged,
874                parts_indices_good,
875                Some(prebuilt),
876                max_stage_degrad,
877            )
878            .unwrap();
879            // Drop any empty (zero-boomerang-stage) partition before it reaches
880            // the flattener. mt-kahypar can leave a requested part-id with no
881            // vertices when the part count is high relative to the graph's
882            // structure; such a partition realizes no endpoints (no work), but
883            // `FlattenedScriptV1::from` indexes `part.stages[0]` unconditionally
884            // (flatten.rs) and panics with "index out of bounds: len is 0" on it.
885            // Dropping them is correctness-neutral and hardens the partition →
886            // flatten boundary for any high-part-count path (finer partitioning,
887            // large `--num-blocks`).
888            effective_parts.retain(|p| !p.stages.is_empty());
889            clilog::info!("after merging: {} parts.", effective_parts.len());
890            effective_parts
891        })
892        .collect::<Vec<_>>()
893}
