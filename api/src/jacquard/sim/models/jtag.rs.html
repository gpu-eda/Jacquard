3//! JTAG `remote_bitbang` peripheral model — consumes a `remote_bitbang`
4//! byte stream (the protocol OpenOCD uses to drive an external JTAG
5//! bridge) and drives the design's TCK/TMS/TDI/TRST input pins from it.
6//!
7//! Two byte sources share one FSM (see [`JtagSource`]):
8//!
9//! - **Replay** (`--jtag-replay <PATH>`, discussion #77 stage 1):
10//!   deterministic file-replay of a recorded stream. `R` (read-TDO)
11//!   commands are counted but never answered — the playback is
12//!   one-directional.
13//! - **Live** (`--jtag-server <PORT>`, stage 2): an accepted
14//!   `remote_bitbang` TCP client drives the same FSM in lock-step, and
15//!   each `R` is answered with the design's live TDO output sampled at
16//!   `tdo_pos` and written back over the socket as ASCII `'0'`/`'1'`.
17//!
18//! Only the byte feed and the `R` response differ; the TCK/TMS/TDI/TRST
19//! decode, IEEE 1149.1 TCK deferral, and override contribution are
20//! identical, so replay stays byte-for-byte unchanged.
21//!
22//! ## `remote_bitbang` byte alphabet
23//!
24//! Per [OpenOCD's `remote_bitbang.c`][openocd-rb]:
25//!
26//! | Byte           | Meaning                                |
27//! |----------------|----------------------------------------|
28//! | `'0'`..=`'7'`  | Set `(TCK, TMS, TDI)` from low 3 bits  |
29//! | `'r'`          | TRST off, SRST off                     |
30//! | `'s'`          | TRST off, SRST on                      |
31//! | `'t'`          | TRST on,  SRST off                     |
32//! | `'u'`          | TRST on,  SRST on                      |
33//! | `'R'`          | Read TDO (replay: counted; live: answered) |
34//! | `'B'` / `'b'`  | Blink on / off (no-op)                 |
35//! | `'Q'`          | Quit (treated as end-of-stream)        |
36//!
37//! [openocd-rb]: https://github.com/openocd-org/openocd/blob/master/src/jtag/drivers/remote_bitbang.c
38//!
39//! ## Pacing
40//!
41//! OpenOCD has no concept of simulation time. Stream bytes are
42//! consumed at chip-clock pace: each byte's drive values are held
43//! on the design's input pins for `hold_edges` cosim edges before
44//! the next byte is consumed. Default `hold_edges = 4` — chosen
45//! against the "chip-clock ≥ 2× TCK" assumption that holds by
46//! construction in any chipflow design running a debug TAP.
47//! Configurable per-cosim via `--jtag-hold-cycles <N>` for the
48//! corner case.

57/// `active_low: true` (the common case for RV32-Debug TAPs) drives

65/// Accept one `remote_bitbang` client and disable Nagle (single-byte TDO
66/// replies must not wait). Shared by the first connect and reconnect;
67/// callers supply their own log line and decide how to handle the error.
68fn accept_client(listener: &TcpListener) -> std::io::Result<(TcpStream, std::net::SocketAddr)> {
69    let (stream, peer) = listener.accept()?;
70    if let Err(e) = stream.set_nodelay(true) {
71        clilog::warn!("jtag server: could not set TCP_NODELAY: {e}");
72    }
73    Ok((stream, peer))
74}

76/// Where a [`JtagReplayModel`] pulls its `remote_bitbang` bytes from.
77///
78/// The model's FSM (TCK/TMS/TDI/TRST decode, IEEE 1149.1 TCK deferral,
79/// override contribution) is identical regardless of source — only the
80/// byte feed differs. J3 adds a `Live` arm wrapping an accepted
81/// `TcpStream` for the interactive `--jtag-server`; replay stays the
82/// deterministic recorded path.
83enum JtagSource {
84    /// Deterministic replay of a recorded stream (`--jtag-replay`).
85    Replay { bytes: Vec<u8>, cursor: usize },
86    /// Live `remote_bitbang` socket from an accepted client
87    /// (`--jtag-server`). The client (OpenOCD) paces the session, so a
88    /// byte read blocks until it sends one, and `R` (read-TDO) commands
89    /// are answered back over the same stream.
90    Live {
91        /// Listener kept for the lifetime of the model so a `persist`
92        /// session can `accept()` a fresh client after one disconnects.
93        listener: TcpListener,
94        /// The currently-connected client.
95        stream: TcpStream,
96        /// When true, a client disconnect re-`accept()`s the next client
97        /// (debugger restart without restarting the slow cosim) instead
98        /// of ending the session. `--jtag-reconnect`.
99        persist: bool,
100        /// Set once the socket reaches EOF or errors (and `persist` is
101        /// off, or re-accept failed) — `finished()` then lets the sim
102        /// free-run instead of blocking forever.
103        closed: bool,
104        /// Set by a reconnect so the model can re-arm its power-on TRST
105        /// pulse for the new debug session.
106        reconnected: bool,
107        /// Bytes consumed so far (diagnostics; Replay derives this from
108        /// its cursor).
109        consumed: usize,
110    },
111}
112
113impl JtagSource {
114    /// Yield the next protocol byte, or `None` when the source is
115    /// exhausted. Replay returns `None` at end-of-buffer; the live arm
116    /// performs a *blocking* socket read (the connected client paces
117    /// the session) and returns `None` on EOF or a fatal socket error.
118    fn next_byte_blocking(&mut self) -> Option<u8> {
119        match self {
120            JtagSource::Replay { bytes, cursor } => {
121                let b = bytes.get(*cursor).copied();
122                if b.is_some() {
123                    *cursor += 1;
124                }
125                b
126            }
127            JtagSource::Live {
128                listener,
129                stream,
130                persist,
131                closed,
132                reconnected,
133                consumed,
134            } => {
135                if *closed {
136                    return None;
137                }
138                let mut buf = [0u8; 1];
139                loop {
140                    match stream.read(&mut buf) {
141                        Ok(0) => {
142                            // Orderly client disconnect.
143                            if *persist {
144                                clilog::info!(
145                                    "jtag server: client disconnected; waiting \
146                                     for the next remote_bitbang client \
147                                     (--jtag-reconnect)…"
148                                );
149                                match accept_client(listener) {
150                                    Ok((s, peer)) => {
151                                        *stream = s;
152                                        *reconnected = true;
153                                        clilog::info!(
154                                            "jtag server: client reconnected from {peer}"
155                                        );
156                                        continue;
157                                    }
158                                    Err(e) => {
159                                        *closed = true;
160                                        clilog::warn!(
161                                            "jtag server: accept failed on \
162                                             reconnect: {e}; ending session"
163                                        );
164                                        return None;
165                                    }
166                                }
167                            }
168                            *closed = true;
169                            clilog::info!("jtag server: client disconnected (EOF)");
170                            return None;
171                        }
172                        Ok(_) => {
173                            *consumed += 1;
174                            return Some(buf[0]);
175                        }
176                        Err(e) if e.kind() == ErrorKind::Interrupted => continue,
177                        Err(e) => {
178                            *closed = true;
179                            clilog::warn!("jtag server: socket read error: {e}; ending session");
180                            return None;
181                        }
182                    }
183                }
184            }
185        }
186    }
187
188    /// Answer an `R` (read-TDO) command by writing one ASCII bit back to
189    /// the client. No-op for replay (the recorded path never answers).
190    fn write_tdo(&mut self, bit: u8) {
191        if let JtagSource::Live { stream, closed, .. } = self {
192            if *closed {
193                return;
194            }
195            let ascii = if bit != 0 { b'1' } else { b'0' };
196            if let Err(e) = stream.write_all(&[ascii]) {
197                *closed = true;
198                clilog::warn!("jtag server: TDO write failed: {e}; ending session");
199            }
200        }
201    }
202
203    /// True once the source can yield no more bytes.
204    fn is_exhausted(&self) -> bool {
205        match self {
206            JtagSource::Replay { bytes, cursor } => *cursor >= bytes.len(),
207            JtagSource::Live { closed, .. } => *closed,
208        }
209    }
210
211    /// True for a live socket in `--jtag-reconnect` (persist) mode.
212    fn is_persist(&self) -> bool {
213        matches!(self, JtagSource::Live { persist: true, .. })
214    }
215
216    /// Take (and clear) the "a new client just reconnected" flag, so the
217    /// model can re-arm its power-on TRST pulse for the new session.
218    fn take_reconnected(&mut self) -> bool {
219        match self {
220            JtagSource::Live { reconnected, .. } => std::mem::replace(reconnected, false),
221            _ => false,
222        }
223    }
224
225    /// Bytes consumed so far — surfaced in end-of-run diagnostics.
226    fn consumed(&self) -> usize {
227        match self {
228            JtagSource::Replay { cursor, .. } => *cursor,
229            JtagSource::Live { consumed, .. } => *consumed,
230        }
231    }
232}

234/// CPU-side state for one JTAG `remote_bitbang` peripheral.
235pub struct JtagReplayModel {
236    /// Peripheral name (e.g. `jtag_0`).
237    name: String,
238    /// State-buffer position for the design's TCK input.
239    tck_pos: u32,
240    /// State-buffer position for the design's TMS input.
241    tms_pos: u32,
242    /// State-buffer position for the design's TDI input.
243    tdi_pos: u32,
244    /// Optional TRST input. Some chips reset the TAP via a five-cycle
245    /// TMS=1 sequence instead of a dedicated pin.
246    trst: Option<TrstPin>,
247    /// Optional state-buffer position for the design's TDO *output*.
248    /// Sampled on each `R` (read-TDO) command in live-server mode and
249    /// written back to the connected client. `None` for replay (which
250    /// never answers `R`) or designs whose TDO isn't mapped.
251    tdo_pos: Option<u32>,
252    /// Returned by `driven_positions()`. Built once at construction
253    /// since `PeripheralModel::driven_positions` returns `&[u32]`.
254    driven_positions_arr: Vec<u32>,
255    /// Cosim edges between consuming successive stream bytes.
256    hold_edges: u32,
257    /// Byte feed — recorded replay or (J3) a live socket.
258    source: JtagSource,
259    /// Cosim edges spent on the current drive. Resets to 0 when a
260    /// new byte is consumed.
261    edges_held: u32,
262    /// Currently-driven TCK value (0 / 1).
263    tck: u8,
264    /// Currently-driven TMS value (0 / 1).
265    tms: u8,
266    /// Currently-driven TDI value (0 / 1).
267    tdi: u8,
268    /// Currently-driven raw TRST value (0 / 1, pre-polarity). Combined
269    /// with `trst.active_low` when contributing the override.
270    trst_active: bool,
271    /// TCK value deferred by one edge so TMS/TDI settle before TCK
272    /// transitions. IEEE 1149.1 requires TMS/TDI stable at TCK
273    /// rising edge; applying all three atomically races the TAP.
274    pending_tck: Option<u8>,
275    /// Total `'R'` (TDO-read) commands seen in the stream — emitted
276    /// once at end-of-replay so the operator sees that stage 2 would
277    /// have responded to them.
278    tdo_read_requests: u64,
279    /// Total `'s'` / `'t'` / `'u'` (SRST-toggling) commands — flagged
280    /// at end-of-replay since SRST isn't modeled.
281    srst_toggles: u64,
282    /// Set true when the stream's `'Q'` (quit) command is reached.
283    quit: bool,
284    /// Set when the just-consumed byte was an `'R'` (read-TDO) command.
285    /// The `PeripheralModel::step_edge` impl checks this after stepping
286    /// and, in live-server mode, samples TDO from `output_state` and
287    /// writes the ASCII bit back to the client. Cleared each edge.
288    pending_tdo_read: bool,
289    /// Cosim edges elapsed since construction (one per `step_edge`),
290    /// used to time the power-on TRST pulse.
291    edges_elapsed: u64,
292    /// Power-on TRST pulse window `[assert_from, deassert_at)` in edges,
293    /// or `None` to never force TRST. The DTM resets on `negedge
294    /// trst_n`; a live `remote_bitbang` client (OpenOCD with the common
295    /// `reset_config none`) never asserts TRST, so the model injects one
296    /// brief assertion at startup — mirroring the recorded stream's
297    /// leading `u` and a real chip's power-on reset of the TAP. Replay
298    /// leaves this `None` (the recorded stream drives TRST itself).
299    trst_power_on: Option<(u64, u64)>,
300}
301
302impl JtagReplayModel {
303    /// Build a replay model. `name` matches the configured
304    /// peripheral name (typically `jtag_0`). `bytes` is the recorded
305    /// `remote_bitbang` stream — captured via `scripts/capture_bitbang.py`
306    /// or hand-authored for synthetic tests.
307    pub fn new(
308        name: String,
309        tck_pos: u32,
310        tms_pos: u32,
311        tdi_pos: u32,
312        trst: Option<TrstPin>,
313        tdo_pos: Option<u32>,
314        hold_edges: u32,
315        bytes: Vec<u8>,
316    ) -> Self {
317        Self::with_source(
318            name,
319            tck_pos,
320            tms_pos,
321            tdi_pos,
322            trst,
323            tdo_pos,
324            hold_edges,
325            // Replay drives TRST from the recorded stream — no injection.
326            None,
327            JtagSource::Replay { bytes, cursor: 0 },
328        )
329    }
330
331    /// Build a live-server model. Blocks on `listener.accept()` for the
332    /// first `remote_bitbang` client, then drives the design from the
333    /// socket and answers `R` with the design's live TDO. `tdo_pos`
334    /// should be `Some` (resolved from `JtagConfig.tdo_gpio`); `None`
335    /// means `R` always reads `0`. With `persist`, a client disconnect
336    /// re-`accept()`s the next client (debugger restart without restarting
337    /// cosim) instead of ending the session.
338    pub fn new_live(
339        name: String,
340        tck_pos: u32,
341        tms_pos: u32,
342        tdi_pos: u32,
343        trst: Option<TrstPin>,
344        tdo_pos: Option<u32>,
345        hold_edges: u32,
346        listener: std::net::TcpListener,
347        persist: bool,
348    ) -> Self {
349        let (stream, peer) =
350            accept_client(&listener).unwrap_or_else(|e| panic!("jtag server: accept failed: {e}"));
351        clilog::info!("JTAG server `{name}`: client connected from {peer}");
352        // Default power-on TRST pulse: hold TRST deasserted for a few
353        // edges (so the assertion is a clean `negedge`), then assert it
354        // for ~4 bitbang-bytes' worth of edges before releasing to the
355        // client — the same shape as the recorded stream's leading `u…r`.
356        // Only meaningful when a TRST pin is configured.
357        let trst_power_on = trst.map(|_| {
358            let from = hold_edges as u64;
359            let to = from + 4 * hold_edges as u64;
360            (from, to)
361        });
362        // Escape hatch for tuning / disabling without a rebuild:
363        //   JACQUARD_JTAG_TRST_PULSE="off"        → no injected pulse
364        //   JACQUARD_JTAG_TRST_PULSE="<from>,<to>" → custom window (edges)
365        let trst_power_on = match std::env::var("JACQUARD_JTAG_TRST_PULSE") {
366            Ok(v) if v.trim() == "off" => None,
367            Ok(v) => v
368                .split_once(',')
369                .and_then(|(a, b)| Some((a.trim().parse().ok()?, b.trim().parse().ok()?)))
370                .or(trst_power_on),
371            Err(_) => trst_power_on,
372        };
373        Self::with_source(
374            name,
375            tck_pos,
376            tms_pos,
377            tdi_pos,
378            trst,
379            tdo_pos,
380            hold_edges,
381            trst_power_on,
382            JtagSource::Live {
383                listener,
384                stream,
385                persist,
386                closed: false,
387                reconnected: false,
388                consumed: 0,
389            },
390        )
391    }
392
393    /// Shared constructor body for both byte sources.
394    fn with_source(
395        name: String,
396        tck_pos: u32,
397        tms_pos: u32,
398        tdi_pos: u32,
399        trst: Option<TrstPin>,
400        tdo_pos: Option<u32>,
401        hold_edges: u32,
402        trst_power_on: Option<(u64, u64)>,
403        source: JtagSource,
404    ) -> Self {
405        assert!(
406            hold_edges > 0,
407            "jtag {name}: hold_edges must be > 0 (got 0)"
408        );
409        // Only the *input* pins the model drives go in driven_positions;
410        // `tdo_pos` is a design output the model reads, not drives.
411        let mut driven = vec![tck_pos, tms_pos, tdi_pos];
412        if let Some(t) = trst {
413            driven.push(t.position);
414        }
415        Self {
416            name,
417            tck_pos,
418            tms_pos,
419            tdi_pos,
420            trst,
421            tdo_pos,
422            driven_positions_arr: driven,
423            hold_edges,
424            source,
425            // Pre-loaded so the first `step_edge` immediately
426            // consumes the stream's first byte. Subsequent bytes
427            // are spaced `hold_edges` apart.
428            edges_held: hold_edges,
429            // Idle drive at startup: TCK=0, TMS=1 (target Run-Test-Idle
430            // entry), TDI=0, TRST deasserted.
431            tck: 0,
432            tms: 1,
433            tdi: 0,
434            trst_active: false,
435            pending_tck: None,
436            tdo_read_requests: 0,
437            srst_toggles: 0,
438            quit: false,
439            pending_tdo_read: false,
440            edges_elapsed: 0,
441            trst_power_on,
442        }
443    }
444
445    /// True while the injected power-on TRST pulse is asserting TRST.
446    fn power_on_trst_asserted(&self) -> bool {
447        matches!(self.trst_power_on, Some((from, to)) if self.edges_elapsed >= from && self.edges_elapsed < to)
448    }
449
450    /// Consume one byte from the stream, updating the driven values.
451    /// Public for unit testing the parser in isolation.
452    pub fn consume_byte(&mut self, byte: u8) {
453        match byte {
454            // (TCK, TMS, TDI) packed in low 3 bits. Bit assignment per
455            // OpenOCD's remote_bitbang.c: TDI=bit0, TMS=bit1, TCK=bit2.
456            b'0'..=b'7' => {
457                let b = byte - b'0';
458                self.tdi = b & 0b001;
459                self.tms = (b & 0b010) >> 1;
460                self.tck = (b & 0b100) >> 2;
461            }
462            // TRST off (deasserted).
463            b'r' => self.trst_active = false,
464            // TRST off, SRST on — SRST not modeled, count it.
465            b's' => {
466                self.trst_active = false;
467                self.srst_toggles += 1;
468            }
469            // TRST on (asserted), SRST off.
470            b't' => self.trst_active = true,
471            // TRST on, SRST on.
472            b'u' => {
473                self.trst_active = true;
474                self.srst_toggles += 1;
475            }
476            // TDO read. Always counted; in live-server mode the
477            // `PeripheralModel::step_edge` impl sees `pending_tdo_read`
478            // and writes the sampled TDO bit back to the client.
479            // Replay leaves it counted-only (no socket to answer).
480            b'R' => {
481                self.tdo_read_requests += 1;
482                self.pending_tdo_read = true;
483            }
484            // Blink LED commands — no-op.
485            b'B' | b'b' => {}
486            // Quit — this client is done. Whether that ends the *model*
487            // is `finished()`'s call (a `--jtag-reconnect` session keeps
488            // going and waits for the next client).
489            b'Q' => self.quit = true,
490            // Unknown byte — ignore but warn so corrupted streams
491            // surface during capture/replay. Count too noisy bytes
492            // up-front would require a HashMap; one-shot at end-of-
493            // replay is plenty.
494            other => clilog::warn!(
495                "jtag {}: unknown remote_bitbang byte 0x{:02X} ignored",
496                self.name,
497                other
498            ),
499        }
500    }
501
502    /// True after consuming the stream's final `'Q'` byte or running
503    /// off the end. Cosim doesn't gate on this — replay just goes
504    /// idle on its own.
505    pub fn finished(&self) -> bool {
506        // `quit` ends the model only when the session won't be reused;
507        // under `--jtag-reconnect` the client's `Q` just ends *its* turn
508        // and we wait for the next one (only a closed source finishes).
509        (self.quit && !self.source.is_persist()) || self.source.is_exhausted()
510    }
511
512    /// Sample the design's current TDO bit from the GPU output state.
513    ///
514    /// `output_state` is the *output* half of the design state
515    /// (`&backend.state()[state_size..]`); `tdo_pos` indexes it
516    /// directly. Returns `None` when no TDO pin is configured (replay,
517    /// or a design that doesn't map TDO) — callers answering an `R`
518    /// command treat that as a low (`'0'`) read.
519    pub fn sample_tdo(&self, output_state: &[u32]) -> Option<u8> {
520        self.tdo_pos.map(|pos| read_bit(output_state, pos))
521    }
522
523    /// Diagnostics counters surfaced at end-of-run.
524    pub fn diagnostics(&self) -> JtagDiagnostics {
525        JtagDiagnostics {
526            bytes_consumed: self.source.consumed(),
527            tdo_read_requests: self.tdo_read_requests,
528            srst_toggles: self.srst_toggles,
529            quit: self.quit,
530        }
531    }
532
533    /// Advance one cosim edge. Drains the next byte from the stream
534    /// when the current drive has been held for `hold_edges`. Public
535    /// for unit testing; the `PeripheralModel` impl also calls it.
536    ///
537    /// Byte application is split into two phases to satisfy IEEE
538    /// 1149.1's requirement that TMS/TDI are stable before TCK rises:
539    ///
540    ///   Phase 1 (consume edge): apply TMS/TDI immediately, defer TCK.
541    ///   Phase 2 (next edge): apply the deferred TCK value.
542    ///
543    /// This guarantees TMS/TDI have been driven into the state buffer
544    /// for at least one full cosim edge before TCK transitions,
545    /// preventing the TAP from sampling a stale TMS on the rising edge.
546    pub fn step_edge(&mut self) {
547        // Count every edge (including deferral/idle edges) so the
548        // power-on TRST pulse is timed in cosim edges.
549        self.edges_elapsed = self.edges_elapsed.saturating_add(1);
550        if let Some(tck_val) = self.pending_tck.take() {
551            self.tck = tck_val;
552            return;
553        }
554        if self.finished() {
555            return;
556        }
557        if self.edges_held >= self.hold_edges {
558            let Some(byte) = self.source.next_byte_blocking() else {
559                // Source drained between the `finished()` guard and the
560                // read — for the J3 live arm this is a socket EOF mid-
561                // session. Nothing to apply this edge.
562                return;
563            };
564            // A `--jtag-reconnect` re-accept may have happened inside the
565            // read above. Start the new session clean: clear the previous
566            // client's `Q`, and re-arm the power-on TRST pulse (by
567            // restarting the edge clock) for a fresh DTM reset, exactly
568            // like the first connection.
569            if self.source.take_reconnected() {
570                self.quit = false;
571                self.edges_elapsed = 0;
572            }
573            let old_tck = self.tck;
574            self.consume_byte(byte);
575            let new_tck = self.tck;
576            if new_tck != old_tck {
577                self.tck = old_tck;
578                self.pending_tck = Some(new_tck);
579            }
580            self.edges_held = 1;
581        } else {
582            self.edges_held += 1;
583        }
584    }

587/// Counters surfaced when the run finishes — useful for confirming
588/// the replay landed where expected.
589#[derive(Debug, Clone, Copy)]
590pub struct JtagDiagnostics {
591    pub bytes_consumed: usize,
592    pub tdo_read_requests: u64,
593    pub srst_toggles: u64,
594    pub quit: bool,
595}
596
597impl PeripheralModel for JtagReplayModel {
598    fn name(&self) -> &str {
599        &self.name
600    }
601
602    fn driven_positions(&self) -> &[u32] {
603        &self.driven_positions_arr
604    }
605
606    fn apply_action(&mut self, action: &QueuedAction) {
607        // The replay model is byte-stream driven from the file, not
608        // input.json. Any action targeted at the jtag peripheral is
609        // a config mistake worth warning about.
610        warn_unhandled(&format!("jtag `{}`", self.name), action);
611    }
612
613    fn step_edge(
614        &mut self,
615        output_state: &[u32],
616        overrides: &mut ModelOverrides,
617        _emitted: &mut Vec<EmittedEvent>,
618    ) {
619        JtagReplayModel::step_edge(self);
620        // Answer a just-consumed `R` (read-TDO). `output_state` is the
621        // design output produced by the previous edge's `run_edges` —
622        // i.e. the TDO the TAP drove in response to the TCK the client
623        // just clocked. `write_tdo` is a no-op for the replay source, so
624        // replay stays unaffected; only the live socket is written.
625        if self.pending_tdo_read {
626            self.pending_tdo_read = false;
627            let bit = self.sample_tdo(output_state).unwrap_or(0);
628            self.source.write_tdo(bit);
629        }
630        self.contribute_overrides(overrides);
631    }
632
633    fn contribute_overrides(&self, overrides: &mut ModelOverrides) {
634        overrides.insert(self.tck_pos, self.tck);
635        overrides.insert(self.tms_pos, self.tms);
636        overrides.insert(self.tdi_pos, self.tdi);
637        if let Some(trst) = self.trst {
638            // Assert TRST when the client drives it *or* during the
639            // injected power-on pulse (live server only — `trst_power_on`
640            // is `None` for replay).
641            let asserted = self.trst_active || self.power_on_trst_asserted();
642            let driven = if trst.active_low {
643                if asserted {
644                    0
645                } else {
646                    1
647                }
648            } else if asserted {
649                1
650            } else {
651                0
652            };
653            overrides.insert(trst.position, driven);
654        }
655    }
656
657    fn is_active(&self) -> bool {
658        !self.finished()
659    }
660}

955    /// End-to-end live-socket loopback: a real `TcpStream` client drives
956    /// the FSM through `JtagSource::Live`, and an `R` is answered with
957    /// the sampled TDO bit written back over the socket. This is the
958    /// deterministic, GPU-free core of the V1 gate — the full-design
959    /// `--jtag-server` equivalence run lives in CI (jtag-minimal-cosim
960    /// server job).
961    #[test]
962    fn live_source_loops_back_tdo_over_socket() {
963        use std::net::TcpListener;
964        use std::thread;
965        use std::time::Duration;
966
967        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
968        let port = listener.local_addr().unwrap().port();
969
970        // Client: drive zeros (`0`), then request TDO (`R`), then quit
971        // (`Q`). Expect exactly one ASCII response byte for the `R`.
972        let client = thread::spawn(move || {
973            let mut s = TcpStream::connect(("127.0.0.1", port)).unwrap();
974            s.set_read_timeout(Some(Duration::from_secs(5))).unwrap();
975            s.write_all(b"0").unwrap();
976            s.write_all(b"R").unwrap();
977            let mut resp = [0u8; 1];
978            s.read_exact(&mut resp).unwrap();
979            s.write_all(b"Q").unwrap();
980            resp[0]
981        });
982
983        // `new_live` accepts the client thread's connection internally.
984        // Any hang is bounded by the client's 5 s read timeout above: a
985        // missing TDO write makes its `read_exact` time out, which drops
986        // the socket → EOF unblocks the model.
987        let mut m = JtagReplayModel::new_live(
988            "jtag_0".into(),
989            10,
990            11,
991            12,
992            None,
993            Some(33), // TDO at output-state bit 33 (word 1, bit 1)
994            1,
995            listener,
996            false, // persist
997        );
998        // Output state with TDO high.
999        let out = vec![0u32, 1u32 << 1];
1000        let mut ov = ModelOverrides::new();
1001        let mut ev = Vec::new();
1002        // Steps: consume '0', consume 'R' (writes TDO), consume 'Q'.
1003        // Extra steps are no-ops once finished.
1004        for _ in 0..8 {
1005            PeripheralModel::step_edge(&mut m, &out, &mut ov, &mut ev);
1006        }
1007        let got = client.join().unwrap();
1008        assert_eq!(got, b'1', "R must read back TDO=1 from output_state");
1009        assert_eq!(m.diagnostics().tdo_read_requests, 1);
1010        assert!(m.finished(), "client 'Q' ends the session");
1011    }
