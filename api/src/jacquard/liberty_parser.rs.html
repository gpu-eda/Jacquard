3//! Liberty (.lib) **timing** extraction (L4 — per-cell-type
4//! characterization).
5//!
6//! The low-level tokenizer and generic group/attribute parser now live in
7//! the [`liberty_parse`] crate — the single Liberty front-end in the repo
8//! (ADR 0019 D6). This module no longer parses bytes: it parses with
9//! `liberty_parse::parse` and **walks** the resulting `LibertyGroup` tree
10//! (see the private `walk` module) to project it onto the timing types
11//! below. It extracts the scalar timing values used in `aigpdk.lib` and
12//! SKY130-style libraries:
13//! - Cell delays (cell_rise, cell_fall) for combinational cells
14//! - Setup/hold constraints for sequential cells (DFF, DFFSR)
15//! - Clock-to-Q delays for sequential cells
16//! - SRAM timing (read delays, setup/hold)
17//!
18//! The public `TimingLibrary` / `CellTiming` / `PinTiming` / `TimingArc`
19//! API is unchanged; only the parsing internals moved.
20

25/// Timing arc from an input pin to an output pin.
26#[derive(Debug, Clone, Default)]
27pub struct TimingArc {

29    pub related_pin: String,
30    /// Timing type (e.g., "setup_rising", "hold_rising", "rising_edge")

33    pub cell_rise_ps: Option<u64>,
34    /// Cell fall delay in picoseconds
35    pub cell_fall_ps: Option<u64>,
36    /// Rise constraint (for setup/hold) in picoseconds
37    pub rise_constraint_ps: Option<u64>,
38    /// Fall constraint (for setup/hold) in picoseconds
39    pub fall_constraint_ps: Option<u64>,
40}

64impl CellTiming {
65    /// Get the maximum propagation delay through this cell (for combinational cells).
66    /// Returns (rise_delay_ps, fall_delay_ps).
67    pub fn max_combinational_delay(&self) -> (u64, u64) {
68        let mut max_rise = 0u64;
69        let mut max_fall = 0u64;
70
71        for pin in self.pins.values() {
72            if pin.direction == "output" {
73                for arc in &pin.timing_arcs {
74                    if arc.timing_type.is_none()
75                        || arc.timing_type.as_deref() == Some("combinational")
76                    {
77                        if let Some(rise) = arc.cell_rise_ps {
78                            max_rise = max_rise.max(rise);
79                        }
80                        if let Some(fall) = arc.cell_fall_ps {
81                            max_fall = max_fall.max(fall);
82                        }
83                    }
84                }
85            }
86        }
87
88        (max_rise, max_fall)
89    }
90
91    /// Get setup time for data pin relative to clock (for sequential cells).
92    /// Returns (rise_setup_ps, fall_setup_ps) for rising edge of clock.
93    pub fn setup_time(&self, data_pin: &str) -> Option<(u64, u64)> {
94        let pin = self.pins.get(data_pin)?;
95        for arc in &pin.timing_arcs {
96            if arc.timing_type.as_deref() == Some("setup_rising") {
97                return Some((
98                    arc.rise_constraint_ps.unwrap_or(0),
99                    arc.fall_constraint_ps.unwrap_or(0),
100                ));
101            }
102        }
103        None
104    }
105
106    /// Get hold time for data pin relative to clock (for sequential cells).
107    /// Returns (rise_hold_ps, fall_hold_ps) for rising edge of clock.
108    pub fn hold_time(&self, data_pin: &str) -> Option<(u64, u64)> {
109        let pin = self.pins.get(data_pin)?;
110        for arc in &pin.timing_arcs {
111            if arc.timing_type.as_deref() == Some("hold_rising") {
112                return Some((
113                    arc.rise_constraint_ps.unwrap_or(0),
114                    arc.fall_constraint_ps.unwrap_or(0),
115                ));
116            }
117        }
118        None
119    }
120
121    /// Get clock-to-Q delay for output pin (for sequential cells).
122    /// Returns (rise_delay_ps, fall_delay_ps).
123    pub fn clock_to_q(&self, output_pin: &str) -> Option<(u64, u64)> {
124        let pin = self.pins.get(output_pin)?;
125        for arc in &pin.timing_arcs {
126            if arc.timing_type.as_deref() == Some("rising_edge") {
127                return Some((arc.cell_rise_ps.unwrap_or(0), arc.cell_fall_ps.unwrap_or(0)));
128            }
129        }
130        None
131    }
132}

159    /// Project a cell-model-IR descriptor's L4 timing (ADR 0019 D5) onto a
160    /// [`TimingLibrary`], selecting one corner.
161    ///
162    /// This is the descriptor-driven counterpart of [`Self::from_file`]: instead
163    /// of parsing a `.lib` at runtime, the per-cell-type timing is taken from
164    /// the IR's pre-extracted, corner-keyed [`cell_model_ir::CellTiming`] arcs.
165    /// The resulting library exposes the *same* aggregate views the runtime
166    /// consumes (`and_gate_delay` / `dff_timing` / `sram_timing` /
167    /// `max_combinational_delay` / `clock_to_q` / `setup_time` / `hold_time`),
168    /// so [`crate::aig::AIG::load_timing_library`] and the flatten timing path
169    /// need no changes.
170    ///
171    /// `corner_index` selects which within-corner derate triple to read; the
172    /// **typ** value is rounded to integer ps (the legacy `TimingLibrary` is
173    /// integer-ps and the runtime's `u16` raw-ps would round anyway — ADR 0019
174    /// D5 notes the sub-ns difference vs the oracle's `.lib` parse). Picoseconds
175    /// are consumed directly (the IR is already ps; `time_unit` is *not*
176    /// re-applied).
177    pub fn from_cell_model_ir(ir: &cell_model_ir::CellModelIr, corner_index: u32) -> Self {
178        // Pick the TimingValue for the selected corner; fall back to the first
179        // entry if the arc was characterized at a different corner set.
180        let pick = |vals: &[cell_model_ir::TimingValue]| -> Option<u64> {
181            vals.iter()
182                .find(|v| v.corner_index == corner_index)
183                .or_else(|| vals.first())
184                .map(|v| v.typ.round() as u64)
185        };
186
187        let mut lib = TimingLibrary {
188            name: ir.library.name.clone(),
189            time_unit: "1ps".to_string(),
190            cells: IndexMap::new(),
191        };
192
193        for cell in &ir.cells {
194            let Some(timing) = &cell.timing else {
195                continue;
196            };
197            let mut ct = CellTiming {
198                name: cell.cell_type.clone(),
199                pins: IndexMap::new(),
200            };
201
202            for arc in &timing.delays {
203                let timing_type = match arc.kind {
204                    cell_model_ir::DelayKind::Combinational => None,
205                    // Clock→output: surface as `rising_edge` (what
206                    // `clock_to_q` / `sram_timing` look up). The reference
207                    // edge is implied by the cell's clock; negedge cells'
208                    // Q-arc still keys here, matching the legacy lookup.
209                    cell_model_ir::DelayKind::ClockToOutput => Some("rising_edge".to_string()),
210                };
211                let pin = ct
212                    .pins
213                    .entry(arc.to_pin.clone())
214                    .or_insert_with(|| PinTiming {
215                        name: arc.to_pin.clone(),
216                        direction: "output".to_string(),
217                        is_clock: false,
218                        timing_arcs: Vec::new(),
219                    });
220                pin.timing_arcs.push(TimingArc {
221                    related_pin: arc.from_pin.clone(),
222                    timing_type,
223                    cell_rise_ps: pick(&arc.rise),
224                    cell_fall_ps: pick(&arc.fall),
225                    rise_constraint_ps: None,
226                    fall_constraint_ps: None,
227                });
228            }
229
230            for arc in &timing.constraints {
231                // Only setup/hold are consumed; map onto the rising-edge
232                // timing-type strings the aggregate views match. The IR's
233                // `edge` distinguishes rising/falling reference clocks; the
234                // runtime only reads `*_rising`, so falling-edge constraints
235                // (negedge cells) project to `*_falling` and are ignored —
236                // identical to a real `.lib` parse.
237                let edge = match arc.edge {
238                    cell_model_ir::ClockEdge::Rising => "rising",
239                    cell_model_ir::ClockEdge::Falling => "falling",
240                };
241                let timing_type = match arc.kind {
242                    cell_model_ir::ConstraintKind::Setup => Some(format!("setup_{edge}")),
243                    cell_model_ir::ConstraintKind::Hold => Some(format!("hold_{edge}")),
244                    cell_model_ir::ConstraintKind::Recovery => Some(format!("recovery_{edge}")),
245                    cell_model_ir::ConstraintKind::Removal => Some(format!("removal_{edge}")),
246                };
247                let pin = ct
248                    .pins
249                    .entry(arc.data_pin.clone())
250                    .or_insert_with(|| PinTiming {
251                        name: arc.data_pin.clone(),
252                        direction: "input".to_string(),
253                        is_clock: false,
254                        timing_arcs: Vec::new(),
255                    });
256                pin.timing_arcs.push(TimingArc {
257                    related_pin: arc.related_pin.clone(),
258                    timing_type,
259                    cell_rise_ps: None,
260                    cell_fall_ps: None,
261                    rise_constraint_ps: pick(&arc.rise),
262                    fall_constraint_ps: pick(&arc.fall),
263                });
264            }
265
266            lib.cells.insert(cell.cell_type.clone(), ct);
267        }
268
269        lib
270    }

407    /// Parse Liberty content from a string.
408    ///
409    /// Errors if the parser succeeded structurally but extracted zero
410    /// cells — this is the silent-failure mode that issue #2 of the
411    /// timing-correctness review called out, and the WS5 deliverable in
412    /// `docs/plans/phase-0-ir-and-oracle.md`. The structural parser
413    /// already rejects input without a `library` block, so this check
414    /// only fires when the input was real-shaped Liberty that yielded no
415    /// usable cells.
416    ///
417    /// Test fixtures that intentionally provide skeletal Liberty content
418    /// can use [`Self::parse_unchecked`] to bypass the check.
419    pub fn parse(content: &str) -> Result<Self, String> {
420        let lib = Self::parse_unchecked(content)?;
421        if lib.cells.is_empty() {
422            return Err(format!(
423                "Liberty input ({} bytes) parsed structurally but yielded no cells. \
424                 This indicates a parser issue or unrecognised dialect. Investigate \
425                 the input, or call TimingLibrary::parse_unchecked if zero-cell parses \
426                 are intentional here.",
427                content.len()
428            ));
429        }
430        Ok(lib)
431    }
432
433    /// Parse Liberty content without the non-empty-cells sanity check.
434    ///
435    /// For test fixtures that intentionally provide minimal Liberty input,
436    /// or for callers that have already validated the input by other means.
437    /// Production paths should prefer [`Self::parse`].
438    pub fn parse_unchecked(content: &str) -> Result<Self, String> {
439        let root = liberty_parse::parse(content)?;
440        Ok(walk::walk_library(&root))
441    }

492    /// Get SRAM timing information.
493    pub fn sram_timing(&self) -> Option<SRAMTiming> {
494        let cell = self.cells.get("$__RAMGEM_SYNC_")?;
495
496        // Get read data output timing (clock to Q)
497        let read_delay = cell
498            .pins
499            .get("PORT_R_RD_DATA")
500            .and_then(|p| {
501                p.timing_arcs.iter().find_map(|arc| {
502                    if arc.timing_type.as_deref() == Some("rising_edge") {
503                        Some((arc.cell_rise_ps.unwrap_or(0), arc.cell_fall_ps.unwrap_or(0)))
504                    } else {
505                        None
506                    }
507                })
508            })
509            .unwrap_or((0, 0));
510
511        Some(SRAMTiming {
512            read_clk_to_data_rise_ps: read_delay.0,
513            read_clk_to_data_fall_ps: read_delay.1,
514            // Setup/hold for address and data inputs (simplified)
515            addr_setup_ps: 0, // These use falling edge timing which is 0.0001ps
516            addr_hold_ps: 0,
517            write_data_setup_ps: 0,
518            write_data_hold_ps: 0,
519        })
520    }

574/// Tree-walking extraction from the generic Liberty AST.
575///
576/// The low-level tokenizer and group/attribute parser now live in the
577/// `liberty-parse` crate (the single Liberty front-end — ADR 0019 D6).
578/// This module's job is purely to *walk* the resulting [`LibertyGroup`]
579/// tree and project it onto the timing types (`TimingLibrary`,
580/// `CellTiming`, `PinTiming`, `TimingArc`). No timing semantics changed in
581/// the refactor: the same groups/attributes are read, the same scalar
582/// extraction and ps conversion are applied.
583mod walk {
584    use super::{CellTiming, PinTiming, TimingArc, TimingLibrary};
585    use liberty_parse::LibertyGroup;
586
587    /// Parse a floating point value and convert to picoseconds (integer).
588    ///
589    /// The Liberty file uses `time_unit : "1ps"`, so values are already in
590    /// ps; we only round to the nearest integer. Ported verbatim from the
591    /// former `LibertyParser::parse_float_to_ps`.
592    fn parse_float_to_ps(value: &str) -> u64 {
593        let value = value.trim().trim_matches('"');
594        if let Ok(f) = value.parse::<f64>() {
595            // time_unit is 1ps, so no scale factor — just round to integer ps.
596            f.round() as u64
597        } else {
598            0
599        }
600    }
601
602    /// Extract the first scalar from a timing-table group such as
603    /// `cell_rise (scalar) { values ("1.0"); }`.
604    ///
605    /// Mirrors the former parser: find the `values` attribute and take the
606    /// first numeric token, splitting on commas/quotes so a list-valued
607    /// `values ("a, b, c")` yields its first entry. 2-D lookup tables (no
608    /// scalar leading value, or absent `values`) yield `None`, matching the
609    /// old "skip and use default timing" behaviour.
610    fn extract_scalar_ps(table: &LibertyGroup) -> Option<u64> {
611        let values = table.attr("values")?;
612        let first = values.first_string()?;
613        // `first_string` already returns the de-quoted text of the first
614        // value. A list packed in one quoted string ("a, b, c") still needs
615        // splitting on the separators the old parser split on.
616        let token = first
617            .trim_start()
618            .trim_start_matches('"')
619            .split([',', '"', ')'])
620            .next()
621            .unwrap_or("")
622            .trim();
623        if token.is_empty() {
624            None
625        } else {
626            Some(parse_float_to_ps(token))
627        }
628    }

630    /// Walk a `timing () { ... }` group into a [`TimingArc`].
631    fn walk_timing_arc(timing: &LibertyGroup) -> TimingArc {
632        let mut arc = TimingArc::default();
633
634        if let Some(rp) = timing.attr("related_pin").and_then(|a| a.first_string()) {
635            arc.related_pin = rp.trim_matches('"').to_string();
636        }
637        if let Some(tt) = timing.attr("timing_type").and_then(|a| a.first_string()) {
638            arc.timing_type = Some(tt.trim_matches('"').to_string());
639        }
640
641        // Timing tables are nested groups: cell_rise / cell_fall /
642        // rise_constraint / fall_constraint. Use the FIRST of each (the old
643        // parser overwrote on repeats too, so first-wins vs last-wins only
644        // differs for malformed duplicate tables; first is what extraction
645        // would observe walking in order).
646        for table in &timing.groups {
647            let slot = match table.group_type.as_str() {
648                "cell_rise" => &mut arc.cell_rise_ps,
649                "cell_fall" => &mut arc.cell_fall_ps,
650                "rise_constraint" => &mut arc.rise_constraint_ps,
651                "fall_constraint" => &mut arc.fall_constraint_ps,
652                _ => continue,
653            };
654            if slot.is_none() {
655                if let Some(ps) = extract_scalar_ps(table) {
656                    *slot = Some(ps);
657                }
658            }
659        }
660
661        arc
662    }

664    /// Walk a `pin (...) { ... }` or `bus (...) { ... }` group into a
665    /// [`PinTiming`]. Buses are treated like pins for timing purposes
666    /// (their direction + timing arcs), matching the former `parse_bus`.
667    fn walk_pin(pin_group: &LibertyGroup) -> PinTiming {
668        let mut pin = PinTiming {
669            name: pin_group.first_name().unwrap_or("").to_string(),
670            ..Default::default()
671        };
672
673        if let Some(dir) = pin_group.attr("direction").and_then(|a| a.first_string()) {
674            pin.direction = dir.trim_matches('"').to_string();
675        }
676        if let Some(clk) = pin_group.attr("clock").and_then(|a| a.first_string()) {
677            pin.is_clock = clk.trim_matches('"') == "true";
678        }
679
680        for timing in pin_group.groups_of_type("timing") {
681            pin.timing_arcs.push(walk_timing_arc(timing));
682        }
683
684        pin
685    }
686
687    /// Walk a `cell (...) { ... }` group into a [`CellTiming`].
688    fn walk_cell(cell_group: &LibertyGroup) -> CellTiming {
689        let mut cell = CellTiming {
690            name: cell_group.first_name().unwrap_or("").to_string(),
691            ..Default::default()
692        };
693
694        // `pin` and `bus` groups both carry pin-level timing. Walk them in
695        // source order; later definitions overwrite earlier ones by name,
696        // matching the former insert-by-name behaviour.
697        for child in &cell_group.groups {
698            match child.group_type.as_str() {
699                "pin" | "bus" => {
700                    let pin = walk_pin(child);
701                    cell.pins.insert(pin.name.clone(), pin);
702                }
703                _ => {}
704            }
705        }
706
707        cell
708    }
709
710    /// Project a parsed Liberty `library` group onto a [`TimingLibrary`].
711    pub(super) fn walk_library(root: &LibertyGroup) -> TimingLibrary {
712        let mut lib = TimingLibrary {
713            name: root.first_name().unwrap_or("").to_string(),
714            ..Default::default()
715        };
716
717        if let Some(tu) = root.attr("time_unit").and_then(|a| a.first_string()) {
718            lib.time_unit = tu.trim_matches('"').to_string();
719        }
720
721        for cell_group in root.groups_of_type("cell") {
722            let cell = walk_cell(cell_group);
723            lib.cells.insert(cell.name.clone(), cell);
724        }
725
726        lib
727    }

890            CellKind, CellModel, CellTiming as IrCellTiming, ClockEdge, ConstraintArc,

895            prefixes: vec!["demo_".into()],

899                name: "ss".into(),
900                process: "ss".into(),

904            cell_model_ir::Corner {
905                name: "tt".into(),

916            min: v,

928                Pin {
929                    name: "D".into(),
930                    direction: IrDir::Input,
931                },
932                Pin {
933                    name: "Q".into(),
934                    direction: IrDir::Output,
935                },
936            ],
937            logic: None,
938            sequential: None,
939            timing: Some(IrCellTiming {
940                delays: vec![DelayArc {
941                    from_pin: "CLK".into(),
942                    to_pin: "Q".into(),
943                    kind: DelayKind::ClockToOutput,
944                    rise: vec![tv(0, 300.0), tv(1, 150.4)],
945                    fall: vec![tv(0, 280.0), tv(1, 140.6)],

947                constraints: vec![
948                    ConstraintArc {
949                        data_pin: "D".into(),
950                        related_pin: "CLK".into(),
951                        kind: ConstraintKind::Setup,
952                        edge: ClockEdge::Rising,
953                        rise: vec![tv(0, 160.0), tv(1, 80.0)],
